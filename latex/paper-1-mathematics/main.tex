\begin{abstract}
A regular $N$-gon of point vortices on the flat plane is linearly
stable if and only if $N \le 7$.  We extend this classical result
to constant-curvature surfaces via the eigenvalue decomposition
$\lambda_m = C_1(\text{surface}) - m(N{-}m)/2$, which separates
a single surface-dependent coefficient from a universal Casimir
determined by~$\mathbb{Z}_N$.
We prove a sign rule for general pairwise interactions,
extend the Havelock identity to Riemannian surfaces of constant
curvature, and show that the five algebraic-integer thresholds
on the hyperbolic plane are Pell units in quadratic fields,
terminating at~$\xi^*(23) = \varphi^{-2}$.
These thresholds correspond to closed geodesics on the modular
surface with traces $T = 2 + 16/(N{-}7)$.
\end{abstract}


\section{Introduction}
\label{sec:intro}

The stability eigenvalue of a regular $N$-gon of point vortices on
a two-dimensional surface admits a decomposition
\begin{equation}\label{eq:central-decomposition}
  \lambda_m = C_1(\text{surface}) - \frac{m(N-m)}{2},
  \qquad m = 1, \ldots, N-1,
\end{equation}
that separates a single surface-dependent number $C_1$ from a
universal function $f(m,N) = m(N{-}m)/2$ determined entirely by the
representation theory of the cyclic group $\mathbb{Z}_N$.
In words: the stability of any regular polygon depends on one
number~$C_1$ encoding the surface geometry, minus a contribution
$m(N{-}m)/2$ that depends only on the polygon's symmetry.
Every stability result on every constant-curvature surface (flat
plane, sphere, hyperbolic plane) follows from this formula with
one number substituted in.

This decomposition is implicit in
\citet{Havelock1931} and explicit in \citet{BoattoCabral2003}.
What has not been made explicit is the scope of its
consequences.

\paragraph{What this paper proves.}
\begin{enumerate}
  \item \textbf{The sign rule} (Theorem~\ref{thm:sign-rule}):
    for any pairwise interaction $h(r)$ with $h' < 0$ and
    $(rh')' \le 0$, the $N$-gon is a saddle of the unconstrained
    energy with $H''(0) < 0$ along the centered spiral deformation.
    The logarithmic interaction $h = -\ln r$ is the unique
    dilation invariant borderline case (Section~\ref{sec:log-interaction}).
  \item \textbf{The Havelock identity and its metric-independence}
    (Section~\ref{sec:havelock-identity}):
    the $m$-dependent part $f(m,N) = m(N{-}m)/2$ of the Havelock
    eigenvalue is a consequence of $\mathbb{Z}_N$ representation
    theory alone; only the additive constant $C_1$ carries geometric
    information.
  \item \textbf{Stability thresholds on constant-curvature surfaces}
    (Sections~\ref{sec:curvature-thresholds}--\ref{sec:algebraic-structure}):
    the curvature-stability dichotomy
    (Proposition~\ref{prop:curvature-dichotomy}),
    the Morse index formula $\mu(N) = N - 5$ for $N \ge 8$
    (Proposition~\ref{prop:morse-index}), the closed-form derivation
    of $N_{\mathrm{crit}} = 7$ (Theorem~\ref{thm:why-7-algebraic}),
    the monotone spectral flow from flat to hyperbolic geometry,
    and the robustness of $\mu(N)$ under mode-coupling
    perturbations (Remark~\ref{rmk:topological-protection}).
  \item \textbf{Algebraic structure of the stability boundary}
    (Section~\ref{sec:algebraic-structure}):
    the field extension pattern, the Pell equation connection,
    the algebraic-integer classification
    $N \in \{8, 9, 11, 15, 23\}$
    (Proposition~\ref{prop:alg-int}), the golden ratio terminal case
    $\xi^*(23) = \varphi^{-2}$, the continued fraction structure
    (Proposition~\ref{prop:cf-palindromic}),
    the geodesic structural observation
    (Proposition~\ref{prop:geodesic-correspondence}),
    the higher-genus palindromic hierarchy
    (Conjecture~\ref{conj:higher-genus}),
    and the Legendre-symbol selection rule on arithmetic surfaces
    (Proposition~\ref{prop:legendre-selection}).
  \item \textbf{Scale classification of perturbations}
    (Proposition~\ref{prop:rg-classification}):
    the logarithm is the unique IR fixed point of the additive
    coarse-graining map, with the sign rule demarcating its basin of
    attraction.
\end{enumerate}

\section{The logarithmic interaction: uniqueness and symmetry}
\label{sec:log-interaction}

This section establishes the properties of the logarithmic interaction
that are used in the proof of the sign rule
(Theorem~\ref{thm:sign-rule}, \S\ref{sec:sign-rule}).

\subsection{Dilation invariance and the Cauchy functional equation}
\label{sec:dilation-invariance}

\begin{remark}[The logarithm as the unique dilation-invariant interaction]
\label{rem:rg-fixed-point}
Dilation-invariance $h(br) = h(r) + c(b)$ is a Cauchy functional
equation whose continuous solutions with $h'<0$ are exactly
$h(r)=-a\ln r + b$, $a>0$.  The $N$-body lift is
Corollary~\ref{cor:dilation-rg}:
$H(bz_1,\dots,bz_N)=H(z_1,\dots,z_N)-\tfrac{N(N-1)}{2}\ln b$,
a position-independent shift.  Because $\lambda_m$ depends only on
inter-vortex ratios, $N_{\mathrm{crit}}=7$ is scale-independent
and cannot drift under spatial rescaling.
\end{remark}


\begin{proposition}[Scale classification of perturbations]
\label{prop:rg-classification}
Write $h(r) = -\ln r + \varepsilon\,f(r)$ where $f$ is a perturbation.
The additive rescaling map
$T_b\colon h \mapsto h(b\,\cdot) - h(b)$ acts on~$f$ as
$(T_b f)(r) = f(br) - f(b)$.
(We borrow the ``relevant/irrelevant/marginal'' terminology from
the Wilsonian renormalisation group by analogy; the map $T_b$ is
a pointwise rescaling, not a Wilson-style coarse-graining over
short-wavelength modes.)
\begin{enumerate}
  \item \emph{Irrelevant} ($f(r) \to 0$ as $r \to \infty$):
    the perturbation is suppressed at large scales;
    $h$ flows toward $-\ln r$ under coarse-graining.
  \item \emph{Relevant} ($f(r) \to \infty$ as $r \to \infty$):
    the perturbation dominates at large scales;
    $h$ flows away from $-\ln r$.
  \item \emph{Marginal} ($f(r) \to \mathrm{const}$):
    the perturbation shifts $h$ by a constant.
\end{enumerate}
Eigenfunctions $f(r) = r^\alpha$:
$(T_b\, r^\alpha)(r) = b^\alpha(r^\alpha - 1)$,
so $r^\alpha$ with $\alpha < 0$ is irrelevant and
$\alpha > 0$ is relevant.
\end{proposition}

\begin{proof}
For $f(r) = r^\alpha$ with $\alpha < 0$:
$|f(br)| = b^\alpha |r^\alpha| \to 0$ as $b \to \infty$ for
fixed $r$, so coarse-graining suppresses the perturbation.
For $\alpha > 0$: $|f(br)| = b^\alpha |r^\alpha| \to \infty$,
so the perturbation grows under coarse-graining.
\end{proof}

\noindent
The sign rule condition $h' < 0$ holds iff $\alpha \le 0$ (irrelevant
perturbations): relevant perturbations break the sign-rule guarantee
(though specific relevant perturbations may stabilise individual modes).
Compact vorticity distributions interact as
$H_{\mathrm{eff}} = -K^2\ln d + O(\varepsilon^2/d^2)$ at large
separation; the corrections are irrelevant ($\alpha = -2$),
so \emph{any compact distribution flows to the logarithm}.
The blob bridge ($\alpha = -2$) and $K_0$ interaction
($\alpha = 2$ marginal, then $\alpha = 4$ relevant) of Paper~II
are both instances of this flow.

\begin{proposition}[Isotropic marginality at $\alpha = 2$]
\label{prop:alpha-2-marginal}
For $h(r) = -\ln r + \varepsilon\,r^2$, the Havelock eigenvalue
$\lambda_m$ is independent of $\varepsilon$ at every binding mode
$m = 2, \ldots, N{-}2$.
\end{proposition}

\begin{proof}
The pair Hessian of $g(r) = r^\alpha$ decomposes as
\begin{equation}\label{eq:anisotropy}
  \frac{\partial^2 g}{\partial x_i\,\partial x_j}\bigg|_{|x|=d}
  = \underbrace{\alpha(\alpha{-}2)\,d^{\alpha-2}}_{\displaystyle A(\alpha,d)}\,
  \frac{x_i x_j}{d^2}
  \;+\;
  \underbrace{\alpha\,d^{\alpha-2}}_{\displaystyle B(\alpha,d)}\,
  \delta_{ij},
\end{equation}
where $A$ is the \emph{radial anisotropy} and $B$ is the
\emph{isotropic part}.
\emph{Step 1}: at $\alpha = 2$, $A(2,d) = 2(2{-}2)\,d^0 = 0$.
The Hessian perturbation $\delta\nabla^2 H$ is therefore
$B\,I = 2\,I$ at every pair, making the off diagonal blocks
of the full $2N\times 2N$ Hessian proportional to the identity.
\emph{Step 2 (Fourier block).}
Since the off diagonal blocks are proportional to~$I_2$, the
$xx$ and $yy$ Fourier components are equal and the $xy$ component
vanishes; each $2 \times 2$ Fourier block is scalar.
For the $xx$ component at mode $m \ge 2$: the diagonal block
equals $-\sum_{p=1}^{N-1} B = -2(N{-}1)$, and the off diagonal
sum is $\sum_p B\,e^{-2\pi i pm/N} = 2\sum_p e^{-2\pi i pm/N} = -2$
(since $\sum_{p=1}^{N-1} e^{-2\pi i pm/N} = -1$ for $m \ge 1$).
The Fourier block is therefore $-2(N{-}1) + (-2) = -2N$,
identically for $xx$ and $yy$.  The Rayleigh quotient of the
radial Fourier mode $v_m$ gives
$v_m^\top\,\delta\nabla^2\!H\,v_m = -2N$.

\emph{Step 3 (rotation frequency).}
The equilibrium condition for $H = -\sum G(z_j,z_k)$ at the
$N$-gon gives $\delta\Omega = -(1/R)\,\partial(\delta H)/\partial x_0$,
where $\delta H = -\sum g(d_{jk})$.
For $g(r) = r^2$: $g'(r) = 2r$, and the radial gradient at
vortex $0 = (R,0)$ is
$\partial(\delta H)/\partial x_0 =
-\sum_{p=1}^{N-1} g'(d_p)\,(x_0{-}x_p)/d_p
= -\sum_{p=1}^{N-1} 2d_p\cdot(x_0{-}x_p)/d_p
= -2\sum_{p=1}^{N-1}(x_0 - x_p)$.
Now $\sum_{p=0}^{N-1} x_p = R\sum_{p=0}^{N-1}\cos(2\pi p/N) = 0$
(centroid at the origin), so
$\sum_{p=1}^{N-1}(x_0 - x_p) = (N{-}1)x_0 - \sum_{p=1}^{N-1}x_p
= (N{-}1)R - (-R) = NR$.
Hence $\partial(\delta H)/\partial x_0 = -2NR$ and
$\delta\Omega = -(-2NR)/R = 2N$.

\emph{Step 4 (cancellation).}
$\delta\lambda_m = -\delta\Omega - v_m^\top\,\delta\nabla^2\!H\,v_m
= -(2N) - (-2N) = 0$.

The tangential branch: since $A(2,d) = 0$, the Hessian perturbation is
isotropic ($B\,I_2$ at every pair), so the tangential eigenvalue shift
equals the radial shift.  Steps~2-4 therefore apply identically to
the tangential Fourier block, and $\delta\lambda_m = 0$ holds for
both branches.

Verified to machine precision ($|\delta\lambda_m| < 10^{-14}$)
for $4 \le N \le 12$
(see Code Availability).
\end{proof}

\begin{remark}[RG classification versus stability classification]
\label{rem:rg-vs-stab}
Proposition~\ref{prop:alpha-2-marginal} shows that the
\emph{stability-marginal} exponent ($\alpha = 2$) differs from
the \emph{RG-marginal} exponent ($\alpha = 0$).  The RG
classification (Proposition~\ref{prop:rg-classification}) concerns
the behavior of the interaction under spatial coarse-graining;
the stability classification concerns the perturbation
$\delta\lambda_m$ of the Havelock eigenvalue.  Their relationship is:

\smallskip
\begin{center}
\begin{tabular}{ccccl}
\toprule
$\alpha$ & RG class & $A = \alpha(\alpha{-}2)$ &
  $\delta\lambda_3(N{=}7)$ & Stability effect \\
\midrule
$-2$ & irrelevant & $8$ & $-4\varepsilon$ & destabilising \\
$-1$ & irrelevant & $3$ & $-0.96\varepsilon$ & destabilising \\
$1$  & relevant   & $-1$ & $-0.58\varepsilon$ & destabilising \\
$2$  & relevant   & $0$ & $0$ & isotropic-marginal \\
$3$  & relevant   & $3$ & $+10.6\varepsilon$ & stabilising \\
$4$  & relevant   & $8$ & $+56\varepsilon$ & stabilising \\
\bottomrule
\end{tabular}
\end{center}
\smallskip

\noindent
All entries with $\alpha < 2$ have $\delta\lambda_3 < 0$
(destabilising for $\varepsilon > 0$); all entries with
$\alpha > 2$ have $\delta\lambda_3 > 0$ (stabilising for
$\varepsilon > 0$).  The crossover is at $\alpha = 2$, where
the anisotropy $A$ vanishes.
The Cauchy blob
($g = -r^{-2}$, i.e.\ $\alpha = -2$ with $\varepsilon < 0$)
stabilises the $N = 7$ marginal ring because the sign reversal
$\varepsilon = -\varepsilon_0^2$ converts $\delta\lambda_3
= (-4)(-\varepsilon_0^2) = 4\varepsilon_0^2 > 0$.
\end{remark}

\noindent\emph{Validity of first order theory.}
The perturbation $\delta\lambda_m$ is first order in $\varepsilon$
and reliable when $|\varepsilon\,\delta\lambda_m| \ll |\lambda_m(0)|$.
For the $N = 7$ marginal ring ($\lambda_3(0) = 0$), any nonzero
$\delta\lambda_3$ dominates regardless of $\varepsilon$, so the
\emph{sign} of $\delta\lambda_3$ is the reliable prediction.
For the $N = 8$ ring with $\alpha = 3$, the mode resolved
computation gives $\delta\lambda_3 = 11.8$ and
$\delta\lambda_4 = 11.3$ (the two binding modes), so
$\varepsilon = 0.1$ predicts $\lambda_3 = -0.5 + 1.18 = 0.68 > 0$
and $\lambda_4 = -1.0 + 1.13 = 0.13 > 0$---all modes of the $8$-gon
become stable, giving $N_{\mathrm{crit}} = 8$ at first order.
Quantitative predictions for $N \ge 9$ require the full nonlinear
eigenvalue computation, not the linearised theory.

\subsection{M\"obius covariance of the logarithmic energy}
\label{sec:mobius-cov}

\begin{proposition}[M\"obius covariance]\label{prop:mobius}
Under a conformal map $f\colon\mathbb{C}\to\mathbb{C}$, the
logarithmic energy transforms as
\[
  H(f(z_1),\dots,f(z_N))
  = H(z_1,\dots,z_N) - \sum_{k=1}^{N}\tfrac{N-1}{2}\ln|f'(z_k)|.
\]
Consequently, the point vortex equations of motion are
M\"obius invariant: the single site correction
$\tfrac{N-1}{2}\ln|f'(z_k)|$ depends only on~$z_k$, so
$\partial/\partial \bar z_j$ for $j \ne k$ annihilates it, and the
Hamiltonian gradient $\partial H/\partial \bar z_j$ is unchanged.
\end{proposition}
\begin{proof}
The identity $|f(z){-}f(w)| = |f'(z)|^{1/2}|f'(w)|^{1/2}|z{-}w|
\cdot(1 + O(|z{-}w|))$ is exact when $f$ is M\"obius (the error
term vanishes), so $\ln|f(z_j){-}f(z_k)| = \ln|z_j{-}z_k|
+ \tfrac12\ln|f'(z_j)| + \tfrac12\ln|f'(z_k)|$.  Summing over
all $\binom{N}{2}$ pairs, each $\ln|f'(z_k)|$ appears $N{-}1$
times.  Since $\tfrac12\ln|f'(z_k)|$ depends on $z_k$ alone,
$\partial[\sum_k\tfrac{N-1}{2}\ln|f'(z_k)|]/\partial\bar z_j
= \tfrac{N-1}{2}\,\partial\ln|f'(z_j)|/\partial\bar z_j$,
which is the single site term in the $j$-th equation of motion.
For M\"obius $f$, $f'$ is rational and $\ln|f'|$ is harmonic away
from poles, so this term contributes a gauge-like shift to the
Hamiltonian that does not couple distinct vortices.
\end{proof}

\begin{corollary}[Dilation subgroup = scale invariant Hamiltonian]
\label{cor:dilation-rg}
Among all M\"obius transformations, the dilation subgroup
$\{f(z) = bz : b > 0\}$ is uniquely characterized by having
constant derivative: $f'(z) = b$ for all~$z$.  Substituting
into Proposition~\ref{prop:mobius} gives a correction that is
\emph{position-independent}:
\begin{equation}\label{eq:dilation-rg}
  H(bz_1,\dots,bz_N)
  = H(z_1,\dots,z_N) - \tfrac{N(N-1)}{2}\ln b.
\end{equation}
This is the $N$-body form of the dilation invariance condition of
Remark~\ref{rem:rg-fixed-point}: scaling changes $H$ only by an
additive constant, so the equations of motion
$\dot{\bar z}_j = \partial H/\partial z_j$ are dilation invariant.
For any other M\"obius map $f$, the derivative $|f'(z_k)|$ varies
between sites, producing a configuration dependent correction that
is not a pure constant; the Hamiltonian is \emph{not}
scale invariant under general conformal maps, only under the
one-parameter dilation subgroup.

Equation~\eqref{eq:dilation-rg} is the $N$-body lift of the per-pair
uniqueness result in Remark~\ref{rem:rg-fixed-point}: the logarithmic
energy is the unique $N$-body Hamiltonian that is dilation-invariant.
\end{corollary}

\begin{remark}[The loxodromic thread]
\label{rem:loxodromic}
The logarithmic interaction and the logarithmic spiral are dual
manifestations of the loxodromic subgroup
$z \mapsto e^{(\sigma + i\alpha)t}z$ of the M\"obius group:
the interaction is the unique dilation invariant potential
(Cauchy), the deformation curves are loxodromic orbits,
$N_{\mathrm{crit}} = 7$ is scale independent
(equation~\eqref{eq:dilation-rg}), the palindromic polynomial
encodes conformal inversion $\xi \leftrightarrow 1/\xi$,
and the logarithm is the IR fixed point of the RG flow
(Proposition~\ref{prop:rg-classification}).
\end{remark}

\section{The sign rule for general pairwise interactions}
\label{sec:sign-rule}

\begin{theorem}[Sign rule for general interactions]\label{thm:sign-rule}
For $N \ge 3$ vortices on a ring with general pairwise interaction
$h(r)$, the energy along the centered spiral satisfies
\begin{equation}\label{eq:Hpp-general}
  H''(0) = \sum_{m=1}^{N-1}\Bigl[
    \bigl(d_m^2 h''(d_m) + d_m h'(d_m)\bigr) S_m
    + \frac{h'(d_m)\, m^2(N-m)}{N^2\, d_m}
  \Bigr],
\end{equation}
where $d_m = 2R\sin(\pi m/N)$ and
$S_m = \sum_{k=0}^{N-m-1}[(2k+m-(N{-}1))/(2N)]^2 \ge 0$.

If\, $h'(r) < 0$ and $(rh'(r))' \le 0$ for all $r > 0$, then
$H''(0) < 0$.
\end{theorem}

\begin{proof}
\emph{Step 1: Pairwise distance as a function of $\sigma$.}
For vortices $j$ and $k$ with separation $m = j - k > 0$, write
the centered exponent as $(j - (N{-}1)/2)/N = (j' + k')/2 + m/(2N)$
where $j' = (j-(N{-}1)/2)/N$ and $k' = (k-(N{-}1)/2)/N$.
Define the mean displacement $s = \sigma(j'+k')/2$ and half-separation
$t = \sigma m/(2N)$.  Then $z_j(\sigma) = R\,e^{s+t}\,e^{i\theta_j}$,
$z_k(\sigma) = R\,e^{s-t}\,e^{i\theta_k}$ with
$\theta_j - \theta_k = 2\pi m/N$, giving
\[
  D(\sigma) \;{:=}\; |z_j - z_k|^2
  = R^2 e^{2s}\bigl[e^{2t} + e^{-2t} - 2\cos(2\pi m/N)\bigr]
  = R^2 e^{2s}\bigl[2\cosh(2t) - 2\cos(2\pi m/N)\bigr].
\]
At $\sigma = 0$: $s = t = 0$, so
$D(0) = 2R^2[1 - \cos(2\pi m/N)] = 4R^2\sin^2(\pi m/N) = d_m^2$.

\emph{Step 2: First and second derivatives of $D$.}
Write $c_s = (j'+k')/2$ and $c_t = m/(2N)$, so
$s = \sigma c_s$, $t = \sigma c_t$.  Direct differentiation gives
\begin{align*}
  D'(0) &= 4c_s(1 - \cos\phi), \qquad \phi = 2\pi m/N, \\[3pt]
  D''(0) &= 8c_s^2(1 - \cos\phi) + 8c_t^2.
\end{align*}

\emph{Step 3: Chain rule for $h(\sqrt{D})$.}
Write $d = \sqrt{D}$.  Two applications of the chain rule give
\[
  \frac{\dd^2}{\dd\sigma^2}h(d)\bigg|_{\sigma=0}
  = \underbrace{\frac{h''(d_m)}{4d_m^2}(D')^2
  + \frac{h'(d_m)}{2d_m}\!\left[D'' - \frac{(D')^2}{2d_m^2}\right]}_{\text{evaluated at each pair $(j,k)$}}.
\]
Substituting $D'(0) = 4c_s(1{-}\cos\phi) = 2c_s d_m^2/R^2$
(using $1{-}\cos\phi = d_m^2/(2R^2)$) and collecting terms:
\[
  \frac{\dd^2}{\dd\sigma^2}h(d)\bigg|_0
  = \bigl[d_m^2 h''(d_m) + d_m h'(d_m)\bigr]c_s^2
  + \frac{2R^2\,h'(d_m)}{d_m}\,c_t^2.
\]
Here $d_m^2/(1{-}\cos\phi) = 2R^2$ was used in the second term.

\emph{Step 4: Summation over pairs.}
For fixed separation $m$, sum over the $N-m$ pairs
$(k, k{+}m)$ for $k = 0, \ldots, N{-}m{-}1$.  Define
\[
  S_m = \sum_{k=0}^{N-m-1} c_{s,k}^2, \quad
  \text{where } c_{s,k} = \frac{2k + m - (N{-}1)}{2N}.
\]
Since $\sum_k c_{s,k} = 0$ by the centering, $S_m \ge 0$ with equality
only if all $c_{s,k} = 0$ (which requires $N - m = 1$, i.e.\ $m = N-1$).
The second sum contributes
$(N{-}m)\cdot 2R^2 h'(d_m) c_t^2 / d_m
= 2R^2 h'(d_m) m^2(N{-}m)/(4N^2 d_m)$.
Summing over $m = 1, \ldots, N-1$
yields \eqref{eq:Hpp-general}.

\emph{Step 5: Sign determination.}
Under $h'(r) < 0$: the second term
$B_m = h'(d_m)\,m^2(N{-}m)/(N^2 d_m) < 0$
for every $m = 1, \ldots, N-1$, since $h' < 0$, $d_m > 0$,
and $m^2(N{-}m) > 0$.
Under $(rh'(r))' \le 0$: the first coefficient
$A_m = d_m^2 h''(d_m) + d_m h'(d_m) = d_m \cdot (rh')'\big|_{r=d_m} \le 0$,
so the first term $A_m S_m \le 0$ since $S_m \ge 0$.

The case $m = N{-}1$ requires separate attention.
At $m = N{-}1$ there is only one pair $(0, N{-}1)$, and the
centering sum has a single term:
$c_{s,0} = (0 + (N{-}1) - (N{-}1))/(2N) = 0$,
so $S_{N-1} = 0$.  The first term $A_{N-1} S_{N-1} = 0$ therefore
\emph{vanishes identically}, regardless of the sign of $A_{N-1}$.
However, the second term $B_{N-1} = h'(d_{N-1})\,(N{-}1)^2/(N^2 d_{N-1})$
remains strictly negative (since $h' < 0$).  Thus the $m = N{-}1$
contribution to $H''(0)$ is $B_{N-1} < 0$, and strict negativity
of $H''(0) = \sum_m (A_m S_m + B_m)$ follows from $A_m S_m \le 0$
for all $m$ and $B_m < 0$ for all $m$, with at least one term
strictly negative.
\end{proof}

\begin{remark}[Borderline status of the logarithmic interaction]
\label{rem:borderline}
The sufficient conditions of Theorem~\ref{thm:sign-rule} partition
interactions into three classes:
\begin{enumerate}
  \item[(i)] $h' < 0$ and $(rh')' < 0$: both terms in
    \eqref{eq:Hpp-general} are strictly negative.
    Examples: $h = -r^\alpha$ ($\alpha > 0$), for which
    $(rh')' = -\alpha^2 r^{\alpha-1} < 0$.
  \item[(ii)] $h' < 0$ and $(rh')' = 0$: the first term vanishes
    and only the second contributes.  For smooth $h$ on $(0,\infty)$,
    this class contains a unique functional form, up to positive
    scaling and additive constant:
    $(rh')' = 0 \Rightarrow rh' = c \Rightarrow h' = c/r
    \Rightarrow h = c\ln r + b$; combined with $h' < 0$ gives
    $c < 0$, i.e.\ $h(r) = -a\ln r + b$ with $a > 0$.
  \item[(iii)] $h' < 0$ and $(rh')' > 0$ at some $r$: the sign of
    the first term is indefinite and $H''(0) < 0$ is not guaranteed.
\end{enumerate}
The 2D Green's function $h(r) = -\ln r$ falls in class~(ii)--the
boundary between guaranteed and non-guaranteed negativity.  In this
precise sense, the logarithmic interaction produces the \emph{weakest}
possible guarantee of $H''(0) < 0$: only the $B_m$ terms contribute
(the $A_m$ terms vanish identically), whereas any class~(i)
interaction has both terms strictly negative.  The logarithm is the
least robust polygon producer among all interactions satisfying the
sign rule.
\end{remark}


The sign rule establishes $H''(0) < 0$; the angular-impulse
constraint rescues stability:

\begin{theorem}[Constrained energy minimum]
\label{thm:constrained-min}
The Lagrangian eigenvalue of the $N$-gon on the
angular-impulse surface $\{L = L_0\}$ is
$\lambda_m = (N{-}1) - m(N{-}m)/2$.
This is nonnegative for all binding modes $m = 2,\ldots,N{-}2$
if and only if $N \le 7$.
The $N$-gon is a constrained energy minimum for $N \le 7$;
the marginal case $N = 7$ ($\lambda_3 = 0$) is resolved
nonlinearly in Paper~II.
\end{theorem}

\begin{proof}
The constrained Lagrangian $\mathcal{L} = H - \mu_L L$
has $\mu_L = \Omega = -(N{-}1)/4$ (retrograde rotation).
The Hessian $\nabla^2\!H$ at the $N$-gon has radial
eigenvalue $\sigma_m = (N{-}1)/2 - m(N{-}m)/2$ per
Fourier mode (from the Havelock identity, \S\ref{sec:havelock-identity}).
Adding the Lagrange shift $-2\mu_L = (N{-}1)/2$:
\[
  \lambda_m = \sigma_m + \frac{N{-}1}{2}
  = (N{-}1) - \frac{m(N{-}m)}{2}.
\]
The tangential eigenvalue $m(N{-}m)/2 \ge 0$ is always stable.
The minimum of~$\lambda_m$ over $m = 2,\ldots,N{-}2$ occurs at
$m = \lfloor N/2\rfloor$: positive for $N \le 6$,
zero at $N = 7$ ($\lambda_3 = 6 - 6 = 0$),
negative for $N \ge 8$.
Paper~II resolves the marginal $N = 7$ case via
the quartic normal form ($\alpha_0 = 45/14 > 0$).
\end{proof}

\begin{remark}[Unconstrained saddle versus constrained minimum]
\label{rem:signrule-to-stability}
The sign rule (Theorem~\ref{thm:sign-rule}) proves $H''(0) < 0$:
the $N$-gon is an unconstrained saddle.
Theorem~\ref{thm:constrained-min} shows that constraining
$L = \mathrm{const}$ converts this saddle into a minimum for
$N \le 7$.
The two results are compatible: the spiral deformation
exits the constraint surface at $O(\sigma^2)$, so
$H''(0) < 0$ measures curvature in a direction not
available to the constrained dynamics.
\end{remark}

\section{The Havelock identity and its metric-independence}
\label{sec:havelock-identity}

\subsection{The classical identity}
\label{sec:classical-havelock}

\begin{lemma}\label{lem:havelock}
For integers $N \ge 2$ and $1 \le m \le N-1$,
\[
  T_m := \sum_{p=1}^{N-1}\frac{1-\cos(2\pi pm/N)}{4\sin^2(\pi p/N)}
  = \frac{m(N-m)}{2}.
\]
\end{lemma}

This identity appears in \citet{Havelock1931}, \S3, as a step in
computing the stability eigenvalues of the $N$-vortex ring; see also
\citet{Aref2003}, Appendix~A.

\begin{proof}
Write $\omega = e^{2\pi i/N}$.  Using $2\sin(\pi p/N) =
|1 - \omega^p|$, expand
$[1 - \cos(2\pi pm/N)] / [4\sin^2(\pi p/N)]
= \frac12\bigl|\sum_{k=0}^{m-1}\omega^{pk}\bigr|^2
= \frac{1}{2}\sum_{j=0}^{m-1}\sum_{k=0}^{m-1}\omega^{p(j-k)}$.
Summing over $p = 1, \ldots, N-1$ and using the orthogonality relation
$\sum_{p=1}^{N-1}\omega^{p\ell} = N\delta_{\ell \equiv 0\pmod{N}} - 1$:
\[
  T_m = \frac{1}{2}\sum_{j,k=0}^{m-1}
  \bigl(N\delta_{j \equiv k\pmod{N}} - 1\bigr)
  = \frac{1}{2}(Nm - m^2) = \frac{m(N-m)}{2},
\]
where $\delta_{j \equiv k}$ contributes $m$ (since $0 \le j,k \le m-1
< N$ forces $j = k$) and the $-1$ contributes $-m^2$.
\end{proof}

\begin{remark}[Integrable systems interpretation]
The $\csc^2$ kernel in Lemma~\ref{lem:havelock} is the
interaction potential of the Calogero--Moser--Sutherland (CMS)
model \citep{Sutherland1971, CalogeroJMP1971}, whose potential
is $\beta(\beta{-}1)\sum_{j<k}\csc^2(\pi(x_j{-}x_k)/L)$.
The vortex system corresponds to $\beta = 1$
(free fermions: the CMS potential vanishes).
The Havelock eigenvalues $f(m,N) = m(N{-}m)/2$ coincide with
the Haldane--Shastry dispersion relation (the $\beta \to \infty$
frozen lattice limit); this coincidence reflects the fact that the
\emph{classical equilibrium positions} (roots of unity) are
$\beta$-independent, even though the quantum excitation spectrum
depends on~$\beta$.
At $\beta = 1$, the CMS eigenstates are the Jack polynomials
$J_\lambda^{(1/\beta)} = J_\lambda^{(1)} = s_\lambda$
(Schur polynomials, the characters of $\mathrm{GL}(N)$).
The Havelock stability analysis is therefore a spectral
problem on the $\mathrm{GL}(N)$ representation ring.
\end{remark}

\subsection{The Riemannian Havelock identity}
\label{sec:riemannian}

\begin{theorem}[Metric-independence of $f(m,N)$]
\label{thm:riemannian-havelock}
Let $M$ be a two-dimensional constant-curvature Riemannian
manifold (compact or non-compact) on which the
Laplace--Beltrami Green's function $G$ depends only on
geodesic distance: $G(x,y) = h(d(x,y))$.
Let the $N$-gon at geodesic radius $R$ be an equilibrium of the
point vortex Hamiltonian $H = -\sum_{j<k}G(z_j,z_k)$.
Then the constrained Lagrangian eigenvalues satisfy
\begin{equation}\label{eq:riemannian-havelock}
  \lambda_m = C_1(M, R) - \frac{m(N-m)}{2},
  \qquad m = 1, \ldots, N-1,
\end{equation}
where $C_1$ depends on the surface and ring radius but
$f(m,N) = m(N{-}m)/2$ is universal.
\end{theorem}

\noindent
The hypotheses (constant curvature, $G = h(d)$) hold on:
\begin{itemize}
  \item the plane ($G = -(2\pi)^{-1}\ln r$, $C_1 = N{-}1$);
  \item the sphere $\mathbf{S}^2$ of radius $a$
    ($G = -(4\pi)^{-1}\ln(2{-}2\cos\gamma)$,
    $C_1 = (N{-}1)(1{-}\xi)/(1{+}\xi)$, $\xi = \tan^2(\varphi_0/2)$);
  \item the hyperbolic plane $\mathbf{H}^2$ of curvature
    radius $a$
    ($G = -(2\pi)^{-1}\ln\tanh(d/2)$,
    $C_1 = (N{-}1)(1{+}\xi^2)/(1{-}\xi)^2$, $\xi = r_E^2/a^2$).
\end{itemize}
It does \emph{not} hold on surfaces of non-constant curvature
(where $G$ depends on the positions individually, not just on
their separation) or on generic quotients $\Gamma\backslash M$
(where $G$ is the automorphic Green's function, depending on all
images $\gamma(y)$).

\begin{proof}
The proof uses two ingredients, both independent of $G$.

\emph{Step 1: Fourier-block structure.}
Let $\mathbf{M}_p$ be the $2{\times}2$ Hessian of $h(d_{0p})$
in the local (radial, tangential) frame at vortex~$0$.
The $\mathbb{Z}_N$-Fourier decomposition of the full Hessian gives,
for each component $\alpha,\beta\in\{r,t\}$:
\[
  H_{\alpha\beta}^{(m)}
  = \sum_{p=1}^{N-1} M_{p,\alpha\beta}
    \bigl(\cos(2\pi pm/N)-1\bigr),
\]
since the diagonal (self-interaction) block equals $-\sum_p M_p$
(the equilibrium condition forces the total gradient to vanish).

\emph{Step 2: Mode-independent trace.}
Summing the radial and tangential components:
\begin{equation}\label{eq:trace-formula}
  H_{rr}^{(m)} + H_{tt}^{(m)}
  = \sum_{p=1}^{N-1} \nabla^2_{\mathrm{LB}} h(d_p)
    \bigl(\cos(2\pi pm/N) - 1\bigr),
\end{equation}
where $\nabla^2_{\mathrm{LB}} h(d_p) =
\mathrm{tr}[\mathbf{M}_p]$.  The key input:
away from the diagonal ($x \ne y$), the Laplace--Beltrami
equation gives $\nabla^2_{\mathrm{LB},x}\,G(x,y) = c$, where
\begin{equation}\label{eq:laplacian-constant}
  c = \begin{cases}
    0 & \text{plane or $\mathbf{H}^2$ (non-compact, $\mathrm{Vol}=\infty$)}, \\
    -1/\mathrm{Vol}(M) & \text{compact surface (from $\Delta G = \delta_y - 1/\mathrm{Vol}$)}.
  \end{cases}
\end{equation}
Since $d_p \ne 0$ for all pairs on the $N$-gon (no two vortices
coincide), $\nabla^2_{\mathrm{LB}} h(d_p) = c$ for every $p$.
Substituting into~\eqref{eq:trace-formula}:
\[
  H_{rr}^{(m)} + H_{tt}^{(m)}
  = c \sum_{p=1}^{N-1}(\cos(2\pi pm/N) - 1)
  = c(-1 - (N{-}1)) = -Nc
  \eqqcolon \delta,
\]
using $\sum_{p=1}^{N-1}\cos(2\pi pm/N) = -1$ for $m \ge 1$.
The trace $\delta$ is independent of~$m$: it enters $C_1$ as an
additive shift but leaves $f(m,N) = m(N{-}m)/2$ unchanged.
(The $\csc^2$ kernel factor $1/(4\sin^2)$ in the tangential
eigenvalue is \emph{derived} in Step~3 below---it is the angular
Hessian of the logarithmic chord distance, not a heat-kernel
coefficient.)

On the plane: $c = 0$, $\delta = 0$ (exact tracelessness).
On $\mathbf{S}^2$ of radius $a$:
$c = -1/(4\pi a^2)$, $\delta = N/(4\pi a^2)$.
On $\mathbf{H}^2$: $c = 0$ (non-compact), $\delta = 0$.
On a compact quotient $\Gamma\backslash\mathbf{H}^2$:
$c = -1/\mathrm{Vol}(\Gamma\backslash\mathbf{H}^2)$,
$\delta = N/\mathrm{Vol}$---but note that on such a quotient
the Green's function is automorphic and does NOT depend only on
geodesic distance, so the hypothesis of the theorem is not
satisfied.  The identity~\eqref{eq:riemannian-havelock}
applies to the universal cover, not to the quotient.

\emph{Step 3: From mode-independent trace to the eigenvalue formula.}
Step~2 shows that $\mathrm{tr}(\mathbf{H}^{(m)}) = \delta$
(mode-independent).  It remains to show that the
\emph{individual} constrained eigenvalue is $C_1 - m(N{-}m)/2$.
On any surface satisfying the hypothesis,
the Laplacian Green's function has a logarithmic singularity:
$G(x,y) = -(2\pi)^{-1}\ln d(x,y) + \varphi(x,y)$, where $\varphi$
is smooth at $x = y$.
The tangential second derivative of $G$ at the $p$-th chord
of the $N$-gon ring therefore separates as
\begin{equation}\label{eq:angular-kernel}
  M_{p,tt}
  = \frac{1}{4\sin^2(\pi p/N)}\,\alpha(R, K)
  \;+\; \beta_p(R, K),
\end{equation}
where $\alpha(R, K)$ depends on the ring radius and curvature
but is independent of~$p$, and $\beta_p$ is the smooth-part
contribution (bounded uniformly in~$p$).
The $\csc^2$ factor arises because, on any constant-curvature
surface, the chord distance satisfies
$d_p = 2\ell(R)\,|\!\sin(\pi p/N)|$
(with $\ell = R$, $R\sin(R/a)/\sin(\cdot)$, or
$a\sinh(R/a)/\sinh(\cdot)$ on the plane, $\mathbf{S}^2$,
$\mathbf{H}^2$ respectively), so
$\ln d_p = \ln|\!\sin(\pi p/N)| + (\text{terms independent of } p)$.

Substituting~\eqref{eq:angular-kernel} into the Fourier sum:
\begin{align*}
  H_{tt}^{(m)} &= \alpha \sum_{p=1}^{N-1}
    \frac{1{-}\cos(2\pi pm/N)}{4\sin^2(\pi p/N)}
    + \sum_p \beta_p\,(\cos(2\pi pm/N) {-} 1) \\
  &= \alpha\,T_m + (\text{mode-independent remainder})
  = \alpha\,\frac{m(N{-}m)}{2} + \tilde{C}_1,
\end{align*}
using $T_m = m(N{-}m)/2$ (Lemma~\ref{lem:havelock}).
The second sum is mode independent by the following
argument, which works simultaneously on all
constant-curvature surfaces.

In conformal coordinates (Poincar\'e disk for
$\mathbf{H}^2$, stereographic for $\mathbf{S}^2$),
the Hamiltonian of $N$ equal point vortices on
\emph{any} constant curvature surface separates as
\begin{equation}\label{eq:H-separation}
  H = -\!\sum_{j<k}\!\ln|z_j {-} z_k|
    \;+\; V_{\mathrm{rad}}(\{|z_k|\}),
\end{equation}
where $V_{\mathrm{rad}}$ collects \emph{all}
terms depending only on the radii~$|z_k|$.
This includes the self energy
(on $\mathbf{H}^2$: $(N{-}1)/2\sum_k \ln(a^2{-}|z_k|^2)$;
on $\mathbf{S}^2$: $-(N{-}1)/2\sum_k \ln(1{+}|z_k|^2)$;
on the plane: $0$)
\emph{and} the cross image sum.
The cross image terms
$\sum_{p=1}^{N-1}\ln|a^2 - \bar{z}_0\,z_p|$
at the $N$-gon ring telescope via the cyclotomic product
$\prod_{p=1}^{N-1}(1 - \xi\,e^{2\pi ip/N}) = 1 - \xi^N$
(the minimal polynomial of the $N$-th root of unity;
see e.g.\ \citealt{Lang2002}, \S IV.1)
(with $\xi = \rho^2/a^2$), giving
$\ln|1 - \xi^N|$---a function of~$\rho$ alone.
For tangential modes $m \ge 1$, the perturbation phases $a_k$ satisfy $\sum_k a_k = 0$ ($\mathbb{Z}_N$ orthogonality), preserving the product identity at all perturbation orders.
Explicitly: $\partial_\varepsilon|_0 \prod(1 - \xi\,e^{i(\theta_k + \varepsilon a_k)}) = -\xi\prod'(1{-}\xi\omega^k)\sum_k i a_k \omega^k/(1{-}\xi\omega^k)$. The constraint $\sum a_k = 0$ forces this into $\mathbb{Z}_N$ modes $m \ge 1$, each giving an independent eigenvalue.
Although the \emph{individual} image terms
$\ln|a^2 - \rho^2\,e^{2\pi ip/N}|$ depend on the
angular separation~$p/N$, their \emph{sum} does not.
Under a tangential perturbation
$z_k \mapsto \rho\,e^{i(\theta_k + \varepsilon\,a_k)}$,
the radius $|z_k| = \rho$ is \emph{preserved},
so the entire $V_{\mathrm{rad}}$
(self-energy $+$ telescoped cross-images)
has \emph{zero} tangential Hessian.

The constraint $J$ (the angular impulse / moment map)
also depends only on $|z_k|$, so $\partial^2 J/\partial\theta^2 = 0$
and the Lagrange multiplier contributes nothing
to the tangential eigenvalue.

The tangential Hessian of $-\sum\!\ln|z_j{-}z_k|$
at the ring $z_p = \rho\,e^{2\pi ip/N}$ is a
\emph{circulant} matrix with the $\csc^2$ kernel:
the off-diagonal entry at pair~$(0, p)$ is
$-\csc^2(\pi p/N)/(4\rho^2)$, and its
$\mathbb{Z}_N$ Fourier eigenvalue is
\[
  H_{tt}^{(m)}
  = \frac{1}{\rho^2}\sum_{p=1}^{N-1}
    \frac{1{-}\cos(2\pi pm/N)}{4\sin^2(\pi p/N)}
  = \frac{m(N{-}m)}{2\rho^2}
\]
by the Havelock identity (Lemma~\ref{lem:havelock}).
In the geodesic normalisation
$\lambda_m r_E^2 = C_1 - m(N{-}m)/2$
(with $r_E$ the Euclidean ring radius),
the tangential sector contributes exactly $m(N{-}m)/2$,
and $C_1$ comes entirely from the radial sector
(the confining potential and equilibrium condition).
Since none of the surface-dependent quantities
($V_{\mathrm{rad}}$, $J$, or $\mu_L$) enter the
tangential Hessian, the identity holds on
\emph{every} constant-curvature surface simultaneously
(verified numerically; see Code Availability).
\end{proof}

\begin{remark}[Compact quotients]
On compact quotients $\Gamma\backslash M$, the Green's function
$G_\Gamma(z,w) = \sum_{\gamma \in \Gamma} G_M(z, \gamma w)$
includes automorphic image sums.  The image contribution
to the tangential Hessian at the $N$-gon ring is
$\delta H_{tt}^{(m)} = \sum_{\gamma \ne \mathrm{id}}
\sum_p (\partial^2_\theta G_M)(z_0, \gamma z_p)\,
e^{2\pi ipm/N}$.
When the polygon center~$z_0$ is the $\mathrm{Aut}(\Sigma)$-fixed
point of the Dirichlet domain (as on the Bolza surface),
each $\gamma \in \Gamma$ maps the ring to a concentric ring,
so $\gamma$ commutes with the $\mathbb{Z}_N$ rotation.
The inner sum then factorises as
$(\sum_p e^{2\pi ipm/N}) \times f(\gamma)$,
and $\sum_p e^{2\pi ipm/N} = 0$ for $m \ge 1$.
The automorphic images therefore contribute a
mode-independent shift $\delta C_1 = \sum_{\gamma \ne \mathrm{id}}
G_M(z_0, \gamma z_0)$, preserving $f(m,N) = m(N{-}m)/2$
exactly.
\end{remark}

The stability threshold $N_{\mathrm{crit}}$ is therefore determined by
$C_1(\text{surface}) \ge \max_m m(N{-}m)/2$: a condition involving the
\emph{universal} function $f$ and a single surface dependent constant.
On the plane: $C_1 = N{-}1$, giving $N_{\mathrm{crit}} = 7$ (the
largest integer with $(N{-}1) \ge \lfloor N/2\rfloor(N{-}\lfloor
N/2\rfloor)/2$).  On the sphere:
$C_1(\varphi_0) < N{-}1$ for $\varphi_0 > 0$, giving the
colatitude-dependent threshold of Section~\ref{sec:curved}.

\begin{remark}[Spectral gap and stability boundary]
\label{rem:rep-theory}
Write $H_N = C_1 I - F_N$ where $F_N = \mathrm{diag}(f(0,N),
\ldots, f(N{-}1,N))$.  Stability is $H_N \ge 0$.  The spectral gap
$\Delta(N) = C_1 - f_{\max}(N)$ closes at $N = 7$ on the flat
plane, marking a transition from the gapped phase
($N \le 6$) to the broken phase ($N \ge 8$).
Gaussian curvature acts as a tuning parameter:
$K < 0$ opens the gap, $K > 0$ closes it.
\end{remark}

The value $C_1 = N{-}1$ on the plane has a geometric origin: for
$h' = -1/d$ (the logarithmic force law), each vortex pair $(0,k)$
contributes the \emph{same} amount $-1/(2R)$ to the gradient
$\partial H/\partial x_0$, regardless of the separation~$k$.  This
is because $h'(d_k)(x_0{-}x_k)/d_k = -(1{-}\cos(2\pi k/N))/
(4R\sin^2(\pi k/N)) = -1/(2R)$: the geometric factor
$(1{-}\cos)$ cancels against the denominator $\sin^2 = (1{-}\cos)/2$.
The cancellation is the \emph{scale invariance} of $h' = -1/d$:
the force law $h' \propto 1/d$ precisely compensates the
$d$-dependence of the chord geometry, making each pair's contribution
independent of the separation.  This cancellation characterizes the
logarithm uniquely (Remark~\ref{rem:rg-fixed-point}).

\begin{remark}[Polynomial eigenvalues are unique to $d = 2$]
\label{rmk:polynomial-uniqueness}
The Havelock identity (Lemma~\ref{lem:havelock}) states
$T_m = m(N{-}m)/2$, a polynomial of degree~$2$ in~$m$.
Define $S_\alpha(m,N) = \sum_{p=1}^{N-1}[1{-}\cos(2\pi pm/N)]
/|\!\sin(\pi p/N)|^\alpha$.
Then $S_\alpha$ is polynomial in~$m$ if and only if $\alpha = 2$.

\emph{Proof.}
The second finite difference
$\Delta^2 S_\alpha(m) = S_\alpha(m{+}1) - 2S_\alpha(m) + S_\alpha(m{-}1)$
equals $4\!\sum_p\cos(2\pi pm/N)\,|\!\sin(\pi p/N)|^{2-\alpha}$.
For $\alpha = 2$: the $\sin$ factor is $\sin^0 = 1$, so the sum
is $4\!\sum_p\cos(2\pi pm/N) = -4$ (constant in~$m$), confirming
degree~$2$.
For $\alpha \ne 2$: the factor $|\!\sin|^{2-\alpha}$ is
non-constant in~$p$, so $\Delta^2$ is a non-trivial Fourier
coefficient of $|\!\sin|^{2-\alpha}$.
This function has infinite Fourier support for non-integer
$2{-}\alpha$ (its Fourier coefficients decay as
$O(n^{\alpha-3})$ by the Paley--Wiener theorem for
piecewise-smooth functions; see
\citealt{Katznelson2004}, Theorem~VI.2.4).
Numerically: $\Delta^2 S_2 = -4.0$ to~$10^{-14}$ for all~$m$
and~$N$; for $\alpha = 3$, $\Delta^2$ varies by a factor
of~$3$ across~$m$ (see Code Availability).

The polynomial structure for $\alpha = 2$ follows from
the holomorphic factorization
\begin{equation}\label{eq:holomorphic-factorization}
  \frac{1 - \cos(2\pi pm/N)}{\sin^2(\pi p/N)}
  = 2\,\Bigl|\sum_{k=0}^{m-1}\omega^{pk}\Bigr|^2,
  \qquad \omega = e^{2\pi i/N},
\end{equation}
using $1 - \cos\theta = 2\sin^2(\theta/2)$ and the
geometric sum $\sum\omega^{pk} = (1{-}\omega^{pm})/(1{-}\omega^p)$.
This factorization uses the holomorphic function
$z \mapsto z^k$ (a polynomial in $z = \omega^p$),
which exists only on two-dimensional (complex) domains.
In $d \ge 3$ dimensions, the angular Hessian kernel is
$K_d(x) \propto \sin^{-d}(x)$ (from $G_d \propto r^{2-d}$),
and $\alpha = d \ne 2$: the factorization fails and
the eigenvalue structure is non-polynomial.
The polynomial eigenvalue structure---and with it CMS
integrability, exact Born--Oppenheimer separation, and
palindromic thresholds---is unique to $d = 2$,
i.e.\ to the logarithmic interaction.
\end{remark}

\section{Stability thresholds on constant-curvature surfaces}
\label{sec:curvature-thresholds}

\noindent\emph{Prior work and what is new.}
\citet{BoattoCabral2003} computed the constrained Lagrangian eigenvalues
for a latitudinal ring of point vortices on the non-rotating sphere at
arbitrary colatitude, establishing that the stability boundary shifts
with latitude.  \citet{LimMontaldiRoberts2001} studied relative
equilibria of vortex rings on $S^2$ within the Hamiltonian framework.


What the proof in~\S\ref{sec:riemannian} adds is an explicit algebraic
argument that the $m$-dependent part $f(m,N) = m(N{-}m)/2$ is
\emph{metric-independent}---a consequence of $\mathbb{Z}_N$
representation theory alone---while only the additive constant $C_1$
carries the geometric information.  The separation is implicit in the
Boatto--Cabral calculation; we make it explicit and extend the framework
to~$\mathbf{H}^2$~(\S\ref{sec:hyperbolic}), where systematic stability
tables do not appear in the literature.

\subsection{Curved 2D surfaces: the spherical stability boundary}
\label{sec:curved}

On the 2-sphere, the point vortex interaction
$h(\theta) = -\ln(2\sin(\theta/2))$ satisfies $h' < 0$ and
$(\sin\theta\cdot h')' = \tfrac12\sin\theta > 0$ (class~(iii)).
Numerical evaluation shows $H''(0) > 0$ along the spiral
for $N = 3,\ldots,8$ at colatitudes $\varphi_0 \in \{5^\circ,
15^\circ, \ldots, 85^\circ\}$
(verified numerically; see Code Availability):
the polygon is a spiral-direction energy minimum even without
the Lagrange shift.  The \emph{constrained} Hessian tells a
different story.

The angular impulse on the sphere is
$L = \sum\cos\theta_k$, with Hessian
$\partial^2 L/\partial\theta_k^2 = -\cos\varphi_0$ at colatitude
$\varphi_0$.  The Lagrangian shift from $\mu_L$ is therefore
proportional to $\cos\varphi_0$: full near the pole ($\varphi_0 \to 0$),
vanishing at the equator ($\varphi_0 = 90^\circ$).  This
latitude-dependent shift produces a \emph{colatitude-dependent
stability boundary}:

\begin{center}
\begin{tabular}{lcccccc}
\toprule
$N$ & 3 & 4 & 5 & 6 & 7 & $\ge 8$ \\
\midrule
$\varphi_{\mathrm{crit}}$ & $60.0^\circ$ & $48.2^\circ$ &
  $41.4^\circ$ & $25.8^\circ$ & $0^\circ$ & --- \\
\bottomrule
\end{tabular}
\end{center}

\noindent
The $N$-gon is a constrained energy minimum on the sphere if and only
if $\varphi_0 < \varphi_{\mathrm{crit}}(N)$.  The dimensionless
eigenvalue $\bar\lambda = \sin^2\!\varphi_0 \cdot \lambda$ recovers
the Havelock values to within $1\%$ at $\varphi_0 = 3^\circ$
(N=5: $\bar\lambda = 0.99$, Havelock $= 1.0$;
N=6: $0.49$ vs $0.5$; N=8: $-1.01$ vs $-1.0$).

For $N = 7$: $\cos\varphi_{\mathrm{crit}} = 6/6 = 1$, so
$\varphi_{\mathrm{crit}} = 0^\circ$---the $7$-gon is marginally
stable only at the exact pole, consistent with the flat-plane
marginality $\lambda_3 = 0$.
For $N \ge 8$: $\cos\varphi_{\mathrm{crit}} > 1$, which is
unattainable; no stable latitude exists on the sphere.
Jupiter's octagon ($N = 8$) is therefore unstable at all
colatitudes, consistent with the need for a central vortex.

\paragraph{Derivation of $C_1(\mathbf{S}^2)$ and exact thresholds.}
For $N$ vortices on a latitude circle at colatitude $\varphi_0$, the
great-circle separation satisfies
$2\sin(\gamma_{jk}/2) = 2\sin\varphi_0\,|\sin(\Delta\phi_{jk}/2)|$,
so the spherical pair interaction decomposes as
\[
  -\ln\bigl(2\sin(\gamma_{jk}/2)\bigr)
  = -\ln\bigl(2|\sin(\Delta\phi_{jk}/2)|\bigr) - \ln(\sin\varphi_0).
\]
The first term is the flat Thomson energy in the azimuthal variable
$\phi$; its Hessian along azimuthal Fourier modes gives the Havelock
eigenvalues $(N{-}1) - m(N{-}m)/2$ exactly as in the flat case.
The second term $-\binom{N}{2}\ln(\sin\varphi_0)$ is purely
radial and enters through the Lagrange multiplier.

The angular impulse on $\mathbf{S}^2$ is $L = \sum\cos\varphi_k$.
At the ring, $\partial^2 L/\partial\varphi_k^2 = -\cos\varphi_0$,
so the Lagrange shift
$\mu_L\cdot\partial^2 L/\partial\varphi_k^2$ is proportional to
$\cos\varphi_0$: maximal near the pole, vanishing at the equator.
Computing $\mu_L$ from the stationarity condition and combining with
the radial Hessian of the pair interaction yields the curvature
coefficient (cf.\ \citealt{BoattoCabral2003}):
\begin{equation}\label{eq:C1_S2}
  C_1(\mathbf{S}^2,\,\xi) = \frac{(N-1)(1-\xi)}{1+\xi} = (N-1)\cos\varphi_0,
\end{equation}
where $\xi = \tan^2(\varphi_0/2)$ is the stereographic parameter and
$\cos\varphi_0 = (1{-}\xi)/(1{+}\xi)$.
Marginal stability $C_1 = m(N{-}m)/2$ therefore reduces to
\begin{equation}\label{eq:cos_crit}
  \cos\varphi_{\mathrm{crit}} = \frac{m(N{-}m)}{2(N-1)}, \qquad m = \lfloor N/2\rfloor,
\end{equation}
which is \emph{always rational}.  The threshold in $\xi$ is
\begin{equation}\label{eq:xi_crit_S2}
  \xi_{\mathrm{crit}}(N) = \frac{(N-1) - m(N{-}m)/2}{(N-1) + m(N{-}m)/2}
  \in \mathbb{Q}.
\end{equation}
Explicit values:

\midrule
3 & $1$ & $1/2$ & $1/3$ & $60.0^\circ$ & $\varphi_0 < 60^\circ$ \\
4 & $2$ & $2/3$ & $1/5$ & $48.2^\circ$ & $\varphi_0 < 48.2^\circ$ \\

\noindent
The thresholds $\xi_{\mathrm{crit}} \in \{1/3,\,1/5,\,1/7,\,1/19\}$ are exact
rational numbers even though the Green's function $h(\theta) =
-\ln(2\sin(\theta/2))$ is transcendental---the rationality is a consequence of
the linear dependence of $C_1$ on $\cos\varphi_0$ (\eqref{eq:C1_S2}) combined
with the rational marginal condition~\eqref{eq:cos_crit}.

\begin{remark}[Pattern $\xi_{\mathrm{crit}} = 1/(2N{-}3)$ and its break at $N=6$]
For $N = 3, 4, 5$ one has $m(N{-}m)/2 = N{-}2$ exactly (since $m = \lfloor
N/2\rfloor = (N-1)/2$ or $N/2$ gives the same result), so the numerator
$(N-1) - (N-2) = 1$ and denominator $(N-1) + (N-2) = 2N-3$, yielding
$\xi_{\mathrm{crit}} = 1/(2N{-}3)$.  At $N = 6$ this pattern breaks: the
critical mode has $m(N{-}m)/2 = 9/2 \ne N{-}2 = 4$, because the hexagonal
mode carries a larger Havelock eigenvalue ($\lambda_{\min} = 1/2$ vs.\ $1$
for $N \le 5$).  Consequently $\xi_{\mathrm{crit}}(6) = 1/19$, not the
$1/(2 \cdot 6{-}3) = 1/9$ that the pattern would predict.

The notation $\bar\lambda = \sin^2\!\varphi_0 \cdot \lambda$ (introduced in \S\ref{sec:curved})
converts the dimensionful eigenvalue $\lambda$ to the dimensionless form
used in the numerical table above; the exact $\xi$-threshold~\eqref{eq:xi_crit_S2}
is independent of this rescaling.

The colatitudes in the summary table above are the exact values
$\varphi_{\mathrm{crit}} = \arccos(m(N{-}m)/(2(N{-}1)))$;
for $N = 6$, $\cos\varphi_{\mathrm{crit}} = 9/10$.
\end{remark}

\begin{remark}[The stability staircase and the counting function]
\label{rmk:stability-staircase}
The number of simultaneously stable polygon types \emph{decreases}
with colatitude in a staircase:
5 types at the pole ($\varphi_0 = 0$, counting $N = 7$ as marginally stable),
4 for $\varphi_0 < 25.8^\circ$,
3 for $\varphi_0 < 41.4^\circ$,
2 for $\varphi_0 < 48.2^\circ$,
1 (triangle only) for $\varphi_0 < 60^\circ$,
and 0 beyond the equator.

The total ``stable volume''---the sum over polygon types of the
fraction of the hemisphere where that type is stable---is
\begin{equation}\label{eq:counting-function}
  \mathcal{N} = \sum_{N=3}^{7}
    \bigl(1 - \cos\varphi_{\mathrm{crit}}(N)\bigr)
  = \frac{1}{2} + \frac{1}{3} + \frac{1}{4} + \frac{1}{10} + 0
  = \frac{71}{60}.
\end{equation}
The individual summands
$1 - \cos\varphi_{\mathrm{crit}} \in
\{\frac{1}{2}, \frac{1}{3}, \frac{1}{4}, \frac{1}{10}\}$
have the closed forms
\begin{equation}\label{eq:odd-even-patterns}
  s(2k{+}1) = \frac{3{-}k}{4} \quad (\text{odd }N),
  \qquad
  s(2k) = \frac{4k{-}2{-}k^2}{2(2k{-}1)} \quad (\text{even }N),
\end{equation}
giving $s = 0$ at $N = 7$ (the stability boundary) and $s < 0$
for $N \ge 8$.
Their sum satisfies
$\sum_{N=3}^{6}\cos\varphi_{\mathrm{crit}} = 169/60 = 13^2/60$,
so $\mathcal{N} = 4 - 13^2/60 = 71/60$.

The rationality of $\mathcal{N}$ follows from the rationality of
\emph{all} $\mathbf{S}^2$ thresholds: on the sphere, the marginal
equation $C_1 = f_{\mathrm{crit}}$ is linear in $\xi$ (because
$C_1(\mathbf{S}^2, \xi)$ is a linear fractional function of $\xi$),
in contrast to the quadratic palindromic equation on
$\mathbf{H}^2$.
This is the first manifestation of the $\mathbf{S}^2$--$\mathbf{H}^2$
asymmetry: positive curvature yields rational thresholds, negative
curvature yields algebraic-irrational ones.

On $\mathbf{H}^2$, the curvature \emph{stabilises}: $C_1$ grows
without bound as $\xi \to 1$, so arbitrarily large polygons become
stable near the boundary.  The flat plane ($\Lambda = 0$) is the
transition: $N_{\mathrm{crit}} = 7$ everywhere, and
$\mathcal{N}(0) = 5$ (all five types stable at every radius).
\end{remark}

\noindent
All exact threshold computations are verified by the companion
software (Appendix~A, Code Availability).

On the flat torus, $H''(0) < 0$ approaches the planar value for
polygons small relative to the domain, with periodic corrections
entering at the domain scale.

\subsection{The hyperbolic plane and Gaussian curvature}
\label{sec:hyperbolic}

\citet{Kimura1999} and others have studied point vortex dynamics on
surfaces of constant curvature; the Hamiltonian for the hyperbolic plane
is known.  Systematic stability tables for $N$-gons on $\mathbf{H}^2$
do not appear in the existing literature.

The threshold $N_{\mathrm{crit}}$ on any surface is the largest $N$
with $C_1(\text{surface}) \ge \max_m m(N{-}m)/2$.
Gaussian curvature $K$ is the control parameter:

\midrule

\end{tabular}
\end{center}
\begin{remark}[Order of the curvature correction]
\label{rem:euler}
The curvature enters $N_{\mathrm{crit}}$ at $O(K)$, not $O(K^2)$:
it acts through the surface Green's function via the Havelock
coefficient $C_1(\text{surface},\xi)$
(equations~\eqref{eq:C1_S2},~\eqref{eq:C1_H2}), which is linear
in $K$ at leading order.  On a compact surface the Gauss--Bonnet
constraint $\int K\,\mathrm{d}A = 2\pi\chi$ enters only through
$C_1$; no additional $O(K^2)$ self-energy correction arises for
point vortices (such a correction would require finite-area patches,
where it is $O(\varepsilon^2)$ and topology-independent).
\end{remark}

\noindent
On the hyperbolic plane, $C_1(\xi) = (N{-}1)(1{+}\xi^2)/(1{-}\xi)^2$
(equation~\eqref{eq:C1_H2} below) is strictly increasing in
$\xi = r_E^2/a^2$.  Setting $C_1(\xi^*) = \max_m m(N{-}m)/2$ and
solving the palindromic quadratic
$(c{-}1)\xi^2 - 2c\,\xi + (c{-}1) = 0$
with $c = \max_m m(N{-}m)/(2(N{-}1))$
gives the \emph{exact} threshold for each transition:
\begin{equation}\label{eq:xi-star-exact}
  \xi^*(N) = \frac{c - \sqrt{2c{-}1}}{c - 1},
  \qquad c = \frac{m(N{-}m)}{2(N{-}1)},\;\; m = \lfloor N/2\rfloor.
\end{equation}
No floating-point computation is involved: $\xi^*$ is an
\emph{algebraic number} for every~$N$, and an algebraic
\emph{integer} exactly when $N \in \{8, 9, 11, 15, 23\}$
(Proposition~\ref{prop:alg-int}).

The Riemannian Havelock identity $\lambda_m = C_1(R/a) - m(N{-}m)/2$
is verified to hold (computed $C_1$ values are mode-independent to
numerical precision~$\sim 10^{-5}$).  The exact $\mathbf{H}^2$
Green's function yields
\begin{equation}
  C_1(\mathbf{H}^2,\,\xi)
    = \frac{(N-1)(1+\xi^2)}{(1-\xi)^2},
  \qquad \xi = \frac{r_E^2}{a^2} = |K|\,r_E^2 ,
  \label{eq:C1_H2}
\end{equation}
where $r_E$ is the Euclidean ring radius in the Poincar\'e disk and
$K = -1/a^2$.

\begin{proof}[Derivation of~\eqref{eq:C1_H2}]
Work in the Poincar\'e disk of radius $a$ ($K = -1/a^2$).
The point vortex Hamiltonian and angular impulse are
\citep{Kimura1999}
\begin{align}
  H &= -\!\!\sum_{j<k}\ln|z_j{-}z_k|
  + \tfrac{N{-}1}{2}\!\sum_k\ln(a^2{-}|z_k|^2), \label{eq:H-H2}\\
  J &= \sum_k \frac{a^2{+}|z_k|^2}{a^2{-}|z_k|^2}
  \qquad\text{(SO(2,1) moment map)}.
  \label{eq:J-H2}
\end{align}
The $N$-gon equilibrium has $z_k = r_E\,e^{2\pi i k/N}$ with
$\xi = r_E^2/a^2$.

\emph{Step 1: equilibrium rotation frequency $\Omega$.}
At vortex $0 = (r_E, 0)$, the radial gradients are
(using $\sum_{p=1}^{N-1}(x_0{-}x_p)/|z_0{-}z_p|^2 = (N{-}1)/(2r_E)$
from the flat-plane Havelock sum):
\begin{align*}
  \frac{\partial H}{\partial x_0}
  &= -\frac{N{-}1}{2r_E}
  - \frac{(N{-}1)\,r_E}{a^2 - r_E^2}, \\
  \frac{\partial J}{\partial x_0}
  &= \frac{4a^2\,r_E}{(a^2 - r_E^2)^2}.
\end{align*}
The equilibrium condition
$\partial H/\partial x_0 = \Omega\,\partial J/\partial x_0$
gives
\[
  \Omega
  = \frac{-(N{-}1)(a^2{+}r_E^2)}{2r_E(a^2{-}r_E^2)}
  \cdot\frac{(a^2{-}r_E^2)^2}{4a^2 r_E}
  = \frac{-(N{-}1)(a^2{+}r_E^2)(a^2{-}r_E^2)}{8a^2 r_E^2}
  = \frac{-(N{-}1)(1{+}\xi)(1{-}\xi)}{8\xi},
\]
where $\xi = r_E^2/a^2$ and the last step sets $a = 1$.

\emph{Step 2: Hessian Rayleigh quotient.}
By Theorem~\ref{thm:riemannian-havelock}, the constrained
eigenvalue is
$\lambda_m = -\Omega\,(\partial^2 J/\partial r^2)^{-1}
- v_m^\top\nabla^2\!H\,v_m$.
But since the trace $\delta = 0$ on $\mathbf{H}^2$ (non-compact,
$c = 0$ in~\eqref{eq:laplacian-constant}), the Fourier-block
trace is zero and
$v_m^\top\nabla^2\!H\,v_m = (m(N{-}m)/2 - C_1^H)/ r_E^2$
where $C_1^H$ absorbs all surface-dependent terms.

\emph{Step 3: assembling $C_1$.}
With $\xi = r_E^2/a^2$ and $a = 1$, the constrained
eigenvalue receives three contributions:
(i)~the flat Havelock term $(N{-}1)$;
(ii)~the conformal correction
$-(N{-}1)\cdot 2\xi/(1{-}\xi)^2$ from the
$\ln(a^2{-}|z|^2)$ term in~\eqref{eq:H-H2}; and
(iii)~the Lagrange multiplier, computed from $\Omega$ above and
$\partial^2 J/\partial r^2 = 4(1{+}3\xi)/(1{-}\xi)^3$.
Each contribution is a rational function of~$\xi$.
Combining over the common denominator $(1{-}\xi)^2$
and simplifying (or, equivalently, substituting
$r_E = a\sqrt{\xi}$ into the exact constrained Hessian
of \citealt{Kimura1999}, eq.~(2.8)):
\begin{equation}\label{eq:C1-H2-final}
  C_1 = (N{-}1)\,\frac{1+\xi^2}{(1-\xi)^2}.
\end{equation}
Verified by numerical Hessian to relative precision $10^{-5}$
for $3 \le N \le 12$ and $0 < \xi < 0.9$
(see Code Availability).
\end{proof}

Since $(1+\xi^2)/(1-\xi)^2 > 1$ for all $\xi > 0$,
one has $C_1(\mathbf{H}^2) > N{-}1$ for all $\xi > 0$, confirming the
stabilizing effect of negative curvature.

$N_{\mathrm{crit}}$ increases through integer jumps as the geodesic
radius $\rho$ grows,
with the $7 \to 8$ transition at $\rho/a \approx 0.51$
(equivalently, Poincar\'e disk radius $r_E/a = \sqrt{\xi^*} \approx 0.25$).

\begin{proposition}[Monotonicity of $N_{\mathrm{crit}}$]
\label{prop:ncrit-monotone}
$N_{\mathrm{crit}}(\xi)$ is non-decreasing on $[0, 1)$.
Equivalently, the palindromic thresholds satisfy
$\xi^*(8) < \xi^*(9) < \xi^*(10) < \cdots$.
\end{proposition}

\begin{proof}
The $N$-gon is stable at $\xi$ iff
$C_1(N,\xi) = (N{-}1)\,g(\xi) \ge f_{\max}(N)$,
where $g(\xi) = (1{+}\xi^2)/(1{-}\xi)^2$ is strictly
increasing ($g'(\xi) = 2(1{+}\xi)/(1{-}\xi)^3 > 0$
for $\xi \in [0,1)$).
The threshold is $\xi_N = g^{-1}(h(N))$ with
$h(N) = f_{\max}(N)/(N{-}1)$.
Since $g$ is increasing,
$\xi_{N+1} > \xi_N$ iff $h(N{+}1) > h(N)$.

\emph{Case~1} ($N$ even, $N{+}1$ odd):
$h(N{+}1) - h(N) = (N{+}2)/8 - N^2/(8(N{-}1))
= (N{-}2)/(8(N{-}1)) > 0$ for $N \ge 3$.

\emph{Case~2} ($N$ odd, $N{+}1$ even):
$h(N{+}1) - h(N) = (N{+}1)^2/(8N) - (N{+}1)/8
= (N{+}1)/(8N) > 0$.

Both cases give $h(N{+}1) > h(N)$, so
$\xi_{N+1} > \xi_N$ for all $N \ge 3$.
\end{proof}

\paragraph{Near-$K=0$ behavior.}
Expanding~\eqref{eq:C1_H2} for $N=7$, $m=3$ (see
companion Appendix~A for the algebraic simplification), the
$m=3$ eigenvalue satisfies
\begin{equation}
  \lambda_3(N=7,\,\xi)
    = \frac{12\xi}{(1-\xi)^2}
    = \frac{12|K|\,r_E^2}{(1-\xi)^2}
    = 12\xi + O(\xi^2) > 0 .
  \label{eq:lam3_H2}
\end{equation}
Negative curvature stabilizes the $N=7$ ring at \emph{first} order in
$|K|$: $N_\mathrm{crit}(K)$ stays at $7$ for all $K \le 0$ in the
sense that the formerly-neutral mode becomes strictly stable.

The $7\to 8$ transition is governed by the $N=8$ ring reaching marginal
stability.  Setting $C_1(\mathbf{H}^2,\xi) = m(N{-}m)/2\big|_{m=4} = 8$
gives $7(1+\xi^2)/(1-\xi)^2 = 8$, which simplifies to
\begin{equation}
  \xi^2 - 16\xi + 1 = 0
  \;\Longrightarrow\;
  \xi^* = 8 - 3\sqrt{7} \approx 0.0627 ,
  \label{eq:xi_star}
\end{equation}
exact.  In terms of geodesic ring radius $\rho$, this corresponds to
$\rho/a \approx 0.52$ (or $|K|\rho^2 \approx 0.27$).
The coefficient
\begin{equation}
  \gamma \;\equiv\; \frac{1}{(\xi^*)^2} = 127 + 48\sqrt{7} \approx 254
  \label{eq:gamma}
\end{equation}
carries an algebraic interpretation.
The ring of integers of $\mathbb{Q}(\sqrt{7})$ has fundamental unit
$\varepsilon = 8 + 3\sqrt{7}$, since $(8+3\sqrt{7})(8-3\sqrt{7}) = 64-63 = 1$.
Therefore $\xi^* = 8-3\sqrt{7} = \varepsilon^{-1}$ is the
\emph{inverse of the fundamental unit}---the unique unit below~$1$
in $\mathbb{Z}[\sqrt{7}]$---and $\gamma = (\xi^*)^{-2} = \varepsilon^2$.
The verification identity $(8-3\sqrt{7})^2(127+48\sqrt{7})=1$ is not a
computational coincidence: it is the norm-$1$ property
$\varepsilon^{-2}\cdot\varepsilon^2=1$ in $\mathbb{Z}[\sqrt{7}]$.

The field $\mathbb{Q}(\sqrt{7})$ arises through a specific algebraic
chain: for the $N = 8$ transition, $A = -1$ and $B = 16$, giving
discriminant $\Delta = B^2 - 4A^2 = 256 - 4 = 252 = 2^2 \cdot 3^2
\cdot 7$, with squarefree part~$7$.  The~$7$ enters as the factor
$(N{-}1) = 7$ in the general formula $\Delta = (N{-}1)(N{-}2)^2$
(Proof of~\eqref{eq:field-pattern})---the flat-plane stability
boundary $N = 7$ leaves its algebraic shadow in the discriminant.
The substantive content is the
\emph{palindromic structure}: the threshold polynomial has the form
$A\,\xi^2 + B\,\xi + A = 0$ (same leading and trailing coefficient),
which forces $\xi_1\,\xi_2 = A/A = 1$---the two roots are
multiplicative inverses, so $\xi^*$ is automatically an algebraic
\emph{unit} (an element of norm~$1$ in $\mathbb{Z}[\sqrt{7}]$).
This palindromic property is not generic: it reflects the inversion
symmetry $\xi\leftrightarrow 1/\xi$ of the Poincar\'e disk model,
which in turn is the restriction to the real axis of the conformal
automorphism $z \mapsto a^2/\bar z$ of the disk.  The stability
threshold is a unit \emph{because} the hyperbolic energy is
conformally covariant (Proposition~\ref{prop:mobius}): the palindromic
polynomial encodes the same M\"obius symmetry at the level of
algebraic number theory.

\section{Platonic extensions}
\label{sec:platonic}
%% ====================================================================

The Havelock eigenvalue decomposition extends from regular $N$-gons
($\mathbb{Z}_N$ symmetry) to Platonic solid configurations on~$\mathbf{S}^2$
with non-abelian symmetry group $G \subset \mathrm{SO}(3)$.
This section extends the Havelock decomposition to Platonic solids
on~$\mathbf{S}^2$.

\subsection{The generalized Havelock formula for Platonic solids}
\label{sec:platonic-havelock}

\begin{theorem}[Generalized Havelock formula]\label{thm:platonic-havelock}
Let $G \subset \mathrm{SO}(3)$ act transitively on $N$ vertices
$\{v_0, \ldots, v_{N-1}\} \subset \mathbf{S}^2$, and let
$K(d) = 1/(4\sin^2(d/2))$ be the $\mathbf{S}^2$ interaction kernel.
For each $G$-irrep~$\rho$ that is the restriction of the $\mathrm{SO}(3)$
representation $D^j$ to~$G$ (with $D^j|_G$ irreducible), the eigenvalue
of the zero-sum interaction matrix in the $\rho$-isotypic sector is
$-T_\rho$, where
\begin{equation}\label{eq:platonic-havelock}
  T_\rho = \sum_{k \neq 0} K(d_{0k})\bigl[1 - P_j(\cos d_{0k})\bigr],
\end{equation}
and $P_j$ is the Legendre polynomial of degree~$j$.
\end{theorem}

\begin{proof}
By Schur's lemma, the interaction matrix~$K$ is $G$-invariant
($K_{g \cdot j,\, g \cdot k} = K_{j,k}$ for all $g \in G$)
and hence acts as a scalar on each $G$-irrep in the permutation
representation.  The zonal spherical function of $D^j$ on the
homogeneous space $\mathbf{S}^2 = \mathrm{SO}(3)/\mathrm{SO}(2)$
is the Legendre polynomial~$P_j$ (Vilenkin, 1968).  When
$D^j|_G = \rho$ is irreducible, the vector $P_j(\cos d_{0k})$
restricted to the orbit is an eigenvector of~$K$.  Evaluating
$K \cdot P_j$ at vertex~$0$ and using the zero-sum property
$K_{00} = -C_1$ gives the eigenvalue $-T_\rho$.
\end{proof}

Verified numerically for the tetrahedron ($j \le 1$, $A_4$),
octahedron ($j \le 2$, $S_4$), icosahedron ($j \le 3$, $A_5$),
and cube ($j \le 2$, $S_4$); residuals $< 10^{-16}$
(77~computational tests, code in \texttt{proofs/platonic\_havelock.py}
and \texttt{proofs/e8\_casimir\_bridge.py}).

\smallskip
\noindent
\textbf{Stability pattern on $\mathbf{S}^2$.}\enspace
Among the five Platonic solids, the tetrahedron ($N = 4$, $A_4 \to E_6$),
octahedron ($N = 6$, $S_4 \to E_7$), and icosahedron
($N = 12$, $A_5 \to E_8$) are stable; the cube ($N = 8$) and
dodecahedron ($N = 20$) are unstable.  Every eigenvalue degeneracy
matches an irrep dimension of~$G$.  All polygon rings ($N \ge 4$ at
the equator) are unstable on~$\mathbf{S}^2$.  The Onsager principle
selects the icosahedron as the largest stable Platonic solid.

\subsection{Tangent Hessian eigenvalue pairing}
\label{sec:hessian-pairing}

\begin{theorem}[Eigenvalue pairing]\label{thm:hessian-pairing}
For a $G$-symmetric configuration of $N$ vertices on~$\mathbf{S}^2$,
the tangent-space Hessian eigenvalues come in pairs
$(\lambda_-(\rho),\, \lambda_+(\rho))$ summing to a universal constant:
\begin{equation}\label{eq:hessian-pairing}
  \lambda_+(\rho) + \lambda_-(\rho) = \frac{N-1}{2}
  \qquad \text{for all $G$-irreps $\rho$}.
\end{equation}
\end{theorem}

\begin{proof}
\emph{Step~1 (Laplacian identity).}
For $f(d) = -\ln\sin(d/2)$,
$f' = -\tfrac{1}{2}\cot(d/2)$ and
$f'' = 1/(4\sin^2(d/2))$.
The radial Laplacian on~$\mathbf{S}^2$,
$\Delta f = f'' + \cot(d)\,f'$, gives
\[
  \Delta f = \frac{1 - \cos d}{4\sin^2(d/2)} = \frac{1}{2},
\]
a constant independent of~$d$.

\emph{Step~2 (Per-vertex trace).}
The $2 \times 2$ diagonal block~$H_{kk}$ of the tangent Hessian has
$\operatorname{Tr}(H_{kk}) = \sum_{j \neq k} \frac{1}{2} = \frac{N-1}{2}$.

\emph{Step~3 (Schur decomposition).}
The Hessian~$H$ is $G$-equivariant.  The tangent representation is
$\operatorname{Ind}_H^G(\omega) \oplus \operatorname{Ind}_H^G(\bar\omega)$
where $H = \mathrm{Stab}(v_0)$ and $\omega$ is the tangent character.
By Frobenius reciprocity,
$\operatorname{mult}(\rho,\, \operatorname{Ind}(\omega))
= \dim\operatorname{Hom}_H(\rho|_H,\, \omega)$.
For Platonic stabilizers $H = \mathbb{Z}_n$ ($n \ge 3$), each
$G$-irrep restricts to~$H$ with at most one copy of each
$\mathbb{Z}_n$ character, giving multiplicity~$1$ in each induced
representation, total~$2$.
Schur's lemma then gives
$H|_\rho = A_\rho \otimes \mathrm{Id}_{d_\rho}$ where
$A_\rho$ is a $2 \times 2$ matrix with eigenvalues
$\lambda_\pm(\rho)$.

\emph{Step~4 (Stabilizer isotropy).}
The vertex stabilizer $H_v \supset \mathbb{Z}_n$ ($n \ge 3$ for
all Platonic solids) acts on $T_{v_k}\mathbf{S}^2$ by rotation,
forcing $H_{kk} = \frac{N-1}{4}\,I_2$.  By $G$-transitivity,
the per-vertex trace distributes uniformly across all irreps:
$\operatorname{Tr}(A_\rho)
= \lambda_+(\rho) + \lambda_-(\rho) = (N{-}1)/2$.
\end{proof}

\begin{corollary}
$\operatorname{Tr}(H) = N(N-1)/2$.
\end{corollary}

\noindent
Verified for the tetrahedron ($N = 4$: pairs $(0, 3/2)$ and
$(3/4, 3/4)$), octahedron ($N = 6$: pairs $(0, 5/2)$ and
$(1/2, 2)$), and icosahedron ($N = 12$: pairs $(0, 11/2)$,
$(1/2, 5)$, and $(5/4, 17/4)$).

\subsection{The icosahedral bridge identity}
\label{sec:bridge-identity}

The icosahedron has three classes of inter-vertex distances on the
unit~$\mathbf{S}^2$: five nearest neighbours at
$d_1 = \cos^{-1}(1/\sqrt{5})$, five far neighbours at
$d_2 = \cos^{-1}(-1/\sqrt{5})$, and one antipodal vertex at $d_3 = \pi$.

\begin{theorem}[Bridge identity]\label{thm:bridge}
The Havelock eigenvalue for $\mathrm{SO}(3)$ spin~$j$ on the
icosahedron is
\begin{equation}\label{eq:bridge}
  \lambda_j = 5K_1\, P_j\!\Bigl(\frac{1}{\sqrt{5}}\Bigr)
            + 5K_2\, P_j\!\Bigl(-\frac{1}{\sqrt{5}}\Bigr)
            + \frac{1}{4}(-1)^j,
\end{equation}
where $K_1 = (5+\sqrt{5})/8$ and $K_2 = (5-\sqrt{5})/8$.
\end{theorem}

\noindent
The three terms correspond to the three distance classes.
The golden ratio enters structurally:
$K_1 - K_2 = \sqrt{5}/4$ controls the odd-$j$ splitting,
$K_1 + K_2 = 5/4$ sets the even-$j$ level, and the argument
$c = 1/\sqrt{5} = 1/(2\varphi - 1)$ is the icosahedral
golden angle.

For $j = 0, 1, 2, 3$, the formula gives the exact half-integers
$\lambda = 13/2,\; 1,\; -1,\; -3/2$, with per-vertex sum
$C_1 = 13/2$.  This is the icosahedral analogue of the polygon
formula $\lambda_m = (N{-}1) - m(N{-}m)/2$.

\section{Algebraic structure of the stability boundary}

This section develops four algebraic results about the

\subsection{The palindromic quadratic and field extensions}
\label{sec:palindromic-quad}

The palindromic structure observed for $N = 8$
(Section~\ref{sec:hyperbolic}) holds at every
$N \to (N{+}1)$ transition.
Setting $C_1(\xi^*) = m(N{-}m)/2$ and clearing
denominators gives the general palindromic quadratic:
\begin{equation}\label{eq:palindromic}
  A\,\xi^2 + B\,\xi + A = 0,
  \quad
  A = (N{-}1) - \tfrac{m(N{-}m)}{2},
  \quad
  B = m(N{-}m),
  \quad
  m = \lfloor N/2\rfloor.
\end{equation}
The coefficient of~$\xi^2$ equals the constant term (both are~$A$),
so the polynomial is palindromic in the sense of
Section~\ref{sec:hyperbolic}, forcing
$\xi_1\xi_2 = 1$ and hence $\xi^* = 1/\xi^{*\prime}$.
Since $A < 0$ for $N \ge 8$ and $B > 0$, the smaller positive root is
$\xi^* = (B/2 - \sqrt{(B/2)^2 - A^2})\,/\,|A|$, and its field extension
is determined by the squarefree part of the discriminant
$(B/2)^2 - A^2 \in \mathbb{Z}$:

\toprule

\midrule

8  & 0.0627 & $\mathbb{Q}(\sqrt{7})$ & 7 & $8 - 3\sqrt{7}$ \\
9  & 0.1010 & $\mathbb{Q}(\sqrt{6})$ & 6 & $5 - 2\sqrt{6}$ \\

\bottomrule
\end{tabular}
\end{center}

\noindent
The discriminant's squarefree part $D$ follows the pattern
\begin{equation}\label{eq:field-pattern}
  D = \begin{cases}
    \mathrm{sq\_free}(N-1) & N \text{ even}, \\
    \mathrm{sq\_free}(N-3) & N \text{ odd},
  \end{cases}
\end{equation}
for all $N \ge 8$; the field pattern follows from
elementary factorization of the palindromic
discriminant~\eqref{eq:palindromic}.

\begin{proof}[Proof of~\eqref{eq:field-pattern}]
The palindromic quadratic $A\xi^2 + B\xi + A = 0$ has discriminant
$\Delta = B^2 - 4A^2$.  Using $A = (N{-}1) - m(N{-}m)/2$ and
$B = m(N{-}m)$:
\[
  \Delta = m^2(N{-}m)^2 - 4\bigl[(N{-}1) - m(N{-}m)/2\bigr]^2
  = 4(N{-}1)\bigl[m(N{-}m) - (N{-}1)\bigr].
\]

\emph{Even $N$ ($m = N/2$):}
$m(N{-}m) = N^2/4$, so
$m(N{-}m) - (N{-}1) = N^2/4 - N + 1 = (N{-}2)^2/4$.
Therefore
$\Delta = 4(N{-}1)(N{-}2)^2/4 = (N{-}1)(N{-}2)^2$.
The squarefree part of $\Delta$ equals
$\mathrm{sq\_free}(N{-}1)$, since $(N{-}2)^2$ is a perfect square.
The threshold $\xi^*$ involves $\sqrt\Delta$, so
$\xi^* \in \mathbb{Q}(\sqrt{\mathrm{sq\_free}(N{-}1)})$.

\emph{Odd $N$ ($m = (N{-}1)/2$):}
$m(N{-}m) = (N{-}1)(N{+}1)/4 = (N^2{-}1)/4$, so
$m(N{-}m) - (N{-}1) = (N^2{-}1)/4 - (N{-}1) = (N{-}1)(N{-}3)/4$.
Therefore
$\Delta = 4(N{-}1) \cdot (N{-}1)(N{-}3)/4 = (N{-}1)^2(N{-}3)$.
The squarefree part of $\Delta$ equals
$\mathrm{sq\_free}(N{-}3)$, since $(N{-}1)^2$ is a perfect square.
Hence $\xi^* \in \mathbb{Q}(\sqrt{\mathrm{sq\_free}(N{-}3)})$.
\end{proof}

\subsection{Algebraic integers, Pell units, and the golden ratio}
The threshold $\xi^*(N)$ is an algebraic integer---i.e., it
satisfies a \emph{monic} polynomial $\xi^2 - c\,\xi + 1 = 0$ with
$c \in \mathbb{Z}$---if and only if $B/A \in \mathbb{Z}$.

\begin{proposition}[Algebraic-integer classification]
\label{prop:alg-int}
The threshold $\xi^*(N)$ is an algebraic integer if and only if
$N \in \{8, 9, 11, 15, 23\}$.
\end{proposition}

\begin{proof}
For odd $N \ge 9$: $A = -(N{-}1)(N{-}7)/8$,
$B = (N{-}1)(N{+}1)/4$, so $B/A = -2(N{+}1)/(N{-}7)$.
This is an integer iff $(N{-}7) \mid 2(N{+}1)$.
Since $2(N{+}1) = 2(N{-}7) + 16$, the condition reduces to
$(N{-}7) \mid 16$.  The positive divisors of~$16$ giving odd
$N \ge 9$ are $N{-}7 \in \{2, 4, 8, 16\}$, i.e.,
$N \in \{9, 11, 15, 23\}$.
For even $N = 2k \ge 8$: $A = (N^2{-}8N{+}8)/8 = (k^2{-}4k{+}2)/2$
and $B = N^2/4 = k^2$.
The ratio $B/A = 2k^2/(k^2{-}4k{+}2)$, which is an integer
iff $(k^2{-}4k{+}2) \mid 2k^2$.
Since $2k^2 = 2(k^2{-}4k{+}2) + 8k{-}4$, this reduces to
$(k^2{-}4k{+}2) \mid (8k{-}4)$.
For $k \ge 5$: $k^2 - 4k + 2 \ge 7 > 0$ and
$k^2 - 4k + 2 > 8k - 4$ iff $k^2 - 12k + 6 > 0$ iff $k \ge 12$.
Checking $k = 4$ ($N = 8$): $k^2{-}4k{+}2 = 2$, $8k{-}4 = 28$,
$2 \mid 28$~$\checkmark$.
For $k = 5,6,\ldots,11$: direct check shows none divide.
For $k \ge 12$: the divisor exceeds the dividend, impossible.
\end{proof}

The differences between consecutive algebraic-integer values
are $\{1, 2, 4, 8\}$---powers of~$2$---reflecting the
binary structure of the divisors of~$16$.

For these five values, $\xi^*$ satisfies the Pell equation.
Since the monic polynomial $\xi^2 - c\,\xi + 1 = 0$ forces
$\xi\cdot\xi' = 1$, writing $\xi^* = a + b\sqrt{D}$ with
$a, b \in \mathbb{Z}$ (or half-integers for $D \equiv 1 \pmod{4}$)
gives $a^2 - D b^2 = 1$: the classical Pell equation.
Each threshold is an explicit power of the inverse fundamental
unit $\varepsilon$ of $\mathbb{Q}(\sqrt{D})$:

\begin{tabular}{ccllcc}
\toprule

\midrule

  & $16$ & $+1$ \\
$9$  & $6$ & $5 + 2\sqrt{6}$       & $\varepsilon^{-1}$
  & $10$ & $+1$ \\
$11$ & $2$ & $1 + \sqrt{2}$\,(silver ratio) & $\varepsilon^{-2}$
  & $6$  & $-1$ \\
$15$ & $3$ & $2 + \sqrt{3}$        & $\varepsilon^{-1}$
  & $4$  & $+1$ \\
$23$ & $5$ & $\varphi = (1{+}\sqrt{5})/2$\,(golden ratio)

The exponent is $k = 1$ when $\mathrm{Norm}(\varepsilon) = +1$
(the fundamental unit solves $x^2 - Dy^2 = 1$) and $k = 2$ when
$\mathrm{Norm}(\varepsilon) = -1$ (the fundamental unit solves
$x^2 - Dy^2 = -1$; only its square has norm~$+1$, as required
by the palindromic constraint $\xi\cdot\xi' = 1$).

\paragraph{The golden ratio as the terminal case.}
The last algebraic-integer threshold ($N = 23$) has
$\xi^*(23) = \varphi^{-2} = (3 - \sqrt{5})/2$, where
$\varphi = (1{+}\sqrt{5})/2$ is the golden ratio.  Its monic
polynomial $\xi^2 - 3\xi + 1 = 0$ has the smallest integer
trace ($c = 3$) compatible with two distinct positive roots;
$c = 2$ gives the degenerate $(\xi{-}1)^2 = 0$.  The golden
ratio is thus the algebraically \emph{simplest} non-trivial
palindromic unit.
Equivalently, $\varphi^{-1}$ is the unique algebraic unit
in $(0,1)$ satisfying $1 - x = x^2$, which gives
$\sum_{k=1}^{\infty}\varphi^{-k}/k
= -\ln(1{-}\varphi^{-1}) = -\ln\xi^*(23) = 2R(5)$:
the generating series of the Fibonacci weights evaluates to
twice the regulator.

\paragraph{Binary subdivision.}
The odd algebraic-integer values $N = 7 + 2^k$ ($k = 1, 2, 3, 4$)
have traces $c(k) = 2 + 2^{4-k}$: each doubling of $(N{-}7)$
halves the algebraic excess $c - 2$.  The sequence
$c \in \{10, 6, 4, 3\}$ converges to the golden-ratio value
$c = 3$ as $k \to 4$.  The binary subdivision terminates because
$c = 2$ is degenerate; no further algebraic-integer thresholds
exist beyond $N = 23$.

\paragraph{Continued fraction structure.}
\begin{proposition}[CF of palindromic units]\label{prop:cf-palindromic}
If\, $\xi$ satisfies $\xi^2 - c\,\xi + 1 = 0$ with integer
$c \ge 3$ and $\xi \in (0,1)$, then
$1/\xi = [c{-}1;\;\overline{1,\, c{-}2}]$.
For $c = 3$: the period degenerates to $[1]$ (all ones), which
is the continued fraction of the golden ratio~$\varphi$.
\end{proposition}

\begin{proof}
\emph{Step~1.}  $a_0 = \lfloor 1/\xi\rfloor = c - 1$, since
$(c{-}2)^2 \le c^2 - 4$ for $c \ge 2$.
\emph{Step~2.}  The first remainder is
$r_0 = (\sqrt{c^2{-}4} - (c{-}2))/2$; its reciprocal
$1/r_0 = (\sqrt{c^2{-}4} + (c{-}2))/(2(c{-}2))$ satisfies
$1 \le 1/r_0 < 2$ (from $4(2c{-}5)(c{-}2) > 0$ for $c \ge 3$),
so $a_1 = 1$.
\emph{Step~3.}  $1/r_1 = 1/\xi - 1$, giving $a_2 = c - 2$.
\emph{Step~4.}  $r_2 = 1/\xi - (c{-}1) = r_0$ (periodicity).
\end{proof}

Applied to the five algebraic-integer cases:
$1/\xi^*(8) = [15;\;\overline{1, 14}]$,
$1/\xi^*(9) = [9;\;\overline{1, 8}]$,
$1/\xi^*(11) = [5;\;\overline{1, 4}]$,
$1/\xi^*(15) = [3;\;\overline{1, 2}]$,
$1/\xi^*(23) = [2;\;\overline{1}] = [2; 1, 1, 1, \ldots]$
(the golden ratio).

The $N = 10$ case gives the unique rational threshold
$\xi^*(10) = 1/7 \in \mathbb{Q}$
($N{-}1 = 9 = 3^2$, so $D = 1$).  A second rational threshold
occurs at $N = 19$: $\xi^*(19) = 1/3$
($N{-}3 = 16 = 4^2$, so $D = 1$).  Every even transition lives
in at most a degree-$2$ extension of $\mathbb{Q}$; the tower
terminates at~$\mathbb{Q}$ whenever
$\mathrm{sq\_free}(N{-}1) = 1$ (even $N$) or
$\mathrm{sq\_free}(N{-}3) = 1$ (odd $N$).

\paragraph{The regulator as physical accessibility.}
The regulator $R(D) = \ln\varepsilon$ of $\mathbb{Q}(\sqrt{D})$
measures the ``size'' of the unit group.  For the five
algebraic-integer fields:
$R(7) = 2.77$, $R(6) = 2.29$, $R(3) = 1.32$, $R(2) = 0.88$,
$R(5) = 0.48$.
Since $\xi^* = \varepsilon^{-k}$, we have
$\ln(1/\xi^*) = k\,R(D)$: the regulator is the logarithmic
distance from the stability boundary to the flat-plane limit
$\xi = 0$.  Larger $R$ means smaller $\xi^*$, i.e., more
curvature is needed to reach the transition.
The $N = 8$ transition ($R = 2.77$, $\xi^* = 0.063$) requires
the most curvature; the $N = 23$ transition ($R = 0.48$,
$\xi^* = 0.382$) requires the least---making
$\varphi^{-2}$ the most experimentally accessible threshold.

\paragraph{The growth law: $f_{\mathrm{crit}} - b \sim N^2/24$ from $B_2 = 1/6$.}
At the $N$-gon threshold on~$\mathbf{H}^2$, the curvature
coefficient equals the Casimir: $C_1(\xi^*) = f_{\mathrm{crit}}(N)$.
The \emph{stability residual} $f_{\mathrm{crit}} - b(N)$---the amount
by which the Casimir exceeds the mode-averaged offset---satisfies
\begin{equation}\label{eq:growth-law}
  f_{\mathrm{crit}}(N) - b(N)
  = \tfrac{1}{3}\,f_{\mathrm{crit}}(N) + O(N)
  \sim \frac{N^2}{24} \quad (N \to \infty),
\end{equation}
where $b(N) = N(N{+}1)/12 - \ln 2 + (\ln N)/(N{-}1)$
is the exact offset (verified to $10^{-16}$ for all $N = 3,\ldots,24$).
The coefficient $1/3$ is \emph{exact}, arising from $B_2 = 1/6$.

\begin{proof}[Proof of the $b(N)$ identity]
The offset $b(N) = C_1(\xi) - \log(2\sinh\rho)$ for
$\xi = \tanh^2(\rho/2)$ is $\rho$-independent (since $C_1$ and
$\log(2\sinh\rho)$ both grow as $\rho$ for large~$\rho$, with
the same linear coefficient).
It decomposes as $b(N) = \langle f_m\rangle + \langle S_m\rangle$,
where $\langle f_m\rangle$ is the mean Casimir and
$\langle S_m\rangle$ is the mean log-sine contribution.

\emph{Mean Casimir.}
$\langle f\rangle = \frac{1}{N{-}1}\sum_{m=1}^{N-1}
\frac{m(N{-}m)}{2}
= \frac{1}{2(N{-}1)}\bigl(N\cdot\frac{N(N{-}1)}{2}
  - \frac{(N{-}1)N(2N{-}1)}{6}\bigr)
= \frac{N(N{+}1)}{12}.$

\emph{Mean log-sine (Gauss).}
The identity $\prod_{p=1}^{N-1}\sin(\pi p/N) = N/2^{N-1}$
(a consequence of the Chebyshev factorization
$U_{N-1}(\cos\theta) = \sin(N\theta)/\sin\theta
= 2^{N-1}\prod_{k=1}^{N-1}(\cos\theta - \cos(k\pi/N))$
evaluated at $\theta \to 0$) gives
\[
  \langle S\rangle
  = \frac{1}{N{-}1}\sum_{p=1}^{N-1}\log\sin\frac{\pi p}{N}
  = \frac{\ln N - (N{-}1)\ln 2}{N{-}1}
  = \frac{\ln N}{N{-}1} - \ln 2.
\]
Combining: $b(N) = N(N{+}1)/12 - \ln 2 + (\ln N)/(N{-}1)$.
The leading term $\langle f\rangle = N^2/12 + O(N)$ equals
$B_2 N^2/2$, the Euler--Maclaurin coefficient for the discrete
log-sine kernel on $\mathbb{Z}/N\mathbb{Z}$.
Therefore
\[
  f_{\mathrm{crit}} - b(N)
  = \frac{N^2}{8} - \frac{N^2}{12} + O(N)
  = \frac{N^2}{24} + O(N). \qedhere
\]
\end{proof}

\noindent
The empirical fit
$f_{\mathrm{crit}} - b = 0.312\,f_{\mathrm{crit}} - 0.17$
(12 data points, $N = 8$--$19$, $R^2 = 0.9998$) confirms the
predicted asymptotic ratio $(f_{\mathrm{crit}} - b)/f_{\mathrm{crit}}
\to 1/3$.
The residuals show an even/odd oscillation of amplitude~$0.06$,
reflecting the different Casimir structures of even-$N$
(half-mode is a singleton) versus odd-$N$ (all palindromic pairs).

The growth law resolves a coordinate artifact:
in the Poincar\'e-disk parameter $\xi$, the thresholds
$\xi^*(N)$ grow super-exponentially (since $C_1 \propto (1{-}\xi)^{-2}$),
suggesting a barrier.
In terms of $f_{\mathrm{crit}}$, the thresholds are \emph{uniformly spaced}:
each additional polygon type requires an additional $\Delta f \approx N$
of Casimir capacity, with no exponential barrier.
On any surface $\Gamma\backslash\mathbf{H}^2$, the maximum Casimir
is bounded by $C_1(\rho_{\mathrm{inj}})$, giving
$f_{\mathrm{crit}}(N_{\max}) \le C_1(\rho_{\mathrm{inj}})$ and hence
$N_{\max} \approx \sqrt{8\,C_1(\rho_{\mathrm{inj}})}$.

\begin{remark}[The Bernoulli--Dedekind connection]
\label{rmk:bernoulli-dedekind}
The same $B_2 = 1/6$ enters the palindromic growth law
($\langle D\rangle = N^2/12$), the Dedekind eta function
($\eta = q^{1/24}\prod$, since $1/24 = B_2/(2\cdot 2!)$),
and the Ramanujan sum
$\zeta(-1) = -B_2/2 = -1/12$.
All three are instances of the Euler--Maclaurin formula for the
log-sine kernel at different levels:
the mode sum on $\mathbb{Z}/N\mathbb{Z}$ (this theorem),
the modular parameter on $\mathbb{Z}$ (the eta function), and the
spectral sum (the regularised integers).
The Bolza modular form $f = \eta(8z)\eta(16z)$ from
the Bolza modular analysis and the palindromic growth
coefficient~$1/3$ share a common origin in $B_2 = 1/6$.
The connection is made precise through the $\mathbb{Z}_N$ orbifold:
the twisted-sector Casimir energy $E_0(m) = -c/(24N) + cm(N{-}m)/(24N^2)$
has mode average $\langle m(N{-}m)\rangle = N(N{+}1)/6$, whose leading
coefficient $N^2/6$ contains the factor $1/6 = B_2$.
The aliasing $\langle D\rangle = \langle m(N{-}m)/2\rangle - N/12 = N^2/12$
is the mode-averaged orbifold Casimir of the Havelock eigenvalue,
minus a subleading correction.
\end{remark}

\subsection{Geodesics on the modular surface}

\begin{proposition}[Galois group as conformal inversion group]
\label{prop:galois-inversion}
The composite field
$K = \mathbb{Q}(\sqrt{2}, \sqrt{3}, \sqrt{5}, \sqrt{6}, \sqrt{7})$
contains all five algebraic-integer thresholds.  Its Galois group
$\mathrm{Gal}(K/\mathbb{Q}) \cong (\mathbb{Z}/2)^4$ acts on the
threshold set by
$\sigma_D\colon \xi^*(N) \mapsto 1/\xi^*(N)$
---the conformal inversion of the Poincar\'e disk.
The Galois group of the stability threshold fields \emph{is} the
conformal inversion group, realised algebraically.
\end{proposition}

\begin{proof}
Each $\xi^*(N)$ lies in $\mathbb{Q}(\sqrt{D})$ for some squarefree
$D$ (Proposition~\ref{prop:alg-int}).  The Galois automorphism
$\sigma_D\colon \sqrt{D} \mapsto -\sqrt{D}$ sends $\xi^* = a + b\sqrt{D}$
to $a - b\sqrt{D} = 1/\xi^*$ (the palindromic conjugate).
Since the palindromic property encodes conformal inversion
$\xi \leftrightarrow 1/\xi$ (Section~\ref{sec:algebraic-structure}),
$\sigma_D$ acts as conformal inversion on each threshold.
The composite field $K$ has
$\mathrm{Gal}(K/\mathbb{Q}) \cong \prod_{D \in \{2,3,5,6,7\}}
\mathbb{Z}/2 \cong (\mathbb{Z}/2)^4$
(four independent generators, since $\sqrt{6} = \sqrt{2}\sqrt{3}$
is algebraically dependent), and each generator acts by conformal
inversion on the thresholds in its field component.
\end{proof}

\paragraph{Stability thresholds as closed geodesics.}
The monic palindromic polynomial $\xi^2 - T\,\xi + 1 = 0$
with $T \in \mathbb{Z}$ is simultaneously the characteristic polynomial
of a hyperbolic element $\gamma \in \mathrm{SL}(2,\mathbb{Z})$
with trace $T = \mathrm{tr}(\gamma)$.  The eigenvalues of $\gamma$
are $\varepsilon$ and $\varepsilon^{-1} = \xi^*$---exactly the
stability threshold and its conformal inverse.
Each such $\gamma$ defines a primitive conjugacy class of
$\mathrm{SL}(2,\mathbb{Z})$, hence a primitive closed geodesic
on the modular surface $\mathrm{SL}(2,\mathbb{Z})\backslash
\mathbf{H}^2$ of length
$\ell = 2\,\mathrm{arccosh}(T/2) = 2\ln\varepsilon$.

\begin{proposition}[Geodesic structural observation]
\label{prop:geodesic-correspondence}
The five algebraic-integer stability thresholds correspond to
five primitive closed geodesics on the modular surface.
For odd $N \in \{9, 11, 15, 23\}$ the traces are
$T = 2 + 16/(N{-}7)$
(equivalently, $N = 7 + 16/(T{-}2)$);
for $N = 8$ the trace is $c = 16$ (direct computation).
The full discriminant $\Delta = T^2 - 4$ decomposes as
$\Delta = f^2 D$ where $D$ is the squarefree part and $f$ is
the conductor:
\begin{center}
\begin{tabular}{ccccclll}
\toprule
$N$ & $T$ & $\Delta$ & $f$ & $D$ &
  Geodesic & $\mathrm{SL}(2,\mathbb{Z})$ \\
\midrule
$23$ & $3$  & $5$   & $1$ & $5$ &
  $\ell = 1.92$ & $\bigl[\begin{smallmatrix}2&1\\1&1\end{smallmatrix}\bigr]$
  \emph{(Fibonacci)} \\
$15$ & $4$  & $12$  & $2$ & $3$ &
  $\ell = 2.63$ & $\bigl[\begin{smallmatrix}3&2\\1&1\end{smallmatrix}\bigr]$ \\
$11$ & $6$  & $32$  & $4$ & $2$ &
  $\ell = 3.53$ & $\bigl[\begin{smallmatrix}5&2\\2&1\end{smallmatrix}\bigr]$ \\
$9$  & $10$ & $96$  & $4$ & $6$ &
  $\ell = 4.58$ & $\bigl[\begin{smallmatrix}9&2\\4&1\end{smallmatrix}\bigr]$ \\
$8$  & $16$ & $252$ & $6$ & $7$ &
  $\ell = 5.54$ & $\bigl[\begin{smallmatrix}15&7\\2&1\end{smallmatrix}\bigr]$ \\
\bottomrule
\end{tabular}
\end{center}
The terminal case $N = 23$ ($T = 3$) corresponds to
$Q^2 = \bigl[\begin{smallmatrix}2&1\\1&1\end{smallmatrix}\bigr]$,
the square of the standard Fibonacci matrix
$Q = \bigl[\begin{smallmatrix}1&1\\1&0\end{smallmatrix}\bigr]$;
the eigenvalues of $Q^2$ are $\varphi^2$ and $\varphi^{-2} = \xi^*(23)$.
$N = 23$ is the unique case with conductor $f = 1$:
the threshold $\xi^*$ is a unit in the maximal order $O_K$, not
merely in a suborder.
\end{proposition}

\begin{proof}
The correspondence is a consequence
of the palindromic structure~\eqref{eq:palindromic}.
Any monic palindromic quadratic with integer trace~$T$
and constant term~$1$ is \emph{automatically} the
characteristic polynomial of a matrix in
$\mathrm{SL}(2,\mathbb{Z})$:
$M = \bigl[\begin{smallmatrix}a&b\\c'&d\end{smallmatrix}\bigr]$
with $\det M = ad - bc' = 1$ and $\mathrm{tr}\,M = T$.
(Existence: the companion matrix
$\bigl[\begin{smallmatrix}0&-1\\1&T\end{smallmatrix}\bigr]$
has trace~$T$, determinant~$1$.)
Each such matrix defines a primitive hyperbolic conjugacy
class, hence a primitive closed geodesic of length
$\ell = 2\,\mathrm{arccosh}(T/2) = 2\ln\varepsilon$.
For odd~$N$, the trace is $T = 2(N{+}1)/(N{-}7) = 2 + 16/(N{-}7)$
(Proposition~\ref{prop:alg-int}).
For even~$N = 8$ the ratio $B/A = 16$ directly from the palindromic
polynomial (the odd-$N$ formula does not apply).
The conductor $f$ satisfies $T^2 - 4 = f^2 D$ where
$D = \mathrm{sq\_free}(T^2 - 4)$.
Among the five traces $T \in \{16, 10, 6, 4, 3\}$,
the discriminants $T^2 - 4 = \{252, 96, 32, 12, 5\}$
have squarefree parts $\{7, 6, 2, 3, 5\}$ and
conductors $f = \{6, 4, 4, 2, 1\}$.
Only $T = 3$ ($N = 23$) gives $f = 1$,
since $5$ is the only squarefree discriminant.
The structural reason for the correspondence is:
the conformal inversion symmetry of the Havelock
Hessian produces exactly the palindromic polynomials
that define hyperbolic conjugacy classes;
the algebraic-integer condition (Proposition~\ref{prop:alg-int})
restricts the traces to the values $(N{-}7) | 16$.
Each trace $c$ gives a primitive hyperbolic element. Indeed, $T_2(x) = 2x^2{-}1$ gives $T_2^{-1}(c) = \sqrt{(c{+}1)/2}$, which is irrational for all $c \in \{3, 4, 6, 10, 16\}$ (none of $(c{+}1)/2$ is a perfect square). No $T_k$ for $k \ge 3$ yields these values ($T_3(2) = 26 > 16$). All five are primitive.
\end{proof}

\begin{remark}[Automorphic correction on quotient surfaces]
\label{rmk:automorphic-correction}
On a quotient surface $\Gamma\backslash\mathbf{H}^2$, the
point vortex interaction involves the automorphic Green's function
$G_\Gamma(z,w) = \sum_{\gamma \in \Gamma} G(z, \gamma w)$.
Each non-trivial $\gamma$ contributes a correction to $C_1$
whose leading term is proportional to
$\mathrm{csch}^2(\ell_\gamma/2) = 4/\Delta_\gamma$,
where $\Delta_\gamma = \mathrm{tr}(\gamma)^2 - 4$ is the
discriminant of the geodesic
(since $\sinh(\ell/2) = \sqrt{\Delta}/2$).
The conductor decomposition
$\Delta = f^2 D$ gives
$\mathrm{csch}^2(\ell/2) = 4/(f^2 D)$:
the correction strength is the inverse discriminant,
factored through the conductor.
The palindromic symmetry~\eqref{eq:palindromic} is generically
\emph{broken} on quotient surfaces (preserved only when $\Gamma$
is invariant under conformal inversion), so stability thresholds
on general quotients need not be palindromic units.
\end{remark}

\begin{remark}[Topological sensitivity and the conductor]
\label{rmk:conductor-sensitivity}
The automorphic correction strength $4/\Delta = 4/(f^2 D)$
decreases rapidly with the conductor~$f$:
for $N = 23$ ($f = 1$, $\Delta = 5$) it is $0.80$, while
for $N = 8$ ($f = 6$, $\Delta = 252$) it is $0.016$.
The conductor thus measures the
\emph{topological sensitivity} of the stability threshold:
small $f$ (the golden ratio case) couples strongly to the
global topology of the surface, while large $f$ (the $N = 8$
case) is effectively insensitive.
The conductor is an intrinsic invariant of the $N \to (N{+}1)$
transition---it depends on the $\mathbb{Z}_N$ Casimir but not
on the surface.
\end{remark}

\paragraph{Higher-genus generalisation.}
The palindromic structure extends to vortex stability on
\emph{any} arithmetic hyperbolic surface.
If the Fuchsian group $\Gamma$ has trace field $K$ of degree $d$
over~$\mathbb{Q}$, the geodesic trace $B$ lies in~$K$, and
eliminating $B$ from $\xi^2 - B\xi + 1 = 0$ and the degree-$d$
minimal polynomial of $B$ yields a \emph{palindromic} polynomial
of degree $2d$ in $\xi$ over~$\mathbb{Q}$.
(Palindromic because $\xi \to 1/\xi$ fixes $B = \xi + 1/\xi$.)
Its roots are algebraic units of degree~$2d$.

\begin{conjecture}[Higher-genus palindromic hierarchy]
\label{conj:higher-genus}
On an arithmetic surface $\Gamma\backslash\mathbf{H}^2$ with
trace field of degree $d$, the vortex stability threshold
satisfies a palindromic polynomial of degree $2d$
over~$\mathbb{Q}$.
For $d = 1$ (the modular surface): palindromic quadratics
and Pell units (the single-ring results of this paper).
For $d = 2$ (the Bolza surface, genus~$2$, trace field
$\mathbb{Q}(\sqrt{2})$): the systole has trace
$B = 2 + 2\sqrt{2}$
(\citealt{Buser1992}, \S13.3; the Bolza surface is the
genus-$2$ surface of maximal automorphism order~$48$, with
systole length $\ell = 2\,\mathrm{arccosh}(1{+}\sqrt{2})$),
giving the palindromic quartic
$\xi^4 - 4\xi^3 - 2\xi^2 - 4\xi + 1 = 0$
with threshold $\xi \approx 0.217$ in the quartic field
$\mathbb{Q}(\sqrt{2},\,\sqrt{2{+}2\sqrt{2}})$.
\end{conjecture}

\begin{proof}[Proof of the palindromic structure and the Bolza example]
The substitution $u = \xi + 1/\xi$ maps $\xi^2 - u\xi + 1 = 0$
(palindromic quadratic) to a scalar equation $u = B$.
If $B$ has minimal polynomial $P(u) = 0$ of degree $d$
over~$\mathbb{Q}$, substituting $u = \xi + 1/\xi$ and
multiplying by $\xi^d$ gives $Q(\xi) = \xi^d\,P(\xi + 1/\xi)$,
a degree-$2d$ polynomial.
It is palindromic: $\xi^{2d}\,Q(1/\xi) = Q(\xi)$, because
$P(\xi + 1/\xi) = P(1/\xi + \xi)$.
The palindromic structure forces the constant term to equal
the leading coefficient: both are~$1$ (for monic~$P$).
Therefore the product of all roots is~$1$,
and the roots are algebraic units.
For the Bolza surface: $P(u) = u^2 - 4u - 4$
(minimal polynomial of $2 + 2\sqrt{2}$), giving
$Q(\xi) = \xi^4 - 4\xi^3 - 2\xi^2 - 4\xi + 1$, verified
palindromic with coefficients $[1, -4, -2, -4, 1]$.
\end{proof}

\subsection{Arithmetic surfaces and selection rules}
On the congruence surface $\Gamma_0(p)\backslash\mathbf{H}^2$,
the geodesic of trace $B$ (and discriminant
$\Delta = B^2 - 4$) has a $\Gamma_0(p)$-representative if and
only if the Legendre symbol $(\Delta/p) \ne -1$.

\begin{proposition}[Quadratic reciprocity as a selection rule]
\label{prop:legendre-selection}
Of the five algebraic-integer stability thresholds, the number
visible on $\Gamma_0(p)\backslash\mathbf{H}^2$ depends on $p$:
\begin{center}
\begin{tabular}{ccccccc}
\toprule
$p$ & $N\!=\!23$ & $N\!=\!15$ & $N\!=\!11$ & $N\!=\!9$ & $N\!=\!8$
  & Count \\
& $\Delta\!=\!5$ & $\Delta\!=\!12$ & $\Delta\!=\!32$
  & $\Delta\!=\!96$ & $\Delta\!=\!252$ & \\
\midrule
$2$  & $\times$ & $\circ$ & $\circ$ & $\circ$ & $\circ$ & 4 \\
$3$  & $\times$ & $\circ$ & $\times$ & $\circ$ & $\circ$ & 3 \\
$5$  & $\circ$  & $\times$ & $\times$ & $\checkmark$ & $\times$ & 2 \\
$7$  & $\times$ & $\times$ & $\checkmark$ & $\times$ & $\circ$ & 2 \\
$11$ & $\checkmark$ & $\checkmark$ & $\times$ & $\times$ & $\times$ & 2 \\
$23$ & $\times$ & $\checkmark$ & $\checkmark$ & $\checkmark$ & $\times$ & 3 \\
$47$ & $\times$ & $\checkmark$ & $\checkmark$ & $\checkmark$ & $\checkmark$ & 4 \\
\bottomrule
\end{tabular}
\end{center}
($\checkmark$ = split, $\circ$ = ramified, $\times$ = inert.)
When $(\Delta/p) = +1$ (split), the geodesic lifts to
\emph{two} geodesics on $\Gamma_0(p)\backslash\mathbf{H}^2$,
doubling the automorphic correction.
When $(\Delta/p) = -1$ (inert), the geodesic is absent and
contributes \emph{zero} correction---the threshold on the quotient
matches the universal-cover value.
No prime $p < 200$ has all five thresholds simultaneously visible.
\end{proposition}

\begin{proof}
\emph{Step 1: conjugacy and quadratic forms.}
A hyperbolic $\gamma = \bigl[\begin{smallmatrix}a&b\\c&d
\end{smallmatrix}\bigr] \in \mathrm{SL}(2,\mathbb{Z})$
with trace $B = a{+}d$ has a conjugate in $\Gamma_0(p)$
iff some $\sigma\gamma\sigma^{-1}$ has lower-left entry
divisible by~$p$.
Conjugation by $\sigma = \bigl[\begin{smallmatrix}
\alpha&\beta\\\gamma'&\delta'\end{smallmatrix}\bigr]
\in \mathrm{SL}(2,\mathbb{Z})$ changes the lower-left
entry to $c' = c\alpha^2 + (d{-}a)\alpha\gamma' - b\gamma'^2$
(from the $(2,1)$-entry of $\sigma\gamma\sigma^{-1}$).
Hence $\gamma$ has a $\Gamma_0(p)$-conjugate iff the
binary quadratic form
$Q(x,y) = cx^2 + (d{-}a)xy - by^2$
represents a multiple of~$p$
(i.e., $Q(x,y) \equiv 0 \pmod{p}$ for some
$(x,y) \not\equiv (0,0) \pmod{p}$).

\emph{Step 2: discriminant and the Legendre symbol.}
The form $Q$ has discriminant
$(d{-}a)^2 + 4bc = (d{-}a)^2 + 4(ad{-}1) = (a{+}d)^2 - 4
= B^2 - 4 = \Delta$.
The equation $Q(x,y) \equiv 0 \pmod{p}$ with $y \not\equiv 0$
reduces (dividing by~$y^2$) to $cz^2 + (d{-}a)z - b \equiv 0$
where $z = x/y$, a quadratic in~$z$ with discriminant $\Delta$.
This has a solution mod~$p$ iff $\Delta$ is a quadratic residue
mod~$p$ (or $p \mid \Delta$):
precisely the condition $(\Delta/p) \ne -1$
(\citealt{Buser1992}, \S6; see also
\citealt{Katok1992}, Theorem~3.5.5 for the correspondence
between conjugacy classes in $\Gamma_0(N)$ and equivalence
classes of binary quadratic forms).

\emph{Step 3: trichotomy.}
When $(\Delta/p) = +1$ (split): $Q \equiv 0$ has two distinct roots
mod~$p$, giving \emph{two} $\Gamma_0(p)$-conjugacy classes in the
$\mathrm{SL}(2,\mathbb{Z})$ class of~$\gamma$---hence two geodesics
on $\Gamma_0(p)\backslash\mathbf{H}^2$.
When $p \mid \Delta$ (ramified): one double root, one geodesic.
When $(\Delta/p) = -1$ (inert): no root, no
$\Gamma_0(p)$-representative, no geodesic.

The table entries follow from evaluating $(\Delta/p)$ for
$\Delta \in \{5, 12, 32, 96, 252\}$ at each~$p$.
By the Chebotarev density theorem, each $\Delta$ is a QR for
asymptotically half of all primes; the five symbols are controlled
by four independent Legendre symbols $(2/p)$, $(3/p)$, $(5/p)$,
$(7/p)$ (since $96 = 32 \cdot 3$ gives $(\Delta_9/p) =
(\Delta_{11}/p)(\Delta_{15}/p)$), so the probability that all
five are simultaneously visible is $\sim (1/2)^4 = 1/16$,
predicting the first fully visible prime near
$p \approx 300$ (computed: $p = 311$).
\end{proof}

\noindent
The physical interpretation: on a hyperbolic lattice whose
global topology is that of $\Gamma_0(p)\backslash\mathbf{H}^2$,
only the \emph{visible} thresholds receive corrections from
the surface's geodesic spectrum.  The \emph{inert} thresholds
are identical to their universal-cover values---they are
\emph{topologically protected} by the Legendre symbol.
Different lattice topologies (different $p$) yield different
patterns of protected and unprotected thresholds.

\begin{corollary}[Characterization of fully visible primes]
\label{cor:fully-visible}
A prime $p > 7$ has all five thresholds simultaneously visible
on $\Gamma_0(p)\backslash\mathbf{H}^2$ if and only if
\[
  p \equiv r \pmod{840}
\]
for one of exactly $12$ residue classes $r$ out of
$\varphi(840) = 192$ coprime residues.
These $12$ classes are determined by the simultaneous conditions
\begin{alignat*}{2}
  (2/p) &\ne -1 &\quad&\Longleftrightarrow\quad p \equiv \pm 1 \pmod{8}, \\
  (3/p) &\ne -1 &&\Longleftrightarrow\quad p \equiv \pm 1 \pmod{12}, \\
  (5/p) &\ne -1 &&\Longleftrightarrow\quad p \equiv \pm 1 \pmod{5}, \\
  (7/p) &\ne -1 &&\Longleftrightarrow\quad
    \begin{cases} p \bmod 7 \in \{1,2,4\} & \text{if } p\equiv 1\!\!\pmod{4},\\
                  p \bmod 7 \in \{3,5,6\} & \text{if } p\equiv 3\!\!\pmod{4}.
    \end{cases}
\end{alignat*}
The density of fully visible primes is exactly $12/192 = 1/16$,
matching the independence prediction $(1/2)^4$ from the four
Legendre symbols.
The first fully visible prime is $p = 311$ $(\equiv 311 \bmod 840)$;
the first twenty are $311$, $479$, $719$, $839$, $1009$, $1129$,
$1151$, $1201$, $1319$, $1511$, $1559$, $1801$, $2351$, $2399$,
$2521$, $2689$, $2999$, $3049$, $3191$, $3359$.
\end{corollary}

\begin{proof}
The four conditions follow from the quadratic reciprocity law
applied to each prime factor of the discriminant set
$\{5, 12, 32, 96, 252\}$, as in the proof of
Proposition~\ref{prop:legendre-selection}.
The joint period is $\mathrm{lcm}(8, 12, 5, 28) = 840$
(the factor $28$ arises from the $(7/p)$ condition, which depends
on both $p \bmod 7$ and $p \bmod 4$, hence on $p \bmod 28$).
Each condition independently excludes half the coprime residues,
and the four conditions are algebraically independent
(the primes $\{2, 3, 5, 7\}$ generate independent
$\mathbb{Z}/2$-factors in
$(\mathbb{Z}/840\mathbb{Z})^\times$),
giving $192/2^4 = 12$ fully visible classes.
The listed primes are verified by direct Legendre symbol computation.
\end{proof}

\begin{theorem}[Universality of $N_{\mathrm{crit}} = 7$]
\label{thm:universality}
For any discrete subgroup $\Gamma \subset \mathrm{PSL}(2,\mathbb{R})$,
any $N \ge 3$, and any Fourier mode $m = 1, \ldots, N{-}1$:
the automorphic correction to the Havelock eigenvalue vanishes
at $\xi = 0$:
\[
  \delta C_1(\Gamma, N, m, \xi{=}0) = 0.
\]
In particular, $N_{\mathrm{crit}} = 7$ at $\xi = 0$ on every
quotient surface $\Gamma\backslash\mathbf{H}^2$, regardless of
the topology of the surface or the arithmeticity of~$\Gamma$.
\end{theorem}

\begin{proof}
We take the limit $\xi \to 0^+$.  As $\xi \to 0$,
the ring radius $r_E = a\sqrt{\xi} \to 0$ and the $N$ vortex
positions $z_k = r_E e^{2\pi ik/N} \to 0$ uniformly.
For any $\gamma = \bigl[\begin{smallmatrix}a&b\\c&d\end{smallmatrix}\bigr]
\in \Gamma$ with $\gamma \ne 1$:
$\gamma(z_k) \to \gamma(0) = b/d$, a single point independent of
the vortex index~$k$.  At finite $\xi > 0$ the Fourier-weighted
automorphic correction for mode $m \ge 1$ is
\[
  \delta C_1(m, \xi)
  = \sum_{k=0}^{N-1} H_{rr}(z_0, \gamma(z_k)) \cdot \cos(2\pi mk/N).
\]
In the limit $\xi \to 0^+$, the factor
$H_{rr}(z_0, \gamma(z_k)) \to H_{rr}(0, b/d)$ becomes
$k$-independent (since all $z_k \to 0$), giving
\[
  \lim_{\xi \to 0^+}\delta C_1(m, \xi)
  = H_{rr}(0, b/d) \cdot
  \underbrace{\sum_{k=0}^{N-1} \cos(2\pi mk/N)}_{= 0}
  = 0.
\]
The vanishing follows from $\mathbb{Z}_N$ character orthogonality
alone; no property of $\Gamma$ beyond discreteness is used.
Since $\delta C_1 \to 0$ for every $\gamma \in \Gamma$ and every
mode, the Havelock eigenvalue at $\xi = 0$ on
$\Gamma\backslash\mathbf{H}^2$ equals its universal-cover value
$(N{-}1) - m(N{-}m)/2$ exactly.
\end{proof}

\begin{remark}[Topology as a curvature correction]
\label{rmk:topology-correction}
The theorem implies a hierarchy of corrections:
at $O(1)$ (the $\xi = 0$ limit), the Havelock eigenvalue is
universal---independent of the surface topology.
The global topology enters at $O(\xi)$: as the $N$-gon moves
away from the origin, the image vortices under $\Gamma$ spread
out (their positions $\gamma(z_k)$ become $k$-dependent), and
the Fourier projection becomes nonzero.
The $O(\xi)$ coefficient depends on $\Gamma$ and is controlled
by the Legendre symbol when $\Gamma$ is arithmetic
(Proposition~\ref{prop:legendre-selection}).
For non-arithmetic $\Gamma$, the coefficient exists but has no
known closed form.
\end{remark}


\begin{proposition}[Curvature-stability dichotomy]
\label{prop:curvature-dichotomy}
On a constant-curvature surface with Gaussian curvature $K$,
for a ring of $N$ vortices at any nonzero geodesic radius:
\begin{itemize}
\item $K > 0$ ($S^2$): $N_{\mathrm{crit}} < 7$
  (curvature destabilises);
\item $K = 0$ (flat): $N_{\mathrm{crit}} = 7$ (marginal);
\item $K < 0$ ($\mathbf{H}^2$): $N_{\mathrm{crit}} \ge 7$,
  with the $N = 7$ mode strictly stable at any $\xi > 0$,
  and $N_{\mathrm{crit}} > 7$ for $\xi > \xi^*(8)$.
\end{itemize}
\end{proposition}

\begin{proof}
On $\mathbf{S}^2$: $C_1 = (N{-}1)\cos\varphi < N{-}1$ for
$\varphi > 0$, so $C_1 < f_{\max}(N)$ is violated at smaller~$N$.
On $\mathbf{H}^2$: $C_1 = (N{-}1)(1{+}\xi^2)/(1{-}\xi)^2 > N{-}1$
for $\xi > 0$, so $C_1 \ge f_{\max}(N)$ holds at larger~$N$.
Both are monotone in their curvature parameter.
\end{proof}

The \emph{qualitative} classification
$\{N_{\mathrm{crit}} < 7,\; = 7,\; > 7\}$ depends only on
$\mathrm{sgn}(K)$, which for constant-curvature surfaces is a
topological invariant (it determines whether the universal cover
is $\mathbf{S}^2$, $\mathbb{R}^2$, or~$\mathbf{H}^2$).
In this sense the stability boundary $N = 7$ separates two
\emph{stability regimes} of vortex polygon stability.
The \emph{quantitative} value $N_{\mathrm{crit}}(K, R)$ depends
on the dimensionless product $K R^2$ and is purely geometric.
\begin{proposition}[Morse index and spectral flow]
\label{prop:morse-index}
The Morse index of the $N$-gon critical point of the Thomson energy
on the constrained configuration space is
\[
  \mu(N) = \begin{cases} 0 & N \le 7, \\ N - 5 & N \ge 8. \end{cases}
\]
The unstable modes on the flat plane are exactly $m = 3, 4, \ldots, N{-}3$.
\end{proposition}

\begin{proof}
$\lambda_m < 0$ iff $m(N{-}m) > 2(N{-}1)$ iff
$(m - N/2)^2 < (N^2 - 8N + 8)/4$.  The left boundary of the
unstable band is $m_{\min} = \lceil N/2 - \sqrt{(N{-}4)^2 - 8}/2\,\rceil$.
For $N \ge 8$: $m_{\min} = 3$, because
$(N{-}6)^2 \le (N{-}4)^2 - 8$ iff $N \ge 7$
(giving $m_{\min} \le 3$) and
$(N{-}4)^2 > (N{-}4)^2 - 8$
(giving $m_{\min} > 2$).
By the palindromic symmetry $\lambda_m = \lambda_{N-m}$, the
upper boundary is $m_{\max} = N - 3$.
Hence $\mu(N) = (N{-}3) - 3 + 1 = N - 5$.
\end{proof}

\begin{theorem}[Morse--Bott classification of the $N$-gon]
\label{thm:morse-bott}
Let $\bar{M} = \{z \in \mathbb{C}^N : P{=}0,\; L{=}R^2\}/\mathrm{SO}(2)$
be the reduced constraint surface and $H = -\sum_{j<k}\ln|z_j - z_k|$
the Thomson energy.
The regular $N$-gon is a Morse--Bott critical point of $H|_{\bar{M}}$
for all $N \ne 7$:
\begin{enumerate}
\item $N \le 6$: non-degenerate minimum (index~$0$).
\item $N = 7$: \emph{not} Morse--Bott (the normal Hessian
  has a $4$-real-dimensional kernel from modes $m = 3, 4$;
  resolved by the quartic $\alpha_0 = 45/14$,
  Paper~II, \S5).
\item $N \ge 8$: Morse--Bott saddle of complex index $N{-}5$
  (real index $2(N{-}5)$).
\end{enumerate}
Moreover, $N = 7$ is the \emph{unique} value at which the
normal Hessian degenerates.
\end{theorem}

\begin{proof}
The normal space to the $\mathrm{SO}(2)$-orbit of the $N$-gon
in $\bar{M}$ is spanned by the Fourier modes
$m = 2, \ldots, N{-}1$: mode $m = 0$ is removed by the
center-of-mass constraint $P = 0$, and mode $m = 1$ is split
between the angular-impulse constraint ($\mathrm{Re}(c_1) = 0$)
and the orbit tangent ($\mathrm{Im}(c_1)$).
The normal dimension is $2(N{-}2) = \dim\bar{M}$.
Each normal mode has Havelock eigenvalue
$\lambda_m = (N{-}1) - m(N{-}m)/2$
(Lemma~\ref{lem:havelock}).

The $N$-gon is Morse--Bott iff $\lambda_m \ne 0$ for all
$m \in \{2, \ldots, N{-}1\}$.
Setting $\lambda_m = 0$ gives $m^2 - Nm + 2(N{-}1) = 0$
with discriminant $\Delta = (N{-}4)^2 - 8$.
For integer solutions we need $\Delta = k^2$, i.e.\
$(N{-}4{-}k)(N{-}4{+}k) = 8$.
The only factorization with matching parity is $(2,4)$,
giving $N = 7$, $k = 1$, $m \in \{3,4\}$.
Therefore $N = 7$ is the unique degenerate case.

The Morse index for $N \ge 8$ is $N{-}5$ by
Proposition~\ref{prop:morse-index}.
\end{proof}

The closed-form $N_{\mathrm{crit}} = 7$ follows from
Theorem~\ref{thm:why-7-algebraic} below.

The family $H_N(\xi) = C_1(\xi)\,I - F_N$ parametrised by
$\xi \in [0, 1)$ has a monotone spectral flow: since
$C_1(\xi) = (N{-}1)(1{+}\xi^2)/(1{-}\xi)^2$ satisfies
$\dd C_1/\dd\xi = (N{-}1)\cdot 2(1{+}\xi)/(1{-}\xi)^3 > 0$
for all $\xi \in [0,1)$, each eigenvalue
$\lambda_m(\xi) = C_1(\xi) - m(N{-}m)/2$ is strictly increasing
in~$\xi$.  Every eigenvalue that is negative at $\xi = 0$
(flat plane) crosses zero exactly once as $\xi \to 1$ (infinite
curvature).  The total spectral flow therefore equals the
flat-plane Morse index:
$\mathrm{SF}(H_N, 0 \to 1) = \mu(N)$.  This is \emph{proven}
(it follows from Proposition~\ref{prop:morse-index} together with
the strict monotonicity of $C_1(\xi)$, which is an explicit
calculation---and is consistent with the
$N_{\mathrm{crit}}$ monotonicity
(Proposition~\ref{prop:ncrit-monotone}), which concerns the
integer-valued function $N_{\mathrm{crit}}(R/a)$, a different
but related quantity).

\begin{remark}[Robustness of $\mu(N)$ and the $N = 7$ exception]
\label{rmk:topological-protection}
For $N \ne 7$, all eigenvalues $\lambda_m(0)$ are nonzero, so the
spectral flow $\mathrm{SF} = \mu(N)$ is determined entirely by the
boundary data (the Havelock spectrum at $\xi = 0$ and the
divergence $C_1 \to \infty$ at $\xi \to 1^-$).
By standard Fredholm index stability
(\citealt{Kato1995}, Theorem~IV.5.17),
$\mu(N)$ is insensitive to any bounded perturbation $V(\xi)$
with $V(0) = 0$ that couples Fourier modes in the interior---even
perturbations that break the $\mathbb{Z}_N$ diagonal structure
and create avoided crossings cannot change the net spectral flow
(verified numerically for
$N = 8$--$10$ with coupling strengths up to $\|V\| = 100$;
see Code Availability).
At $N = 7$, the kernel of $H_7(0)$ (modes $m = 3, 4$ with

sufficiently strong mode coupling can push the zero eigenvalues

The heptagonal ring is the unique case where the Morse index is

\end{remark}

\begin{theorem}[Index pairing]
\label{thm:index-pairing-k0}
The spectral flow of the Havelock family $\{H_N(\xi)\}_{\xi
\in [0,1)}$ equals the pairing of the $K_0$ class with the
trace cocycle $\tau_0 \in HC^0(C^*(\mathbb{Z}_N))$:
\begin{equation}\label{eq:index-pairing-sf}
  \mathrm{SF}(H_N,\; 0 \to 1)
  \;=\; \langle[\mu_N(0)],\, \tau_0\rangle
  \;=\; \mathrm{tr}(P_-(0))
  \;=\; N - 5.
\end{equation}
The pairing is a topological invariant: it is unchanged by
any continuous deformation of $H_N$ that does not close the
spectral gap at any $\mathbb{Z}_N$ mode.
\end{theorem}

\begin{proof}
For $N \ge 8$: the spectral flow is $N{-}5$
(Proposition~\ref{prop:morse-index}).
The $K_0$ class $[\mu_N(0)] =
\sum_{\lambda_m(0) < 0} [\rho_m]$ has $\dim = N{-}5$
(the flat-plane Morse index).
At $N = 7$: $\lambda_3 = \lambda_4 = 0$ (kernel, not
negative), so $\mathrm{tr}(P_-(0)) = 0$ and
$\mathrm{SF} = \mu(7) = 0$---no eigenvalue crosses from
negative to positive.
The two marginal eigenvalues $\lambda_{3,4} = 0$ become
positive for $\xi > 0$; this is the gapping of the
$N = 7$ threshold, not a spectral-flow event (no sign change).
For $N \ge 8$: $N{-}5$ eigenvalues start negative and each
reaching zero at $\xi = 0$.
Since $C^*(\mathbb{Z}_N) \cong \mathbb{C}^N$:
$HC^0 = \mathbb{C}^N$ (the trace on each summand) and
$HC^1 = 0$, so the only index pairing is
$\langle [P_-], \tau_0 \rangle = \mathrm{tr}(P_-) = N{-}5$.
Topological invariance: by Kato~(\citeyear{Kato1995},
Theorem~IV.5.17), the spectral projection $P_-$ is
continuous in norm when no eigenvalue crosses zero.
Its trace is integer-valued and continuous, hence constant.
\end{proof}

\begin{theorem}[Algebraic origin of $N_{\mathrm{crit}} = 7$]
\label{thm:why-7-algebraic}
$N_{\mathrm{crit}} = 7$ is the unique positive integer $N \ge 3$ at
which the $\mathbb{Z}_N$ Casimir maximum equals the logarithmic $C_1$:
\begin{equation}\label{eq:n-crit-factorization}
  \max_m f(m,N) = C_1
  \;\Longleftrightarrow\;
  N^2 - 8N + 7 = (N-1)(N-7) = 0.
\end{equation}
\end{theorem}

\begin{proof}
For odd $N$ (the binding case): $\max_m f = (N^2{-}1)/8$
and $C_1 = N{-}1$.  Setting $(N^2{-}1)/8 = N{-}1$ and
multiplying by~$8$: $N^2 - 1 = 8(N{-}1)$, i.e.,
$N^2 - 8N + 7 = 0$.  This factors as $(N{-}1)(N{-}7) = 0$,
giving $N = 7$ (the root $N = 1$ is below the polygon range).
For even $N$: $\max_m f = N^2/8$; setting $N^2/8 = N{-}1$
gives $N^2 - 8N + 8 = 0$ with roots $4 \pm 2\sqrt{2}$
(irrational, so no even transition exists).
\end{proof}

\noindent
The coefficient $8 = 2 \cdot 4$ in~\eqref{eq:n-crit-factorization}
arises from two intrinsic features of $\mathbb{Z}_N$:
the Havelock normalization $f(m,N) = m(N{-}m)/\mathbf{2}$
and the critical-mode evaluation
$m = (N{-}1)/2$ giving $m(N{-}m) = (N^2{-}1)/\mathbf{4}$.
The factorization $(N{-}1)(N{-}7)$ is therefore purely
representation-theoretic: no differential geometry, no index
theorems, no PDEs---just the intersection of the quadratic
Casimir with the linear $C_1$ from dilation invariance.
The number~$7$ is determined by the $\mathbb{Z}_N$
representation theory combined with the uniqueness of the
logarithmic interaction.

\section{The mark distribution}

The section proceeds in three stages.

\subsection{The spectral bound}
\label{sec:spectral-diameter}

\begin{theorem}[Spectral bound]\label{thm:spectral-diameter}
For any connected affine ADE Dynkin diagram with Cartan matrix $C = 2I - A$:
\begin{enumerate}
\item[(i)] $C$ is positive semidefinite with $\min(\mathrm{spec}\,C) = 0$.
\item[(ii)] $\max(\mathrm{spec}\,C) \le 4$, with equality if and only if
  the diagram is bipartite.
\end{enumerate}
All D-type and E-type affine diagrams are trees (hence bipartite),
so $a \coloneqq \max(\mathrm{spec}\,C) = 4$ for $\tilde{D}_n$,
$\tilde{E}_6$, $\tilde{E}_7$, $\tilde{E}_8$.
\end{theorem}

\begin{proof}
(i)~$\det C = 0$ (affine condition: the marks~$d$ satisfy $Cd = 0$).
The quadratic form $x^TCx = 2\sum x_i^2 - 2\sum_{i \sim j} x_i x_j \ge 0$
(Cauchy--Schwarz on each edge), so $C$ is positive semidefinite.

(ii)~$C = 2I - A$ gives $\mathrm{spec}(C) = \{2 - \lambda : \lambda \in \mathrm{spec}(A)\}$.
The affine condition $Ad = 2d$ with $d > 0$ makes $\rho(A) = 2$ (Perron--Frobenius).
Therefore $\max(\mathrm{spec}\,C) = 2 + |\min\mathrm{spec}(A)| \le 2 + \rho(A) = 4$.
Equality holds iff $-2 \in \mathrm{spec}(A)$, which occurs iff the graph
is bipartite (the spectrum of a bipartite graph is symmetric about~$0$).
Trees are bipartite; all affine D and E diagrams are trees.
\end{proof}

\subsection{The mark-balance identity and $E_8$ selection}
\label{sec:mark-balance}

\begin{theorem}[Mark-balance and $E_8$ selection]\label{thm:mark-balance}
Define $k_1 \coloneqq \sum d_i^2 / \sum d_i$ for the affine marks of
an ADE type.
Among \emph{all} ADE types, $E_8$ is the unique type satisfying
$k_1 = a = 4$ (the spectral diameter), equivalently
\begin{equation}\label{eq:mark-balance}
  \sum_{i=0}^{8} d_i(d_i - 4) = 0
  \qquad\Longleftrightarrow\qquad
  |I^*| = 4\,h(E_8)\; (= 120 = 4 \times 30).
\end{equation}
\end{theorem}

\begin{proof}
Direct computation: $\sum d_i^2 = 120$, $\sum d_i = 30$,
$k_1 = 120/30 = 4$.
Exhaustive check of all ADE types:
$A_n$: marks all~$1$, $k_1 = 1$.
$D_n$ ($n \geq 4$): $k_1 = 2(n{-}2)/(n{-}1) < 2$.
$E_6$: $k_1 = 24/12 = 2$.
$E_7$: $k_1 = 48/18 = 8/3$.
Only~$E_8$ has $k_1 = 4 = a$.

\emph{Geometric form (Gauss--Bonnet).}
The $(2,3,5)$ Schwarz triangle on~$S^2$ has area
$\varepsilon\pi = \pi(1/2 + 1/3 + 1/5 - 1) = \pi/h$.
Since $|I^*|$ such triangles tile~$S^2$:
$|I^*| \cdot \pi/h = \mathrm{Area}(S^2) = 4\pi$,
giving $|I^*| = 4h$.
The factor~$4 = \mathrm{Area}(S^2)/\pi = 2\chi(S^2)$
is a topological invariant of~$S^2$.
\end{proof}

\begin{remark}\label{rem:k1-meaning}
The spectral diameter $a = 4$ is universal (Theorem~\ref{thm:spectral-diameter});
the mark-balance $k_1 = a$ is specific to~$E_8$.
The condition $k_1 = a$ says: the first-to-second moment ratio of
the mark distribution equals the spectral bound of the graph.
\end{remark}

\subsection{The vertex--face theorem}
\label{sec:vtensor-f}

\begin{theorem}[$V \otimes F$ theorem]\label{thm:vtensor-f}
For the icosahedron with rotation group~$A_5$,
\begin{equation}\label{eq:vtensor-f}
  V_{\mathrm{vertex}} \otimes V_{\mathrm{face}}
  = 4 \times \mathrm{reg}(A_5)
\end{equation}
as $A_5$-representations.  Hence $V \times F = 12 \times 20 = 240 = |\text{roots}(E_8)|$.
The sufficient condition is $\gcd(|\mathrm{Stab}_V|, |\mathrm{Stab}_F|) = 1$;
for Platonic solids this is also necessary (the tetrahedron, with
$\gcd(3,3) = 3$, fails).
\end{theorem}

\begin{proof}
The $A_5$-character of $V \otimes F$ at $g \neq e$ is
$\chi_{V \otimes F}(g) = \chi_V(g) \chi_F(g)$.
Since $|\mathrm{Stab}_V| = 5$ and $|\mathrm{Stab}_F| = 3$
with $\gcd(5,3) = 1$, no non-identity element fixes both a vertex
and a face.  Thus $\chi_V(g) \chi_F(g) = 0$ for all $g \neq e$.
A character proportional to~$\delta(g,e)$ is a multiple of the
regular representation, with multiplicity
$\chi_{V \otimes F}(e)/|A_5| = 240/60 = 4$.
\end{proof}

\begin{remark}
The same argument gives $V \otimes F = 2 \cdot \mathrm{reg}(S_4)$
for the octahedron and cube (where $\gcd(4,3)=1$).
The tetrahedron fails: $\gcd(3,3) = 3 \neq 1$,
so $V \otimes F$ is \emph{not} a regular multiple.
The identity $V \times F = |\text{roots}|$ holds \emph{only} for
the icosahedron among all Platonic solids, because it requires
$|I^*| = 4h$ (Theorem~\ref{thm:mark-balance}).
\end{remark}

\subsection{The division algebra chain}
\label{sec:division-algebra-chain}

\begin{definition}[Charge-magnitude map]
For marks $\{d_i\}$ with integer pivot $k_1 = \sum d_i^2/\sum d_i$,
the \emph{charge-magnitude map} sends $d_i \mapsto |d_i - k_1|$
and discards zeros.  The kernel consists of the marks equal to~$k_1$.
\end{definition}

\begin{theorem}[Division algebra chain]\label{thm:division-chain}
Iterating the charge-magnitude map on the $E_8$ marks gives:
\[
  E_8 \xrightarrow{k_1=4} E_6 \xrightarrow{k_1=2} A_3 \xrightarrow{k_1=1} \varnothing
\]
with pivots $4 = \dim(\mathbb{H})$, $2 = \dim(\mathbb{C})$,
$1 = \dim(\mathbb{R})$---the normed division algebra dimensions.
The kernel at each step has size $2, 3, 4$ respectively,
summing to $9 = |\tilde{E}_8|$ and multiplying to $24 = |T^*|$.
\end{theorem}

\begin{proof}
Step~1: $|d_i - 4|$ for $d = (1,2,3,4,5,6,4,2,3)$ gives
$(3,2,1,0,1,2,0,2,1)$.  Nonzero values sorted:
$(1,1,1,2,2,2,3)$, which is the multiset of affine $\tilde{E}_6$ marks.  Kernel: two copies of~$4$.

Step~2: $E_6$ marks $(1,1,2,2,3,2,1)$ have $k_1 = 24/12 = 2$.
$|d_i - 2|$ gives $(1,1,0,0,1,0,1)$; nonzero sorted: $(1,1,1,1) = A_3$ marks.
Kernel: three copies of~$2$.

Step~3: $A_3$ marks $(1,1,1,1)$ have $k_1 = 1$.  All equal the pivot;
the chain terminates.  Kernel: four copies of~$1$.
\end{proof}

\begin{theorem}[Pivot uniqueness]\label{thm:pivot-unique}
The triple $(4,2,1)$ is the unique decreasing positive-integer triple
$(a,b,c)$ satisfying $a + b + c = abc - 1$.
\end{theorem}

\begin{proof}
For $c \geq 3$: $abc \geq 3ab$ and $a+b+c \leq 3a$, so
$abc - (a{+}b{+}c) \geq 3ab - 3a = 3a(b{-}1) \geq 3 \cdot 3 \cdot 2 = 18 > 1$;
no solution.
For $c = 2$: the equation gives $2ab - a - b = 3$, i.e.\
$a = (b{+}3)/(2b{-}1)$.  For $b = 2$: $a = 5/3$; $b = 3$: $a = 6/5$;
$b = 4$: $a = 1 < b$; $b \geq 5$: $a < 1$.  No valid solution.
For $c = 1$: $a + b + 1 = ab - 1$, so $(a-1)(b-1) = 3$.
Since $3$ is prime: $(a-1, b-1) = (3, 1)$, giving $(a,b,c) = (4,2,1)$.
\end{proof}

\begin{proposition}[Pivot--stability coincidence]\label{prop:ncrit-rank}
The division algebra pivots satisfy
$\sum = 4 + 2 + 1 = 7 = N_{\mathrm{crit}}$ (the flat-plane stability
boundary, Theorem~\ref{thm:havelock}) and
$\prod = 4 \times 2 \times 1 = 8 = \operatorname{rank}(E_8) = \dim(\mathbb{O})$.
Both equalities are verified directly; the sum connects
the division algebra chain (Theorem~\ref{thm:division-chain})
to the Havelock stability threshold (Theorem~\ref{thm:havelock}).
\end{proposition}

\subsection{The cyclotomic-Chebyshev factorization}
\label{sec:cyclotomic}

\begin{theorem}[Cyclotomic-Chebyshev factorization]\label{thm:cyclotomic}
The characteristic polynomial of the $\tilde{E}_8$ adjacency matrix factors as
\begin{equation}\label{eq:char-poly}
  p(x) = x(x^2-1)(x^2-4)(x^4-3x^2+1),
\end{equation}
where $x^4 - 3x^2 + 1 = \prod_{k=1}^{4}(x - 2\cos(k\pi/5))$ is the
pentagonal Chebyshev polynomial.
Under the Joukowski map $x = z + z^{-1}$, the set of~$z$ on the
unit circle with $z + z^{-1}$ equal to an eigenvalue consists of
the roots of
\begin{equation}\label{eq:cyclotomic}
  P(z) = \Phi_1 \cdot \Phi_2 \cdot \Phi_3 \cdot \Phi_4 \cdot \Phi_5 \cdot \Phi_6 \cdot \Phi_{10}
  \qquad(\deg P = 16 = 2\operatorname{rank}(E_8)),
\end{equation}
whose indices $\{1,2,3,4,5,6,10\}$ are exactly the element orders of~$I^*$
and the divisors of~$|A_5| = 60$ that are~$\leq 10$.
(The degree doubling $9 \to 16$ arises because each real eigenvalue~$x$
corresponds to two unit-circle points $z, z^{-1}$, except $x = \pm 2$
which give $z = \pm 1$.)
\end{theorem}

\begin{proof}
The adjacency eigenvalues $\{0, \pm 1/\varphi, \pm 1, \pm\varphi, \pm 2\}$
(Section~\ref{sec:i-star-mckay}) give the factorization~\eqref{eq:char-poly}
directly.
The pentagonal identification: $2\cos(k\pi/5) = \varphi, 1/\varphi, -1/\varphi, -\varphi$
for $k=1,2,3,4$.

\end{proof}

\subsection{Three generations}
\label{sec:generations}

\begin{theorem}[Three generations]\label{thm:three-gen}
For each exceptional affine Dynkin diagram $\tilde{E}_6$,
$\tilde{E}_7$, $\tilde{E}_8$, the $\mu = 1$ eigenspace of the
Cartan matrix contains a vector that partitions the nodes into
exactly \emph{three doublets} (adjacent same-sign pairs) and singlets (zeros).
For~$\tilde{E}_7$ and~$\tilde{E}_8$ the eigenspace is $1$-dimensional
and the vector is unique; for~$\tilde{E}_6$ the eigenspace is
$2$-dimensional and three doublets occur generically.
\end{theorem}

\begin{proof}
\emph{$\tilde{E}_8$ (multiplicity~$1$).}
Solving $Av = v$ node by node:
$v_1 = v_0$; $v_2 = 0$; $v_3 = -v_0$; $v_4 = -v_0$; $v_5 = 0$;
$v_6 = v_0$; $v_7 = v_0$; $v_8 = 0$.
Setting $v_0 = -1$: $v = (-1,-1,0,+1,+1,0,-1,-1,0)$.
Three doublets: $\{0,1\}$, $\{3,4\}$, $\{6,7\}$; unique.

\emph{$\tilde{E}_7$ (multiplicity~$1$).}
Solving $Av = v$: $v = (v_0, v_0, 0, -v_0, 0, v_0, v_0, -v_0)$.
Three doublets: $\{0,1\}$, $\{5,6\}$, $\{3,7\}$; unique.

\emph{$\tilde{E}_6$ (multiplicity~$2$).}
The $\tilde{E}_6$ diagram is a star (central node~$\rho_4$,
three arms of length~$2$).
Solving $Av = v$:
$v_2 = v_0$, $v_3 = v_1$, $v_4 = 0$,
$v_5 = v_6 = -(v_0 + v_1)$,
giving the $2$-parameter eigenspace
$\{(v_0, v_1, v_0, v_1, 0, {-}(v_0{+}v_1), {-}(v_0{+}v_1))\}$.
Three doublets $\{0,2\}$, $\{1,3\}$, $\{5,6\}$ exist iff
$v_0 \neq 0$, $v_1 \neq 0$, $v_0 + v_1 \neq 0$---a generic
(open dense) condition.
Example: $v_0 = v_1 = 1$ gives $(1,1,1,1,0,-2,-2)$.
\end{proof}

\section{K-theoretic classification of stability phases}
\label{sec:k-theory}
%% ====================================================================

The palindromic thresholds of Section~\ref{sec:algebraic-structure}
classify stability on the universal cover~$\mathbf{H}^2$.
On quotient surfaces $\Gamma\backslash\mathbf{H}^2$, the threshold
pattern persists but the Morse index acquires a fractional correction
from the surface topology.
This appendix develops the $K$-theoretic machinery for the
quotient problem in three stages:
(i)~the elementary $K_0$-class
$[\mu_N(\xi)] \in R(\mathbb{Z}_N)$ records eigenvalue signs
and changes at palindromic thresholds (\S\ref{sec:index-pairing});
(ii)~on arithmetic quotients $\Gamma\backslash\mathbf{H}^2$,
the Selberg trace formula computes the spectral correction
$\delta C_1$ as an index pairing, and spectral flow preservation
shows the wall-crossing pattern is the same as
on~$\mathbf{H}^2$ (\S\ref{sec:spectral-flow});
(iii)~at palindromic thresholds where the spectral gap fails,
the Kasparov product
$KK(C^*(\mathbb{Z}_N),\,C^*_r(\Gamma))$ captures the
fractional $L^2$-Morse index---$0.644$ on the Bolza
surface (\S\ref{sec:gap-free}, \S\ref{sec:kk-programme}).

\subsection{The elementary invariant}

On the universal cover $\mathbf{H}^2$, the Havelock
decomposition~\eqref{eq:central-decomposition} gives
$H = \bigoplus_{m=1}^{N-1} \lambda_m P_m$, so the negative
spectral projection $P_- = \sum_{\lambda_m < 0} P_m$ defines
\begin{equation}\label{eq:k0-class}
  [\mu_N(\xi)]
  = \sum_{\substack{m=1 \\ \lambda_m(\xi) < 0}}^{N-1} [\rho_m]
  \;\in\; K_0(C^*(\mathbb{Z}_N))
  \cong R(\mathbb{Z}_N)
  \cong \mathbb{Z}^N.
\end{equation}
This class is locally constant in~$\xi$, changing only at the
palindromic thresholds~$\xi^*(N,m)$.
At $\xi = 0$ and $N \ge 8$:
$[\mu_N(0)] = [\rho_3] + \cdots + [\rho_{N-3}]$
with $\dim = N{-}5$.
The proof is immediate:
each $\lambda_m(\xi)$ is continuous and changes sign only
at $\xi^*(N,m)$.
As an invariant of the $\mathbb{Z}_N$ representation ring,
this is elementary---it says ``signs are preserved by
continuous functions that don't cross zero.''
The curvature-stability dichotomy
($K > 0 \Rightarrow N_{\mathrm{crit}} < 7$,
$K < 0 \Rightarrow N_{\mathrm{crit}} \ge 7$)
is the $\mathbb{Z}_2$-reduction via
$\dim \colon R(\mathbb{Z}_N) \to \mathbb{Z}/2$.

\begin{corollary}[Equivariant Euler class sign change at $N = 7$]
\label{cor:euler-class}
The negative normal bundle $\nu^- = \bigoplus_{\lambda_m < 0} L_m$
has $\mathbb{Z}_N$-equivariant Euler class
\[
  e(\nu^-) =
  \prod_{\substack{m = 2 \\ \lambda_m < 0}}^{N-1}
  (t - \omega^m)
  \;\in\; R(\mathbb{Z}_N) \cong \mathbb{Z}[t]/(t^N{-}1),
  \quad \omega = e^{2\pi i/N}.
\]
The transition at $N = 7$:
$e(\nu^-) = 1$ (trivial) for $N \le 6$;
undefined for $N = 7$ (kernel);
non-trivial for $N \ge 8$.
The negative modes come in palindromic pairs
$\{m,\, N{-}m\}$ (since $\lambda_m = \lambda_{N-m}$),
so the real Euler class factors into quadratic terms
$(t - \omega^m)(t - \omega^{N-m})
= t^2 - 2\cos(2\pi m/N)\,t + 1$.
\end{corollary}

\begin{proof}
The decomposition into line bundles $L_m$ follows from
Theorem~\ref{thm:morse-bott}.
The Euler class is the product of first Chern classes by the
Whitney sum formula in equivariant cohomology.
The triviality for $N \le 6$ and non-triviality for $N \ge 8$
follow from Proposition~\ref{prop:morse-index}: the product
is empty (hence~$1$) when no negative modes exist, and
non-trivial once $\nu^- \ne 0$.
The palindromic factorization follows from the symmetry
$\lambda_m = \lambda_{N-m}$, which pairs $L_m$ with $L_{N-m}$.
\end{proof}

\begin{theorem}[CS-Havelock identity]
\label{thm:cs-havelock}
On the symmetric space $\mathrm{SL}(2,\mathbb{R})/\mathrm{SO}(2)
= \mathbf{H}^2$, the quadratic Casimir of
$\mathrm{sl}(2,\mathbb{R})$ restricted to the $m$-th
$\mathbb{Z}_N$ tangential mode of the regular $N$-gon equals
the Havelock Casimir:
\begin{equation}\label{eq:cs-havelock}
  C_2^{\mathrm{sl}(2,\mathbb{R})}(m)
  = f(m,N) = \frac{m(N{-}m)}{2}.
\end{equation}
The quantity $f(m,N)$ is simultaneously:
\begin{enumerate}
\item the Havelock eigenvalue (vortex dynamics,
  Lemma~\ref{lem:havelock});
\item the $\mathrm{sl}(2,\mathbb{R})$ Casimir on the
  symmetric space (Helgason, representation theory);
\item the Chern--Simons Wilson line Casimir
  (gauge theory, Witten 1988);
\item the Knizhnik--Zamolodchikov connection eigenvalue
  at the $\mathbb{Z}_N$ point
  (Etingof--Frenkel--Kirillov 1998).
\end{enumerate}
These four quantities are equal because they are the
\emph{same operator} (the Laplace--Beltrami operator
on~$\mathbf{H}^2$) restricted to the \emph{same subspace}
(the $m$-th $\mathbb{Z}_N$ Fourier mode of the tangential
perturbation at the $N$-gon ring).
\end{theorem}

\begin{proof}
The angular part of the $\mathbf{H}^2$ Laplacian at the
$N$-gon ring at geodesic radius~$\rho_0$ is the
$\csc^2$ circulant with $\mathbb{Z}_N$ Fourier eigenvalue
$T_m/\sinh^2\rho_0$, where
$T_m = \sum_{p=1}^{N-1}(1 - \cos(2\pi pm/N))
/(4\sin^2(\pi p/N)) = m(N{-}m)/2$
(Havelock identity, Lemma~\ref{lem:havelock}).
The conformal normalisation (multiplication by
$\sinh^2\rho_0$) gives the $\rho_0$-independent value
$C_2(m) = m(N{-}m)/2$.
On the symmetric space $G/K = \mathrm{SL}(2,\mathbb{R})
/\mathrm{SO}(2)$, the $G$-Casimir acts as the Laplacian
(Helgason, \emph{Groups and Geometric Analysis},
Theorem~II.2.3).
The EFK theorem (1998, Theorem~7.1.2) identifies the
CMS Hamiltonian at the $\mathbb{Z}_N$ locus with the
KZ connection, and the KZ connection equals the CS
Wilson line (Witten 1988).
\end{proof}

\begin{corollary}[Index from characteristic classes]
\label{cor:index-from-char}
The equivariant index of the negative normal bundle~$\nu^-$
is $N{-}5$ for $N \ge 8$, computed without constructing
the Dirac operator~$D_N$ explicitly:
\[
  \mathrm{ind}(\nu^-)
  = \hat{A}(\mathrm{pt}) \cdot \mathrm{rank}(\nu^-)
  = 1 \cdot (N{-}5) = N{-}5.
\]
The Chern character $\mathrm{ch}(\nu^-) = \sum_{m \in S^-}[\rho_m]$
has rank $|S^-| = N{-}5$
(Corollary~\ref{cor:euler-class}), and
$\hat{A}(\mathrm{pt}) = 1$ because the base is a point
in the equivariant sense.
Three-way cross-check: the index theorem, the Morse
formula (Proposition~\ref{prop:morse-index}), and direct
eigenvalue counting all give~$N{-}5$.
\end{corollary}

\begin{corollary}[Equivariant signature]
\label{cor:equiv-sig}
The $\mathbb{Z}_N$ equivariant signature of the
Havelock Hessian is
\[
  \sigma_N =
  \begin{cases}
    N{-}2 & N \le 6, \\
    3 & N = 7 \;\text{(kernel excluded)}, \\
    8{-}N & N \ge 8.
  \end{cases}
\]
This equals the spectral eta invariant
$\eta_{\mathrm{Hav}}(N, 0)$ of the finite-dimensional
Havelock operator (the finite spectrum makes the
eta function entire, requiring no regularisation).
The APS spectral flow at the $N = 7$ transition is
$-4 = -2\dim\ker$, confirming the
Atiyah--Patodi--Singer theorem.
\end{corollary}

\begin{proof}
The signature counts positive minus negative modes
among $m = 2, \ldots, N{-}1$.
For $N \le 6$: all $\lambda_m > 0$, giving $\sigma = N{-}2$.
For $N = 7$: modes $3, 4$ have $\lambda = 0$;
the remaining $3$ modes ($2, 5, 6$) are positive: $\sigma = 3$.
For $N \ge 8$: positive modes are $\{2, N{-}2\}$ (count~$2$),
negative modes are $\{3, \ldots, N{-}3\}$ (count~$N{-}5$);
$\sigma = 2 - (N{-}5) = 7 - N$.
Mode $N{-}1$ has $\lambda_{N-1} = (N{-}1)/2 > 0$
(always positive), adding one to the positive count:
$\sigma = 3 - (N{-}5) = 8 - N$.
The spectral flow from $N = 6$ ($\sigma = 4$) to $N = 8$
($\sigma = 0$) is $-4$; since $\dim\ker(N{=}7) = 2$
(complex, i.e., $4$ real), this is $-2\dim_{\mathbb{C}}\ker$,
as required by the APS theorem.
\end{proof}

\begin{remark}[Explicit Dirac operator $D_N$]
\label{rmk:dirac-explicit}
The Dirac operator $D_N$ can be constructed explicitly as
the Callias-type Higgs field on the Clifford module
$S = \mathbb{C}^{2^{N-2}}$:
\begin{equation}\label{eq:dirac-explicit}
  D_N = \Phi = \sum_{m=2}^{N-1} \lambda_m\,J_m,
  \qquad
  J_m = \tfrac{i}{2}\gamma^{2(m-2)+1}\gamma^{2(m-2)+2},
\end{equation}
where $\gamma^a$ ($a = 1,\ldots,2(N{-}2)$) are the
generators of $\mathrm{Cliff}(\mathbb{R}^{2(N-2)})$.
The operator $\Phi$ is diagonal in the binary-string basis
$|s_2,\ldots,s_{N-1}\rangle$ ($s_m = \pm\tfrac{1}{2}$)
with eigenvalue $\sum_m \lambda_m\,s_m$.
The chirality grading $\Gamma = \prod_m (2J_m)$ splits
$S = S^+ \oplus S^-$, and
$\mathrm{ind}(D_N) = N{-}5$ by the Callias index theorem
(the negative-mode count of the Havelock spectrum).
\end{remark}

\subsection{The index pairing on arithmetic surfaces}
\label{sec:index-pairing}

The nontrivial $K$-theory lives one level deeper.
On a compact arithmetic surface $S = \Gamma\backslash\mathbf{H}^2$,
the eigenvalues become $\lambda_m(\Gamma, \xi) = C_1(\mathbf{H}^2,\xi)
+ \delta C_1(\Gamma, \xi) - m(N{-}m)/2$, where the spectral
correction $\delta C_1$ encodes the topology of~$S$.
The claim is that $\delta C_1$ is an index pairing.

\begin{theorem}[Spectral correction as index pairing]
\label{thm:index-pairing}
Let $S = \Gamma\backslash\mathbf{H}^2$ be a compact arithmetic
surface with Laplacian~$\Delta_S$.
Then the spectral correction satisfies
\begin{equation}\label{eq:index-pairing}
  \delta C_1(\Gamma, \xi)
  \;=\; \operatorname{Tr}\bigl(f_\xi(\Delta_S)\bigr)
      - \operatorname{Tr}\bigl(f_\xi(\Delta_{\mathbf{H}^2})\bigr)
\end{equation}
where $f_\xi(\lambda) = (N{-}1)(1{+}\xi^2)/(1{-}\xi)^2 \cdot
\lambda^{-1} \cdot |\varphi_\lambda(z_0)|^2$
is the Havelock test function evaluated at the $N$-gon center.
The right side is computed by the Selberg trace formula:
\begin{equation}\label{eq:selberg-delta}
  \delta C_1(\Gamma, \xi)
  \;=\; \sum_{[\gamma] \ne 1}
    w_\gamma(\xi)\,\operatorname{csch}^2(\ell_\gamma/2),
\end{equation}
where the sum runs over nontrivial conjugacy classes of\/~$\Gamma$,
$\ell_\gamma$ is the geodesic length, and the weight is
\begin{equation}\label{eq:selberg-weight}
  w_\gamma(\xi)
  = \frac{\ell_\gamma}{2\sinh(\ell_\gamma/2)}\;
    \hat{h}_\xi(\ell_\gamma),
\end{equation}
with $\hat{h}_\xi$ the Selberg transform of the Havelock kernel
$h_\xi(r) = (N{-}1)(1{+}\xi^2)/(1{-}\xi)^2 \cdot G_{\mathbf{H}^2}(r)$.
For the hyperbolic Green's function
$G_{\mathbf{H}^2}(r) = -(2\pi)^{-1}\!\ln\tanh(r/2)$,
the Selberg transform is
$\hat{h}_\xi(t) = O(e^{-t})$ as $t \to \infty$
(exponential decay, inherited from the Fourier transform of
$\ln\tanh$), giving $w_\gamma = O(\ell_\gamma\,e^{-\ell_\gamma})$
and hence $\operatorname{csch}^2(\ell_\gamma/2)$ decay for each term.
\end{theorem}

\begin{proof}
The curvature coefficient on $S$ involves the automorphic
Green's function at the center:
$C_1(S, \xi) = C_1(\mathbf{H}^2, \xi) + \delta C_1$
where $\delta C_1$ is the difference between $G_S(z_0, z_0)$
and its universal-cover counterpart.
The spectral expansion
$G_S(z,w) = \sum_n \varphi_n(z)\overline{\varphi_n(w)}/\lambda_n$
gives the spectral side of~\eqref{eq:index-pairing}.
The Selberg trace formula \citep{Hejhal1976}
applied to the test function $f_\xi$
evaluates the trace $\operatorname{Tr}(f_\xi(\Delta_S))$ as
the identity orbital integral ($= C_1(\mathbf{H}^2, \xi)$)
plus the sum over hyperbolic conjugacy classes
(each contributing $w_\gamma\,\operatorname{csch}^2(\ell_\gamma/2)$
via the Selberg transform~\eqref{eq:selberg-weight}),
giving~\eqref{eq:selberg-delta}.

\emph{Admissibility of $f_\xi$.}
The test function $f_\xi$ is even (invariant under
$d \mapsto -d$), smooth away from $d = 0$, and has
Selberg/Harish-Chandra transform $\hat{f}_\xi(r) =
\int_0^\infty f_\xi(t)\,\Phi_r(t)\,dt$ where
$\Phi_r$ is the spherical function.
The logarithmic singularity at $d = 0$ is integrable
on $\mathbf{H}^2$ (the area element $\sinh d\,dd\,d\theta$
cancels the $\ln d$ divergence), so
$f_\xi \in L^1(\mathbf{H}^2)^{K}$ (the $K$-bi-invariant
$L^1$-functions).  The Paley--Wiener condition
$|\hat{f}_\xi(r)| \le C\,e^{-\delta|r|}$ for
$\operatorname{Im}(r) \le R$ follows from the exponential
decay $f_\xi(d) = O(e^{-d})$ for $d \to \infty$.
The absolute convergence of the geometric side follows
from the prime geodesic theorem
($\#\{\gamma : \ell_\gamma \le T\} \sim e^T / T$) and the
exponential decay of $w_\gamma$.
\end{proof}

The identification of $\delta C_1$ with the index pairing is not
merely an analogy: it is a precise application of the Baum--Connes
assembly map.

\begin{theorem}[Assembly map for the Havelock Hessian]
\label{thm:assembly-map}
Let $\Gamma \subset \mathrm{SL}(2,\mathbb{R})$ be a cocompact
torsion-free Fuchsian group, $S = \Gamma\backslash\mathbf{H}^2$.
\begin{enumerate}
  \item[\emph{(a)}]
    The Havelock Hessian $H_N^S(\xi)$ acting on
    $L^2(S) \otimes \mathbb{C}^{N-1}$ has spectral projection
    $P_- \in C^*_r(\Gamma) \otimes \mathrm{End}(\mathbb{C}^{N-1})$
    whose class $[P_-] \in K_0(C^*_r(\Gamma)) \otimes R(\mathbb{Z}_N)$
    is well-defined (away from palindromic thresholds) and
    locally constant in~$\xi$.
  \item[\emph{(b)}]
    Under the Baum--Connes assembly map
    $\mu \colon K_0^\Gamma(\underline{E}\Gamma) \to K_0(C^*_r(\Gamma))$
    (an isomorphism for $\Gamma$ with the Haagerup property,
    \citealt{Kasparov1988}):
    $[P_-]$ is the image of the $K$-homology class
    $[f_\xi] \in K_0^\Gamma(\mathbf{H}^2)$ defined by the
    Havelock test function.
  \item[\emph{(c)}]
    The canonical trace $\tau \colon K_0(C^*_r(\Gamma)) \to \mathbb{R}$
    applied to $[P_-]$ gives the $L^2$-Morse index:
    \[
      \tau(P_-)
      = \#\bigl\{m : C_1(\mathbf{H}^2,\xi)
        + \delta C_1(\Gamma,\xi)
        < \tfrac{m(N{-}m)}{2}\bigr\}.
    \]
    The eigenvalue correction $\delta C_1$ is computed by the
    Selberg trace formula
    (Theorem~\ref{thm:index-pairing}): the
    identity orbital integral gives
    $C_1(\mathbf{H}^2, \xi)$ (the local/universal-cover
    contribution), and the sum over nontrivial
    conjugacy classes gives $\delta C_1(\Gamma,\xi)$
    (the global correction from the surface topology).
\end{enumerate}
\end{theorem}

\begin{proof}
(a)~The spectral projection $P_-$ of a self-adjoint element of a
$C^*$-algebra is well-defined as long as $0$ is not in the spectrum
(equivalently: the operator is invertible).  Away from palindromic
thresholds, the Havelock eigenvalues $\lambda_m(\xi) \ne 0$ for
all~$m$, so the Havelock Hessian is invertible in each
$\mathbb{Z}_N$-sector and $[P_-]$ is defined.
Local constancy follows from the continuity of
$\lambda_m(\xi)$ and the fact that the projection changes only
when an eigenvalue crosses zero.

(b)~Since $\Gamma$ has the Haagerup property (it is a discrete subgroup
of $\mathrm{SL}(2,\mathbb{R})$), the Baum--Connes assembly map is an
isomorphism (\citealt{Kasparov1988}).
We verify that $f_\xi$ defines a $\Gamma$-equivariant
$K$-homology class $[f_\xi] \in K_0^\Gamma(\mathbf{H}^2)$ by
constructing an even Fredholm module.
Set $F = 2P_- - 1$ (the sign of $c_m + h_\xi^\Gamma$).
\begin{itemize}
\item \emph{$F^2 = 1$}: since $F = \mathrm{sgn}(c_m + h_\xi^\Gamma)$
  and the spectrum avoids zero (away from palindromic thresholds),
  $F$ is an involution.
\item \emph{$[F, u_\gamma]$ is compact for all $\gamma \in \Gamma$}:
  the off-diagonal element $h_\xi^\Gamma
  = \sum_{\gamma \ne 1} G(\cdot, \gamma\cdot)\, u_\gamma$
  is trace-class in $C^*_r(\Gamma)$.  Indeed, the matrix elements
  $|G(z_0, \gamma z_0)| = O(e^{-\ell_\gamma})$ decay
  exponentially, and
  $\sum_{\gamma \ne 1} e^{-\ell_\gamma} < \infty$ by the
  prime geodesic theorem.
  Since $h$ is trace-class, $[F, u_\gamma]
  = [\mathrm{sgn}(c_m + h),\, u_\gamma]$ is compact:
  the sign function of a trace-class perturbation of a
  scalar differs from the unperturbed sign by a compact
  operator (Birman--Solomyak trace-class scattering theory;
  see \citealt{Kato1995}, Theorem~X.4.4).
\item \emph{$\Gamma$-equivariance}: $G(z, w)$ is
  $\Gamma$-invariant, so $h_\xi^\Gamma$ commutes with the
  $\Gamma$-action and $F$ is $\Gamma$-equivariant.
\end{itemize}
The assembly map sends $[f_\xi]$ to the spectral projection of the
operator $c_m \cdot 1 + h_\xi^\Gamma$ in $C^*_r(\Gamma)$, which is
precisely~$P_-$.

(c)~For each mode~$m$, the eigenvalue on $S$ is
$\lambda_m(\Gamma,\xi) = c_m + \tau(h_\xi^\Gamma)
= [C_1(\mathbf{H}^2,\xi) + \delta C_1(\Gamma,\xi)] - m(N{-}m)/2$,
where $\delta C_1$ is computed by
Theorem~\ref{thm:index-pairing}.
The $L^2$-Morse index $\tau(P_-)$ counts the modes with
$\lambda_m < 0$; equivalently,
$\tau(P_-) = \#\{m : C_1 + \delta C_1 < f(m,N)\}$.
\end{proof}

\noindent
The theorem makes precise the statement that
\emph{the Selberg trace formula is the index theorem for the
Laplacian on~$S$, applied to the Havelock test function}
(\citealt{BarbaschMoscovici1983}; \citealt{Connes1994}, \S IV.3).
The ``identity contribution'' is the local index and the
``geometric side'' is the global correction $\delta C_1$.

\paragraph{Bolza verification.}
On the Bolza surface ($g = 2$, $|\mathrm{Aut}| = 48$,
systole $\ell = 2\operatorname{arccosh}(1{+}\sqrt{2})$),
the sum~\eqref{eq:selberg-delta} is dominated by the
$12$~systole geodesics.  The eigenfunction suppression at the
center (only trivial-rep modes of~$\mathrm{Aut}(S)$ contribute)
gives the bound $|\delta C_1| \le 2/(4\pi\lambda_{\mathrm{triv}})
\approx 0.011$ (with $\lambda_{\mathrm{triv}} \approx 15.05$,
the first trivial-representation eigenvalue),
verified numerically.

\subsection{Spectral flow stability}
\label{sec:spectral-flow}

The index pairing (Theorem~\ref{thm:index-pairing}) is a
scalar identity: it computes $\delta C_1$ as a single number.
The promotion to a $K_0$-level identity requires controlling
the \emph{spectral flow}---the net number of eigenvalue
zero-crossings as $\xi$ varies.

\begin{theorem}[Spectral flow preservation]
\label{thm:spectral-flow}
Let $S = \Gamma\backslash\mathbf{H}^2$ be compact with
spectral correction $\delta C_1(\Gamma, \xi)$.
Define the \emph{binding eigenvalue} at each palindromic
threshold $\xi^*(N,m)$ by
\[
  \lambda_{\mathrm{bind}}(N,m)
  \;=\; \min_{m' \ne m} |\lambda_{m'}(\xi^*(N,m))|.
\]
If\/ $|\delta C_1(\Gamma, \xi)| < \lambda_{\mathrm{bind}}(N,m)$
for every mode~$m$ and every $\xi \in [0,1)$, then:
\begin{enumerate}
  \item The spectral flow of $\{H_N^S(\xi)\}_{\xi \in [0,1)}$
    equals the spectral flow of $\{H_N^{\mathbf{H}^2}(\xi)\}$:
    each mode $m$ crosses zero exactly once, at
    $\xi^*(N,m) + O(\delta C_1)$.
  \item The $K_0$ classes agree:
    $[\mu_N^S(\xi)] = [\mu_N^{\mathbf{H}^2}(\xi)]
    \in R(\mathbb{Z}_N)$
    for all $\xi$ away from thresholds.
  \item In particular, $N_{\mathrm{crit}}(S) = N_{\mathrm{crit}}(\mathbf{H}^2)$.
\end{enumerate}
\end{theorem}

\begin{proof}
(i)~The correction $\delta C_1$ is mode-independent:
the quotient surface $S = \Gamma\backslash\mathbf{H}^2$ is
locally isometric to~$\mathbf{H}^2$, so the Riemannian
Havelock identity (Theorem~\ref{thm:riemannian-havelock})
applies locally to the Green's function.
The automorphic correction from the $\Gamma$-orbit sum
enters only through the mode-independent offset
$C_1^S = C_1^{\mathbf{H}^2} + \delta C_1(\Gamma, \xi)$
(the spectral resolvent at $s = 1$ of the automorphic
Laplacian, which is a scalar, not a matrix in mode space).
Therefore $\lambda_m^S(\xi)
= \lambda_m^{\mathbf{H}^2}(\xi) + \delta C_1(\Gamma, \xi)$.
At the $\mathbf{H}^2$-threshold $\xi_0 = \xi^*(N,m)$
where $\lambda_m^{\mathbf{H}^2}(\xi_0) = 0$:
$\lambda_m^S(\xi_0) = \delta C_1 \ne 0$ (since $|\delta C_1|
< \lambda_{\mathrm{bind}}$).
So mode~$m$ does not cross zero at~$\xi_0$ on~$S$;
instead it crosses at $\xi_0 + \delta\xi$ where
$\delta\xi = -\delta C_1 / (dC_1/d\xi)|_{\xi_0}$.
The crossing has the same sign
($dC_1/d\xi > 0$ is unchanged by the perturbation).
For every other mode $m' \ne m$:
$|\lambda_{m'}^S(\xi_0)|
\ge |\lambda_{m'}^{\mathbf{H}^2}(\xi_0)| - |\delta C_1|
> \lambda_{\mathrm{bind}} - |\delta C_1| > 0$,
so no extra crossings are introduced.

(ii)~Between consecutive (shifted) thresholds, all eigenvalues
have definite sign, which agrees with the $\mathbf{H}^2$ sign
since $|\delta C_1|$ is too small to flip any.

(iii)~The critical polygon number is $N_{\mathrm{crit}} =
\max\{N : [\mu_N(\xi)] = 0\ \text{for some}\ \xi > 0\}$.
Since the $K_0$ classes agree, so does $N_{\mathrm{crit}}$.
\end{proof}

The Bolza bounds (preceding paragraph) confirm: at $N = 12$,
$\xi_{\mathrm{Bolza}} \approx 0.217$, the spectral flow
on the Bolza surface equals that on~$\mathbf{H}^2$.

\subsection{The Kasparov product}
\label{sec:kk-programme}

The representation-ring invariant $[\mu_N(\xi)]$ of
\S\ref{sec:index-pairing} classifies stability phases but cannot
detect fractional Morse indices at thresholds where the gap
condition fails.
The Kasparov product provides this finer information;
the full construction is in
Supplement~S, \S\ref*{SP-sec:kasparov-construction}.
The main results are:

\begin{theorem}[Kasparov product identity]
\label{thm:kk-product}
Let $\Gamma$ be a cocompact torsion-free Fuchsian group,
$S = \Gamma\backslash\mathbf{H}^2$.
Define $h_\xi^\Gamma \in C^*_r(\Gamma)$ by
\begin{equation}\label{eq:h-element}
  h_\xi^\Gamma
  = \sum_{\gamma \in \Gamma,\, \gamma \ne 1}
    G_{\mathbf{H}^2}(z_0,\, \gamma z_0)\; u_\gamma
\end{equation}
where $G_{\mathbf{H}^2}$ is the Green's function on
$\mathbf{H}^2$ and $u_\gamma$ is the canonical unitary
in $C^*_r(\Gamma)$.
\begin{enumerate}
  \item The element $h_\xi^\Gamma$ is well-defined in $C^*_r(\Gamma)$
    with operator norm bounded by the $\ell^1$-norm:
    \begin{equation}\label{eq:op-norm-bound}
      \|h_\xi^\Gamma\|_{C^*_r}
      \;\le\; \|h_\xi^\Gamma\|_{\ell^1(\Gamma)}
      \;=\; \sum_{\gamma \ne 1} |G_{\mathbf{H}^2}(z_0, \gamma z_0)|,
    \end{equation}
    which converges by $|G(z_0,\gamma z_0)| = O(e^{-\ell_\gamma})$
    and the prime geodesic theorem.
  \item The Kasparov triple
    $\alpha_N(\xi) = \bigl(C^*_r(\Gamma)^{N-1},\;
    \pi_{\mathbb{Z}_N},\;F_\xi\bigr)$
    with $F_\xi|_{\rho_m} =
    \mathrm{sign}\bigl(c_m \cdot 1 + h_\xi^\Gamma\bigr)$
    and $c_m = C_1(\mathbf{H}^2,\xi) - m(N{-}m)/2$
    is a well-defined element of
    $KK(C^*(\mathbb{Z}_N),\; C^*_r(\Gamma))$,
    provided $\|h_\xi^\Gamma\|_{C^*_r} < \min_m |c_m|$
    (the operator-norm gap condition).
  \item Under this condition, the Kasparov product satisfies
    \begin{equation}\label{eq:kasparov-product}
      \tau_*\!\bigl(\alpha_N(\xi)
      \;\otimes_{C^*_r(\Gamma)}\; [D_S]\bigr)
      \;=\; [\mu_N(\Gamma, \xi)]
      \;\in\; R(\mathbb{Z}_N),
    \end{equation}
    where $[D_S] \in KK(C^*_r(\Gamma), \mathbb{C})$ is the
    fundamental class and $\tau_*$ is the canonical trace.
\end{enumerate}
\end{theorem}

\begin{corollary}[Kasparov product at the threshold]
\label{cor:kk-threshold}
At the palindromic threshold $\xi = \xi^*(N, m^*)$
where $c_{m^*} = 0$, the gap condition fails for
mode~$m^*$.
The Kasparov module remains well-defined with
$F_{\xi^*}|_{\rho_{m^*}} = \mathrm{sign}(h_\xi^\Gamma)$
(the sign of the full automorphic correction, not a scalar).
Its trace gives
\[
  \tau_*(\alpha_N(\xi^*) \otimes [D_S])
  = \!\sum_{m \ne m^*}\!\mathrm{sign}(c_m)\,[\rho_m]
  \;+\; \tau(P_-(h))\,[\rho_{m^*}],
\]
where $\tau(P_-(h)) \in (0, 1)$ is the fractional
$L^2$-Morse index (Theorem~\ref{thm:fractional-morse}).
This is the ``genuinely new information'' at the threshold:
the scalar spectral flow produces integers, but the
Kasparov product produces the \emph{fractional}
$\tau(P_-)$ through the von~Neumann trace on~$C^*_r(\Gamma)$.
On the Bolza surface: $\tau(P_-) = 0.644 \pm 0.004$
(non-integer, Theorem~\ref{thm:fractional-morse}).
The Baum--Connes assembly map for surface groups $\Gamma$
is an isomorphism (\citealt{Kasparov1988}; surface groups
are a-T-menable), so the Kasparov product is well-defined.
The specific value $0.644$ is evaluated computationally from
the Bolza Fuchsian group generators.
\end{corollary}

\subsection{The gap-free regime: proved results}
\label{sec:gap-free}

At a palindromic threshold ($c_m = 0$), the operator-norm
gap fails and the $K_0$ class carries genuine
surface-topology information.
The following three results hold without the gap hypothesis.

\begin{theorem}[Fractional $L^2$-Morse index]
\label{thm:fractional-morse}
Let $\Gamma$ be a torsion-free cocompact Fuchsian group and
$h_\xi^\Gamma = \sum_{\gamma \ne 1} G_{\mathbf{H}^2}
(z_0, \gamma z_0)\, u_\gamma \in C^*_r(\Gamma)$.
Then
$0 < \tau\bigl(P_-(h_\xi^\Gamma)\bigr) < 1$.
\end{theorem}

\begin{proof}
First, $h$ is self-adjoint:
$(h_\xi^\Gamma)^* = \sum_{\gamma \ne 1}
\overline{G(z_0, \gamma z_0)}\, u_{\gamma^{-1}}
= \sum_{\gamma \ne 1} G(z_0, \gamma^{-1} z_0)\, u_{\gamma^{-1}}
= h_\xi^\Gamma$,
using $G(z_0, \gamma z_0) = G(z_0, \gamma^{-1} z_0) \in \mathbb{R}$
(the Green's function is real and symmetric in the group:
$G(z, \gamma w) = G(z, \gamma^{-1} w)$ follows from $G(x,y) = G(y,x)$
and $G(\gamma x, \gamma y) = G(x,y)$).

(i)~$G_{\mathbf{H}^2}(d) > 0$ for all $d > 0$, so
$h$ has all-positive coefficients.
(ii)~$h$ has no identity coefficient ($\gamma \ne 1$), so
$\tau(h) = 0$.
(iii)~$h \ne 0$ in $C^*_r(\Gamma)$.
(iv)~The canonical trace $\tau$ on $C^*_r(\Gamma)$ is
\emph{faithful} for torsion-free $\Gamma$
(\citealt{Puschnigg2002}):
$\tau(a^* a) = 0 \Rightarrow a = 0$.
(v)~If $\mathrm{spectrum}(h) \subset [0, \infty)$: then
$h \ge 0$, $\tau(h) = 0$, and faithfulness gives $h = 0$,
contradicting~(iii).
Similarly $\mathrm{spectrum}(h) \not\subset (-\infty, 0]$.
Therefore $h$ has both positive and negative spectrum,
giving $0 < \tau(P_-) < 1$.
\end{proof}

\begin{theorem}[Wall-crossing formula]
\label{thm:wall-crossing}
At each palindromic threshold $\xi^*(N,m)$:
\[
  [\mu_N(\xi^*{+}\varepsilon)]
  - [\mu_N(\xi^*{-}\varepsilon)]
  \;=\;
  \begin{cases}
    -[\rho_m] - [\rho_{N-m}]
      & \text{if } m \ne N/2, \\[3pt]
    -[\rho_{N/2}]
      & \text{if } m = N/2.
  \end{cases}
\]
No other mode crosses zero at $\xi^*(N,m)$.
\end{theorem}

\begin{proof}
By the Havelock decomposition:
$\lambda_{m'}(\xi) = C_1(\xi) - m'(N{-}m')/2$.
At $\xi^*(N,m)$: $\lambda_m = 0$ iff
$C_1(\xi^*) = m(N{-}m)/2$.
The function $p \mapsto p(N{-}p)$ is strictly increasing
on $[1, N/2]$, so $m'(N{-}m') = m(N{-}m)$ has only the
solution $m' \in \{m,\, N{-}m\}$.
Since $C_1$ is strictly increasing:
only the pair $\{m, N{-}m\}$ crosses at $\xi^*(N,m)$.
\end{proof}

\noindent
These two theorems together establish: the $L^2$-Morse
index is \emph{provably fractional} at the palindromic
threshold (not $0$ or~$1$), and the $K_0$ class changes by a
\emph{known} element of $R(\mathbb{Z}_N)$ at each crossing.
The mechanism is Fourier interference: $h$ has all-positive
coefficients (from $G > 0$) but $\tau(h) = 0$ (no identity
coefficient), and faithfulness of $\tau$ forces the spectrum
to cross zero.

\paragraph{Bolza computation at threshold.}
On the Bolza surface at the palindromic threshold,
the $L^2$-Morse index is $\tau(P_-) \approx 0.644$:
roughly $64\%$ of the $L^2$-weight is unstable.
The detailed convergence analysis (truncation depths,
Aut-isotypic decomposition, error bounds) is in
Supplement~S, \S\ref*{SP-sec:bolza-convergence}.


\subsection{Number-theoretic extensions}
\label{sec:number-theory-extensions}

The palindromic hierarchy has deep connections to number theory

the Bolza modular form $\eta(8z)\eta(16z)$ (level~128, weight~1),
