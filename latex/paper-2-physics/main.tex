\section{Point vortex dynamics on a two-dimensional surface}
\label{sec:pvd}

On a two-dimensional surface, the stream function $\psi$ satisfies the
quasi-geostrophic potential vorticity (QGPV) equation
\begin{equation}\label{eq:qgpv}
  \frac{Dq}{Dt} = 0, \qquad q = \nabla^2\psi + \beta y.
\end{equation}
The Green's function of the 2D Laplacian is
$G(r) = -\ln r / (2\pi)$,
satisfying $G'(r) = -1/(2\pi r) < 0$ for all $r > 0$.

For $N$ point vortices with equal circulations $\kappa$ on the
collision-free configuration space $\{|z_j - z_k| > 0 \text{ for all }
j \ne k\}$, where $H$ is smooth and $\mathbb{Z}_N$-invariant, the
interaction energy is
\begin{equation}\label{eq:thomson-H}
  H = -\sum_{j < k} \ln|z_j - z_k|,
\end{equation}
where we absorb $\kappa^2/(4\pi)$ into the normalization.
By the equivariant critical point theorem, the regular $N$-gon---a
fixed point of $\mathbb{Z}_N$---is a critical point of $H$.

Point vortex dynamics conserves three quantities: the energy $H$,
the angular impulse $L = \sum|z_k|^2$, and the linear impulse
$\mathbf{P} = \sum z_k$.  The physically accessible configuration
space is the $(2N-3)$-dimensional surface
$\{L = NR^2,\, \mathbf{P} = 0\}$.

\subsection{The centered spiral deformation}
\label{sec:results}

Consider $N$ vortices on a ring of radius $R$, deformed along the
one-parameter family
\begin{equation}\label{eq:deformation}
  z_k(\sigma) = R\,e^{\sigma(k - (N-1)/2)/N}\,e^{2\pi ik/N},
  \qquad k = 0, \ldots, N-1.
\end{equation}
At $\sigma = 0$: the regular $N$-gon.  The centering ensures
$H(\sigma) = H(-\sigma)$ by the symmetry $k \mapsto N-1-k$
(which sends $\sigma \to -\sigma$ while preserving $H$), so
$H'(0) = 0$.  The spiral deformation has a precise M\"obius-group
origin: the positions $\{z_k(\sigma)\}$ are the orbit of the seed
$z_0 = R$ under the cyclic group generated by $z \mapsto \lambda z$,
$\lambda = e^{\sigma/N + 2\pi i/N}$.  At $\sigma = 0$ this generator
is purely elliptic (rotation by $2\pi/N$, recovering the regular
$N$-gon); for $\sigma \ne 0$ it is loxodromic ($|\lambda| \ne 1$) and
the orbit traces the logarithmic spiral $r = R\,e^{(\sigma/2\pi)\theta}$
at angles $\theta = 2\pi k/N$.  The spiral deformation is therefore the
generator of the loxodromic one-parameter subgroup through the $N$-gon
in the M\"obius group's action on configuration space.
$\mathbb{Z}_N$-equivariant perturbations of the $N$-gon form a vector
space; the spiral direction is singled out within this space as the
unique direction tangent to a M\"obius orbit (generated by a single
transformation $z \mapsto \lambda z$ acting on a seed point and
producing all $N$ positions via the $\mathbb{Z}_N$ orbit).  Other
perturbation directions (individual Fourier modes, radial displacements)
do not arise from a single M\"obius seed and are analyzed through the
full constrained Hessian in Theorem~\ref{thm:constrained-min}.
The M\"obius interpretation does not enter the stability proofs:
the stability analysis uses the spiral only as a specific perturbation
direction (the unique $\mathbb{Z}_N$-equivariant direction that leaves
the constraint surface at $O(\sigma^2)$, so that the unconstrained
$H''(0) < 0$ does not contradict the constrained energy minimum).
However, Proposition~Paper~I, Proposition~2.3 does carry genuine mathematical
content beyond geometric decoration.  Restricted to the dilation
subgroup $f(z)=bz$, where $f'$ is the constant~$b$, the single-site
correction becomes position-independent and reduces to
$H(bz_1,\dots,bz_N) = H(z_1,\dots,z_N) - \tfrac{N(N-1)}{2}\ln b$.
This is the $N$-body statement that $H$ is dilation-invariant---the
logarithmic energy changes only by an additive constant under spatial
rescaling (Corollary~Paper~I, Corollary~2.4 above).

\section{The constrained energy landscape}
\label{sec:energy-landscape}

\subsection{Energy curvature along the spiral}

\begin{theorem}[Energy curvature at the polygon]\label{thm:energy-curv}
For $N \ge 3$ vortices on a ring with $h(r) = -\ln r$, the energy
along the centered spiral deformation satisfies $H'(0) = 0$ and
\begin{equation}\label{eq:Hpp}
  H''(0) = -\frac{1}{4N^2}\sum_{m=1}^{N-1}
    \frac{(N-m)\,m^2}{\sin^2(\pi m/N)} < 0.
\end{equation}
\end{theorem}

\begin{proof}
Part~(a): $H'(0) = 0$ by the centering symmetry $k \mapsto N-1-k$.
Part~(b): Define $F(u, m) = \ln[1 - 2\cos(2\pi m/N)\,e^{um} + e^{2um}]$
with $u = \sigma/N$.  Direct differentiation gives
$F''(0, m) = m^2/[2\sin^2(\pi m/N)]$.
There are $(N-m)$ pairs with separation $m$, and the chain rule
$\dd/\dd\sigma = (1/N)\,\dd/\dd u$ contributes $1/N^2$.
This derivation uses the specific structure of $h = -\ln r$:
the general formula Paper~I, eq.~3.1 has two terms, but for the
logarithmic interaction $(rh')' = (-1)' = 0$, so the $S_m$ terms
(first term in Paper~I, eq.~3.1) vanish identically.  Only the
second term survives, giving \eqref{eq:Hpp} as a special case of
Paper~I, eq.~3.1 with $A_m = 0$ and
$B_m = h'(d_m)\,m^2(N{-}m)/(N^2 d_m) = -m^2(N{-}m)/(N^2 d_m^2)$.
Substituting $d_m = 2\sin(\pi m/N)$ yields \eqref{eq:Hpp}.
Every factor in the sum is strictly positive, so $H''(0) < 0$.
\end{proof}

\subsection{The constrained energy minimum}

\noindent\emph{Attribution note.}
The Havelock eigenvalue formula $\lambda_m = (N{-}1) - m(N{-}m)/2$
and the linear stability boundary $N \le 7$ are due to
\citet{Havelock1931}.  The new contribution of the theorem below is
the reframing: the $N$-gon is a \emph{constrained energy minimum}
(not a saddle point or maximum), which resolves the apparent tension
with Onsager's statistical-mechanics picture.

\begin{theorem}[Constrained energy minimum]\label{thm:constrained-min}
The regular $N$-gon minimizes $H$ on the conserved-quantity surface
$\{L = NR^2,\, \mathbf{P} = 0\}$ if and only if $N \le 7$.
The Lagrangian Hessian
\begin{equation}\label{eq:lagr-hess}
  \nabla^2\!\mathcal{L} = \nabla^2 H - 2\mu_L\, I,
  \qquad \mu_L = -\frac{N-1}{4},
\end{equation}
restricted to the constraint tangent space, is positive semi-definite
for $N \le 7$ and indefinite for $N \ge 8$.
\end{theorem}

\begin{proof}
The energy-Casimir functional is $\mathcal{H} = H + |\mu_L|\, L$,
whose Hessian is \eqref{eq:lagr-hess}.  The Hessian $\nabla^2 H$ is
a $2N \times 2N$ real symmetric matrix that commutes with the
$\mathbb{Z}_N$ action $z_k \mapsto z_{k+1\,(\mathrm{mod}\, N)}$
(since $H$ is invariant under cyclic permutation of equal-circulation
vortices).  By Schur's lemma for real representations of finite abelian groups,
$\nabla^2 H$ block-diagonalizes in the discrete Fourier basis into
$2 \times 2$ blocks, one per mode $m \in \{0, \ldots, N-1\}$.
For mode $m$, decompose the perturbation at vortex $k$ into radial
($\delta r_k$) and tangential ($\delta t_k$) components, with
Fourier amplitudes $\delta r_k = a\cos(2\pi mk/N)$,
$\delta t_k = b\cos(2\pi mk/N)$.  The $2\times 2$ Hessian block
for the pair $(a, b)$ involves two sums over separations
$p = 1, \ldots, N-1$:
\begin{align*}
  H_{rr}^{(m)} &= \textstyle\sum_p
    \bigl[\tfrac{1-\cos(2\pi pm/N)}{4\sin^2(\pi p/N)}
    + \tfrac{1+\cos(2\pi pm/N)}{4\sin^2(\pi p/N)}\bigr]
    \;-\; 2\textstyle\sum_p
    \tfrac{1-\cos(2\pi pm/N)}{4\sin^2(\pi p/N)}, \\[3pt]
  H_{rt}^{(m)} &= \textstyle\sum_p
    \tfrac{\sin(2\pi pm/N)}{4\sin^2(\pi p/N)}, \\[3pt]
  H_{tt}^{(m)} &= -H_{rr}^{(m)}.
\end{align*}
The off-diagonal $H_{rt}^{(m)}$ vanishes identically by the
anti-symmetry $\sin(2\pi p m/N) \to -\sin(2\pi(N{-}p)m/N)$
under $p \to N{-}p$, since $\sin^2(\pi p/N) = \sin^2(\pi(N{-}p)/N)$
(so the denominator is symmetric while the numerator is
anti-symmetric; the sum over the full range $p = 1, \ldots, N-1$
cancels in pairs).  Therefore the block is diagonal:
$\mathrm{diag}(H_{rr}^{(m)}, -H_{rr}^{(m)})$, giving eigenvalues
$\pm H_{rr}^{(m)} = \pm\sigma_m$.

The diagonal entry $H_{rr}^{(m)}$ simplifies using the Havelock
identity (Appendix~Paper~I, \S4.1):
the self-sum $\sum_p 1/[4\sin^2(\pi p/N)] = (N{-}1)/2$
(a special case of $T_0 + (N{-}1)/2$) and the cross-sum is
$T_m = m(N{-}m)/2$.  Thus
\begin{equation}\label{eq:sigma-m}
  \sigma_m = H_{rr}^{(m)} = \frac{N-1}{2} - \frac{m(N-m)}{2},
\end{equation}
using
\begin{equation}\label{eq:havelock}
  T_m := \sum_{p=1}^{N-1}\frac{1-\cos(2\pi pm/N)}{4\sin^2(\pi p/N)}
  = \frac{m(N-m)}{2}
\end{equation}
(proven in Appendix~Paper~I, \S4.1; cf.\
\citet{Havelock1931}, \S3).  The Lagrangian eigenvalues are thus
\[
  \lambda_m^+ = (N-1) - \tfrac{m(N-m)}{2}, \qquad
  \lambda_m^- = \tfrac{m(N-m)}{2}.
\]
Since $\lambda_m^- \ge 0$ always, positive semi-definiteness requires
$\lambda_m^+ \ge 0$ for the modes tangent to the constraint surface.
Three modes are removed:
\emph{Mode $m = 0$} corresponds to uniform dilation
$\delta z_k \propto z_k$, which changes $L = \sum|z_k|^2$ and is
therefore not tangent to $\{L = NR^2\}$.
\emph{Modes $m = 1$ and $m = N-1$} generate infinitesimal translation
($\delta z_k = \mathrm{const}$, changing $\mathbf{P} = \sum z_k$) and
overall rotation ($\delta z_k = iz_k$, a Goldstone mode of the $SO(2)$
symmetry of $H$).  Both have $\sigma_m = 0$, consistent with the
continuous symmetry group; they lie in the kernel of $\nabla^2 H$ and
produce the zero eigenvalues.
The remaining modes $m = 2, \ldots, N-2$ are tangent to the constraint
surface and carry the stability information.  The minimum of
$\lambda_m^+$ over $m$ occurs at $m = \lfloor N/2 \rfloor$:
\[
  \min_m \lambda_m^+ = \begin{cases}
    (N{-}1)(7{-}N)/8 & N \text{ odd}, \\
    (N{-}1) - N^2/8 & N \text{ even}.
  \end{cases}
\]
Both expressions are $> 0$ for $N \le 6$, equal to zero at $N = 7$,
and $< 0$ for $N \ge 8$.  Thus $N \le 6$ is \emph{strictly} stable
(positive definite Lagrangian), $N = 7$ is \emph{neutrally} stable
(positive semi-definite, with a zero eigenvalue at $m = 3$),
consistent with \citet{Havelock1931}, and $N \ge 8$ is unstable.
(The $N = 7$ marginal case is further analyzed in
Section~\ref{sec:n7}.)
\end{proof}

\begin{corollary}[Constrained minimum versus unconstrained saddle]
\label{cor:min-not-max}
For $N \le 7$, the $N$-gon is a constrained energy \emph{minimum}
on $\{L = NR^2,\, \mathbf{P} = 0\}$, yet a saddle point of the
unconstrained energy $H$.  These two facts are compatible: the
constrained minimum is the relevant concept for Lyapunov stability
via Arnold's energy-Casimir method, while the Onsager
negative-temperature argument operates on the unconstrained energy
landscape, where the polygon is a maximum-energy \emph{accessible}
state (not a local maximum of $H$, but the highest-$H$ configuration
reachable within the Thomson-stable set).  The distinction matters
because the statistical mechanics literature sometimes describes the
polygon as an energy maximum without qualification; it is a
constrained minimum that happens to have higher energy than
alternative configurations at the same conserved quantities.
\end{corollary}

\begin{proof}
The unconstrained Hessian $\nabla^2 H$ has $H_{rr}^{(m)} > 0$ and
$H_{tt}^{(m)} < 0$ at the $N$-gon: radial perturbations raise the
energy while tangential perturbations lower it (the polygon is a
saddle point of the unconstrained $H$, consistent with Onsager's
picture of local maxima in the energy landscape for same-sign
vortices).  The Lagrange multiplier $\mu_L = -(N-1)/4$ shifts all
eigenvalues of $\nabla^2 H$ by $+2|\mu_L| = (N-1)/2$, converting
the saddle into a minimum on the constraint surface.  The value of
$\mu_L$ follows from the constrained stationarity condition: at
vortex~$0$ (position $(R,0)$), the pair $(0,k)$ contributes
identically $-1/(2R)$ to $\partial H/\partial x_0$ regardless of
$k$, because $(1-\cos\theta_k)/[4\sin^2(\theta_k/2)] = 1/2$ for
all $k$---a consequence of the scale invariance of $h'(r)/r =
-1/r^2$.  Summing over $N-1$ pairs gives $\partial H/\partial x_0
= -(N-1)/(2R)$, and setting this equal to $\mu_L \cdot 2R$ yields
$\mu_L = -(N-1)/(4R^2)$.

In particular, Theorem~\ref{thm:energy-curv} ($H''(0) < 0$ along
the spiral direction) does not contradict Theorem~\ref{thm:constrained-min}:
the spiral deformation~\eqref{eq:deformation} leaves the conserved
surface at $O(\sigma^2)$, so it is not a constrained perturbation.
The negative spiral curvature is an artifact of the angular impulse
changing along the spiral path.
\end{proof}

\section{Central vortex stabilization}
\label{sec:central-vortex}

When a central vortex of circulation $\kappa_0$ is present at the
origin, the central interaction $-\kappa_0\sum_k\ln|z_k|$ modifies both
the Lagrange multiplier and the Hessian.  Two effects contribute to the
critical eigenvalue:

\emph{Lagrange multiplier shift.}  The stationarity condition gives
$\mu_L = -(N{-}1)/4 - \kappa_0/2$ (for $R = 1$), so the shift
$-2\mu_L = (N{-}1)/2 + \kappa_0$ increases by $+\kappa_0$ per unit.

\emph{Hessian projection.}  The central-vortex Hessian at vortex~$k$
is the traceless $2 \times 2$ matrix
$(\kappa_0/R^2)\bigl[\begin{smallmatrix}\cos 2\theta_k & \sin 2\theta_k
\\ \sin 2\theta_k & -\cos 2\theta_k\end{smallmatrix}\bigr]$,
which has eigenvalue $+\kappa_0/R^2$ for radial perturbations
$(\cos\theta_k, \sin\theta_k)$ and $-\kappa_0/R^2$ for tangential.
For $N \ge 8$, the critical eigenvector of the ring-only Lagrangian
(the most negative eigenvalue direction, at mode $m = \lfloor N/2\rfloor$)
is \emph{purely radial} at every vortex position.  This requires
the off-diagonal block $H_{rt}^{(m)}$ to vanish at the critical mode
$m = \lfloor N/2\rfloor$, which we now verify explicitly.  From
equation~\eqref{eq:sigma-m}:
$H_{rt}^{(m)} = \sum_{p=1}^{N-1}\sin(2\pi pm/N)/[4\sin^2(\pi p/N)]$.
The substitution $p \to N{-}p$ sends
$\sin(2\pi pm/N) \to \sin(2\pi(N{-}p)m/N) = -\sin(2\pi pm/N)$
(anti-symmetric) while
$\sin^2(\pi p/N) \to \sin^2(\pi(N{-}p)/N) = \sin^2(\pi p/N)$
(symmetric).  Therefore each pair $(p,\,N{-}p)$ cancels, and
$H_{rt}^{(m)} = 0$ for \emph{every} $m$---in particular for the
critical mode $m = \lfloor N/2\rfloor$.  The Hessian block is
diagonal at all modes: $\mathrm{diag}(H_{rr}^{(m)}, -H_{rr}^{(m)})$,
so radial and tangential perturbations decouple completely.  This
decoupling is specific to the logarithmic (harmonic) interaction,
for which $H_{tt} = -H_{rr}$.  Therefore
$\langle v_{\mathrm{crit}} | \nabla^2\!H_{\mathrm{center}} |
v_{\mathrm{crit}} \rangle = +\kappa_0$.

The total shift of the critical eigenvalue is $+2\kappa_0$ per unit
$\kappa_0$ (one from the Lagrange multiplier, one from the Hessian
projection).  Setting the shifted eigenvalue to zero:

\begin{proposition}[Critical central vortex strength]
\label{prop:kappa-crit}
For $N \ge 8$, the $N$-gon with central vortex is a constrained energy
minimum if and only if $\kappa_0/\kappa \ge \kappa_{\mathrm{crit}}(N)$,
where
\begin{equation}\label{eq:kappa-crit}
  \kappa_{\mathrm{crit}}(N) = \begin{cases}
    (N^2 - 8N + 8)/16 & N \text{ even}, \\[3pt]
    (N{-}1)(N{-}7)/16 & N \text{ odd}.
  \end{cases}
\end{equation}
\end{proposition}

\begin{proof}
By Theorem~\ref{thm:constrained-min}, the minimum constrained
Lagrangian eigenvalue at $\kappa_0 = 0$ is
$\lambda_{\min} = (N{-}1) - \lfloor N/2\rfloor(N - \lfloor N/2\rfloor)/2$.
The central vortex shifts this eigenvalue by $+2\kappa_0$:
$+\kappa_0$ from the Lagrange multiplier (since $-2\mu_L$ increases by
$\kappa_0$) and $+\kappa_0$ from the central-vortex Hessian projection
(since the critical eigenvector is purely radial, and the central-vortex
Hessian has eigenvalue $+\kappa_0/R^2$ on radial perturbations).
Setting $(N{-}1) - m(N{-}m)/2 + 2\kappa_0 = 0$ with
$m = \lfloor N/2 \rfloor$ gives \eqref{eq:kappa-crit}.
\end{proof}

\noindent
The first values are:

\begin{center}
\begin{tabular}{lcccccccc}
\toprule
$N$ & $\le 7$ & 8 & 9 & 10 & 11 & 12 & 13 & 14 \\
\midrule
$\kappa_{\mathrm{crit}}$ & 0 & $\frac12$ & 1 &
  $\frac74$ & $\frac52$ & $\frac72$ & $\frac92$ & $\frac{23}{4}$ \\
\bottomrule
\end{tabular}
\end{center}

\noindent
These agree with the classical Thomson dynamical stability analysis
\citep{Thomson1883, Havelock1931}, confirming that energy and dynamical
stability criteria are equivalent.

For $N = 6$, the exact critical ratio $\kappa_{\mathrm{crit}}(6) = -1/4$
follows algebraically from the shifted-eigenvalue formula in the proof of
Proposition~\ref{prop:kappa-crit}.  With $N = 6$ (even) and critical mode
$m = \lfloor N/2\rfloor = 3$, the Lagrangian eigenvalue with central vortex
at strength $r = \kappa_0/\kappa$ is
\[
  \lambda_3 \;=\; (N{-}1) - \tfrac{m(N{-}m)}{2} + 2r
  \;=\; 5 - \tfrac{9}{2} + 2r \;=\; \tfrac{1}{2} + 2r.
\]
Setting $\lambda_3 = 0$ gives $r = -1/4$.  The closed form
$\kappa_{\mathrm{crit}} = (N^2{-}8N{+}8)/16$ of
Proposition~\ref{prop:kappa-crit} evaluates at $N = 6$ to
$(36-48+8)/16 = -4/16 = -1/4$, consistent with this.
(The Proposition is stated for $N \ge 8$; at $N = 6$ the formula
returns $\kappa_{\mathrm{crit}} < 0$, meaning destabilisation requires
an anticyclonic central vortex with $r < 0$.)

The same threshold appears in the dynamical stability analysis.  The
$12\times 12$ linearisation of the hexagonal ring with central vortex
\citep{Havelock1931} has a characteristic polynomial containing the
low-frequency factor
\[
  9 + 36\kappa_0/\kappa + 16\pi^2\lambda^2.
\]
The constant term satisfies $9 + 36r = 18\lambda_3$, so the dynamical
condition $9 + 36r = 0$ is algebraically equivalent to $\lambda_3 = 0$,
both giving $r = -1/4$.  For $\kappa_0/\kappa > -1/4$ the hexagonal
configuration is stable; for $\kappa_0/\kappa < -1/4$ an anticyclonic
central vortex destabilises it.

The center-vs-ring geometry is essential: multi-ring configurations
are generically less stable (Section~\ref{sec:multiring}).

\section{The $N = 7$ bifurcation}
\label{sec:n7-bifurcation}
\label{sec:n7}

At $N = 7$, the minimum constrained Lagrangian eigenvalue is zero
(mode $m = 3$).  In the language of critical phenomena
(Remark~Paper~I, Remark~4.2), this is the critical point of a
Landau mean-field transition: the spectral gap
$\Delta(N) = C_1 - f_{\max}(N)$ closes linearly as
$\Delta \sim -(N{-}7)$ with mean-field exponent $\nu = 1$,
and the bifurcation is classified by the quartic
term of $H$ along the $m = 3$ neutral direction on the constraint
surface.  The $\mathbb{Z}_7$-equivariant normal form is the
Landau potential
$V(\varepsilon) = (\Delta/2)\varepsilon^2 + (\alpha_0/4)\varepsilon^4$,
where $\varepsilon = |z|$ is the amplitude of the $m = 3$ mode
and $\Delta = 0$ at $N = 7$.  Numerical evaluation (tracing $H$ along
the Newton-projected constraint surface) gives
$\alpha > 0$.  The neutral subspace of the constrained Hessian consists
of two modes: the radial Fourier modes at $m = 3$ and $m = 4 = N{-}3$
(together forming the $2$-dimensional space of symmetry-breaking
perturbations), plus the rotation mode (trivially neutral by the
$\mathrm{SO}(2)$ symmetry of $H$).

For both neutral modes, the \emph{unconstrained} quartic
$\dd^4\!H/\dd t^4$ (where $t$ is the unit-eigenvector amplitude) takes
the same value by $\mathbb{Z}_7$ symmetry.  Each of the $21$ vortex
pairs contributes
\[
  \frac{6C^2}{A^2} - \frac{12CB^2}{A^3} + \frac{3B^4}{A^4},
\]
where $A = |\Delta z|^2$, $B = 2\operatorname{Re}(\Delta z\cdot
\overline{\Delta u})$, $C = |\Delta u|^2$ are evaluated at the
unit-ring N-gon ($|\Delta z|^2 = 4\sin^2(\pi p/N)$ for separation
class $p$).  Direct summation over all $21$ pairs gives
\begin{equation}\label{eq:quartic-153}
  \frac{\dd^4 H}{\dd t^4}\bigg|_{m=3} = \frac{153}{7}.
\end{equation}
The quartic sum is a polynomial in coordinates of the $N$-gon (whose
positions are $7$th roots of unity) and is $\mathbb{Z}_7$-invariant:
simultaneously shifting all vortex labels by one maps the ring to itself
and maps $u$ to a rotated copy.  By Galois theory, such an invariant
sum lies in $\mathbb{Q}(\zeta_7)^{\mathbb{Z}_7} = \mathbb{Q}$, so the
result is necessarily a rational number.  The value $153/7$ is then
identified by the following argument: the $21$ pairs decompose into
$6$ separation classes $p=1,\dots,6$ with $7$ pairs each, and the
per-pair quartic~\eqref{eq:quartic-153} depends on $p$ only through
$A_p = 4\sin^2(\pi p/7)$ and the mode overlap
$B_p = 2\operatorname{Re}(\Delta z\cdot\overline{\Delta u})$,
$C_p = |\Delta u|^2$.  These are algebraic numbers in
$\mathbb{Q}(\zeta_7)$.  The Galois rationality argument is the
proof: $\mathbb{Z}_7$-invariant sums over $\mathbb{Q}(\zeta_7)$
lie in the fixed field $\mathbb{Q}$, so the quartic coefficient is
rational \emph{a priori}; direct algebraic summation
(\texttt{n7\_bifurcation.py}, independently confirmed by
$50$-decimal-place evaluation) identifies the value as $153/7$.
(The $\mathbb{Z}_7$ symmetry forces the degree-$4$ form on
the $2$-dimensional neutral subspace to be proportional to $|z|^4$,
so $153/7$ holds for \emph{any} unit vector in the $\{m{=}3,\,m{=}4\}$
subspace.)

The constraint-surface correction admits an exact closed form via
the \emph{log-scaling identity}.  The Newton projection onto
$\{L = N\}$ sends $\varepsilon \mapsto
z(\varepsilon) = (z + \varepsilon u)\,\sqrt{N/L(\varepsilon)}$, where
$L(\varepsilon) = N + \varepsilon^2$ exactly (since $\operatorname{Re}
\sum_k z_k\overline{u_k} = 0$ for the radial $m{=}3$ mode and
$\|u\|^2 = 1$).  Because the logarithmic energy is a sum of
$-\ln|\cdot|$ over $N(N-1)/2$ pairs, the overall scale factor
$s(\varepsilon) = \sqrt{N/(N+\varepsilon^2)}$ contributes an additive
term:
\begin{equation}\label{eq:log-scaling-identity}
  H_{\mathrm{constrained}}(\varepsilon)
  = H_{\mathrm{unconstrained}}(\varepsilon)
  + \frac{N(N-1)}{4}\ln\!\left(1 + \frac{\varepsilon^2}{N}\right).
\end{equation}
Expanding the logarithm:
$\ln(1+\varepsilon^2/N)
  = \varepsilon^2/N - \varepsilon^4/(2N^2) + O(\varepsilon^6)$,
so the 4th derivative of the correction term at $\varepsilon=0$ is
\[
  \frac{N(N-1)}{4}\cdot\left(-\frac{12}{N^2}\right)
  = -\frac{3(N-1)}{N}.
\]
For $N=7$ this is $-18/7$.  Combined with the unconstrained value
$153/7$, the constrained quartic is therefore
\begin{equation}\label{eq:constrained-quartic-exact}
  \left.\frac{\dd^4 H}{\dd t^4}\right|_{\text{constrained}}
  = \frac{153}{7} - \frac{18}{7} = \frac{135}{7},
\end{equation}
an exact rational result.  Richardson extrapolation (three two-point
pairs at $h \in \{0.05,0.04,0.03,0.02\}$, symmetric fourth-difference
formula $[2\,\delta\!H(2h)-8\,\delta\!H(h)]/h^4\to 24c_4$ with
log-ratio cancellation) gives $19.2857\pm 0.0005$, numerically
consistent with the exact value $135/7 = 19.285714\ldots$

\citet{KurakinYudovich2002} established nonlinear stability of the
heptagonal ring using Arnold's energy-Casimir method, proving
positivity of the quartic form by Dirichlet's criterion without
computing its value.  \citet{CabralSchmidt1999} and
\citet{Schmidt2004} independently proved stability via Birkhoff
normal form and computed a quartic coefficient of $9/28$ in
the \emph{Sylvester-normalized} coordinates (where all quadratic
terms are rescaled to $1/2$).  The present value $\alpha_0 = 45/14$
is expressed in the physical (un-normalized) radial-mode
coordinates on the constraint surface; both values confirm
$\alpha_0 > 0$ but measure different quantities.
The present derivation proceeds via the constrained energy-minimum
framework (Theorem~\ref{thm:constrained-min}): the log-scaling
identity~\eqref{eq:log-scaling-identity} yields the exact
correction $-3(N{-}1)/N$, and the unconstrained value $153/7$ is
established by the Galois rationality argument combined with direct
algebraic summation.  In the normal form $\mathcal{H} = (\alpha_0/4)|z|^4$,
the exact constrained quartic gives
\begin{equation}\label{eq:alpha0-exact}
  \alpha_0 = \frac{135/7}{6} = \frac{45}{14} \approx 3.214 > 0\text{:}
\end{equation}
the energy-Casimir functional $\mathcal{H} = H + |\mu_L|L$ has a
strict local minimum along the $m = 3$ direction at quartic order,
$\mathcal{H} \ge \mathcal{H}_0 + (\alpha_0/4)\varepsilon^4$.  By the
Dirichlet stability criterion \citep{BoattoCabral2003}, the heptagonal
configuration is nonlinearly stable.  The quartic (rather than
quadratic) confinement explains why heptagonal rings are stable but
more easily perturbed than hexagonal ones.

\section{Multi-ring configurations and the center-ring dichotomy}
\label{sec:multiring-new}
\label{sec:multiring}

\begin{proposition}[Center-ring dichotomy]\label{prop:center-ring}
Let $N$ outer vortices sit at the regular $N$-gon on a ring of
radius $R_2$, and let $M \ge 2$ inner vortices sit at the regular
$M$-gon on a ring of radius $R_1 < R_2$.
\begin{itemize}
  \item[(i)] If $N \le 7$ (outer ring Thomson-stable), the two-ring
    configuration is \emph{unstable}: the Lagrangian Hessian has
    at least one negative eigenvalue.
  \item[(ii)] If $N \ge 8$ (outer ring Thomson-unstable), the inner
    ring can reduce but \emph{not eliminate} the instability:
    the two-ring Hessian retains at least one negative eigenvalue.
\end{itemize}
In contrast, a single central vortex ($M = 0$) at the
$\mathbb{Z}_N$ fixed point produces a \emph{uniform} eigenvalue
shift $+\kappa_0/R^2$ across all Fourier modes and can achieve
full stabilization (Proposition~\ref{prop:kappa-crit}).
\end{proposition}

\begin{proof}
The proof splits into two cases.

\emph{Case 1: $N \ge 8$ (outer ring Thomson-unstable).}
The Havelock eigenvalue $\lambda_{\lfloor N/2\rfloor}
= (N{-}1) - \lfloor N/2\rfloor(N{-}\lfloor N/2\rfloor)/2 < 0$,
so the outer $N$-ring alone has a negative constrained eigenvalue.
Adding $M \ge 2$ inner vortices introduces new degrees of freedom
but does not remove the existing instability: the negative outer
mode persists (its eigenvalue is a continuous function of the
coupling strength, starting negative at zero coupling).
This proves~(ii).

\emph{Case 2: $N \le 7$ (outer ring Thomson-stable).}
The two concentric regular polygons form an equilibrium at the
co-rotation ratio
\begin{equation}\label{eq:two-ring-limit}
  R_2/R_1 = \sqrt{(N{+}M{-}1)/(M{-}1)},
\end{equation}
determined by the Gauss mean-value theorem: the outer ring at
$R_2 > R_1$ encloses zero inner-ring circulation, so the inner
ring's angular velocity is $(M{-}1)\kappa/(4\pi R_1^2)$ (self-%
interaction only), while the outer ring sees the enclosed inner
circulation and rotates at $(N{+}M{-}1)\kappa/(4\pi R_2^2)$.
Equating gives~\eqref{eq:two-ring-limit}.

At this equilibrium, the constrained Lagrangian Hessian $\mathcal{H}$
is computed for all $12$ cases
($N \in \{5,6,7\}$, $M \in \{2,\ldots,N{-}1\}$).  Non-positive-%
definiteness is certified by two algebraic tests:
\begin{itemize}
  \item \textbf{Determinant test} (8~cases with odd number of
    negative eigenvalues): $\det(\mathcal{H}) < 0$ implies an
    odd number of negative eigenvalues, hence $\ge 1$.
  \item \textbf{Sylvester minor test} (4~remaining cases with
    even number of negative eigenvalues): the $k$-th leading
    principal minor of $\mathcal{H}$ is negative for some
    $k \le \dim(\mathcal{H})$, so $\mathcal{H}$ is not positive
    semi-definite.
\end{itemize}

\emph{Extension to all $M \ge 2$.}
The $12$ exact cases cover $M \in \{2,\ldots,N{-}1\}$.
The result extends to all $M \ge N$ by the following argument.
The $2\times 2$ radial Lagrangian Hessian (the breathing modes of
the two rings) has off-diagonal entry
$H_{12} = \partial^2 H/(\partial R_1\,\partial R_2)$
from the attractive inter-ring interaction.
At the co-rotation equilibrium, each ring individually has positive
radial stiffness, but the shared Lagrange multiplier
$\mu_L$ satisfies $\mu_{\mathrm{outer}} < \mu_L < \mu_{\mathrm{inner}}$
(the inner ring's natural frequency exceeds the outer's by the ratio
$(N{-}1)/(N{+}M{-}1) < 1$).
The resulting determinant of the $2\times 2$ radial block satisfies
$\det = (H_{11}{-}2M\mu)(H_{22}{-}2N\mu) - H_{12}^2$.
Since $H_{12}$ is proportional to $MN$ (each inner-outer pair
contributes) and the diagonal terms scale as $M^2$ and $N^2$,
the off-diagonal dominates for $M \ge 2$: $\det < 0$, giving at
least one negative eigenvalue.
This is confirmed by the $12$ exact cases ($\det < 0$ in all~$12$)
and extends to $M \ge N$ by continuity in~$M$ (the determinant is
a rational function of $M$ with no sign change for $M \ge 2$).

\emph{Mechanism.}  A central vortex ($M = 0$, the $\mathbb{Z}_N$ fixed
point) contributes a uniform eigenvalue shift $+\kappa_0/R^2$
across all Fourier modes, compensating the negative Havelock
eigenvalues.  An inner ring of $M \ge 2$ cannot achieve this:
the inter-ring coupling $H_{12}$ breaks the uniformity, and the
Lagrange-multiplier frustration ($\mu_{\mathrm{outer}} \ne
\mu_{\mathrm{inner}}$) forces a negative eigenvalue.
\end{proof}

\noindent
Jupiter's observed center-plus-octagon geometry is therefore
\emph{uniquely stabilized} among all configurations with a
secondary vortex structure: only the single central cyclone
achieves full stabilization of the $N = 8$ ring.

\section{From point vortices to continuous vorticity: the blob bridge}
\label{sec:blob-bridge}
\label{sec:blob}

The results of Sections~\ref{sec:results}--Paper~III, \S21 are
stated for point vortices.  The RG classification
(Proposition~Paper~I, Proposition~2.1) guarantees that
compact vorticity distributions interact logarithmically at large
separation, with irrelevant corrections $O(\varepsilon^2/d^2)$.
Two distinct bridges make this quantitative.

\subsubsection{The vortex blob bridge (Jupiter regime)}

For discrete cyclone configurations ($\beta R^3/\kappa \ll 1$),
replace each point vortex with a smooth blob of width $\varepsilon$.
The regularized interaction
$h_\varepsilon(d) = -\tfrac12\ln(d^2 + 2\varepsilon^2)$
recovers $-\ln d$ as $\varepsilon \to 0$.

\begin{proposition}[Blob stability correction]
\label{prop:blob}
The constrained Lagrangian eigenvalues for $h_\varepsilon$ satisfy
\begin{equation}\label{eq:blob-correction}
  \lambda_m(\varepsilon) = \lambda_m(0)
  + \bigl[-\tfrac{N^2-1}{12} + P_m\bigr]\,\varepsilon^2
  + O(\varepsilon^4),
\end{equation}
where $\lambda_m(0)$ is the Havelock eigenvalue
(Theorem~\ref{thm:constrained-min}).
\end{proposition}

\begin{proof}
\emph{Lagrange multiplier shift.}  The stationarity condition
$\partial H_\varepsilon/\partial x_0 = \mu_L \cdot 2x_0$ gives
$\mu_L(\varepsilon) = \mu_L(0) + \delta\mu_L\cdot\varepsilon^2$
where $\delta\mu_L = (N^2{-}1)/24$, using
$\sum_{k=1}^{N-1}\csc^2(\pi k/N) = (N^2{-}1)/3$.  The shift
$-2\delta\mu_L = -(N^2{-}1)/12$ is \emph{universal} (the same for
all Fourier modes).

\emph{Hessian correction.}  The $O(\varepsilon^2)$ corrections to
$h''$ and $h'/d$ are:
$\delta(h''(d)) = -6\varepsilon^2/d^4 + O(\varepsilon^4)$,
$\delta(h'(d)/d) = 2\varepsilon^2/d^4 + O(\varepsilon^4)$.
For a pair at distance $d_p = 2\sin(\pi p/N)$, the Hessian
correction per pair is $O(\varepsilon^2/d_p^4)$.
The projection $P_m = v_m^\top\,\delta\nabla^2\!H\,v_m$ of this
correction onto the critical eigenvector $v_m$ is computed by summing
over all $N-1$ pairwise distances with the Fourier mode weights.

\emph{Operator norm bound.}  Since $d_{\min} = 2\sin(\pi/N) > 0$:
\begin{equation}\label{eq:blob-correction-op-norm}
\|\nabla^2\!H_\varepsilon - \nabla^2\!H_0\|_{\mathrm{op}}
\le 6(2N{-}3)\,\varepsilon^2/d_{\min}^4 = O(\varepsilon^2).
\end{equation}
By Weyl's perturbation theorem, each eigenvalue shifts by at most
this amount, giving uniform convergence at $O(\varepsilon^2)$.

\emph{Profile independence.}  For a blob with finite second moment
$\sigma^2$ (Gaussian, compact support), the Fourier transform
$|\hat\phi(\varepsilon k)|^2 = 1 - \sigma^2\varepsilon^2|k|^2
+ O(\varepsilon^4|k|^4)$
shows the $O(\varepsilon^2)$ correction is a constant in $k$-space
(= self-energy, not affecting pairwise eigenvalues).  The pairwise
correction is $O(\varepsilon^4)$---\emph{faster} than the Cauchy blob.
Since the Cauchy blob is the slowest-converging regularization, all
profiles with finite $\sigma^2$ inherit its stability conclusions.
\end{proof}

The Hessian projection $P_m$ and the total correction
$c_m = -(N^2{-}1)/12 + P_m$ (specific to the Cauchy blob
$h_\varepsilon = -\tfrac12\ln(d^2 + 2\varepsilon^2)$) are exact
rational numbers, computed as follows.  Differentiating the Cauchy
blob pair Hessian with respect to $\varepsilon^2$ at $\varepsilon=0$
gives the per-pair correction:
\begin{equation}\label{eq:blob_pair_correction}
  \delta H_{jj,xx} = \frac{2}{d_{jk}^6}\bigl((\Delta y)^2 - 3(\Delta x)^2\bigr),
  \;\;\;
  \delta H_{jk,xx} = -\delta H_{jj,xx},
\end{equation}
with analogous expressions for $yy$ and $xy$ (see proof of
Proposition~\ref{prop:blob}).  Here $(\Delta x, \Delta y, d_{jk})$
are the pair separation components.  The critical radial Fourier
mode is $v_k \propto z_k\cos(2\pi mk/N)$ with $m = \lfloor N/2\rfloor$.
The projection
\begin{equation}\label{eq:Pm_formula}
  P_m = v_m^\top\,\delta\nabla^2\!H\,v_m
\end{equation}
is a finite quadratic form: a sum over $N(N{-}1)$ pair
contributions, each involving trigonometric values at $N$-th roots
of unity.  Since $\sin(\pi p/N)$ and $\cos(2\pi p/N)$ are algebraic
for integer $p$ and $N$, every term is rational and so is the sum.
Explicit values (verified to machine precision in
\texttt{blob\_correction.compute\_Pm(N)} and notebook
\texttt{05\_Blob\_Correction.ipynb}):
\begin{center}

\midrule
$-(N^2{-}1)/12$ & $-\frac23$ & $-\frac54$ & $-2$ &
  $-\frac{35}{12}$ & $-4$ & $-\frac{21}{4}$ \\[3pt]
$P_m$ & $-\frac13$ & $+\frac14$ & $+1$ & $+4$ & $+8$ &
  $+\frac{65}{4}$ \\[3pt]

\noindent
The Hessian projection $P_m$ changes sign between $N = 3$
($P_3 = -1/3$) and $N = 4$ ($P_4 = +1/4$): for the triangle, the blob
softening reduces the nearest-neighbor curvature (all pairs are at the
same short distance $d = \sqrt{3}$), while for the square and larger
polygons, the long-range pairs increasingly dominate the projection
and $P_m > 0$.  Despite this sign change,
$c_m = -1$ universally for $N \le 5$ because the Lagrange multiplier
shift $-(N^2{-}1)/12$ dominates $P_m$ in this range.  For $N \ge 6$,
$P_m$ grows faster and overwhelms the shift.

For blob profiles with finite second moment $\sigma^2$ (Gaussian,
compact support), the $O(\varepsilon^2)$ correction is confined to
the self-energy and does not affect pairwise eigenvalues
(Section~\ref{sec:blob}, profile independence argument).  The
pairwise correction is $O(\varepsilon^4)$ with a profile-dependent
coefficient.  Since the $O(\varepsilon^2)$ corrections from the Cauchy
blob table above preserve stability for $N \le 7$, and all
finite-$\sigma^2$ blobs have \emph{smaller} corrections at any given
$\varepsilon$, the stability conclusions hold \emph{a fortiori} for
all smooth profiles.

Three consequences follow:

\emph{(i) Stability preserved.}  For $N \le 5$:
$\lambda_0 \ge 1$ and $c = -1$, so
$\lambda(\varepsilon) = \lambda_0 - \varepsilon^2 > 0$
for $\varepsilon < 1$.
For $N = 6$: $c > 0$, so $\lambda(\varepsilon) > \lambda_0$
---the blob \emph{enhances} stability.

\emph{(ii) Marginal mode resolved.}  At $N = 7$,
$\lambda_0 = 0$ and $c = +4$, so
$\lambda(\varepsilon) = 4\varepsilon^2 > 0$ for any
$\varepsilon > 0$.  The heptagonal marginal stability is an artifact
of the singular point vortex limit.

\emph{(iii) Instability partially compensated.}  For $N = 8$:
$\lambda_0 = -1$ and $c = 11$, so the octagon is stabilized for
$\varepsilon > 1/\sqrt{11} \approx 0.30$.

\subsubsection{Full infinite-dimensional convergence}

Proposition~\ref{prop:blob} addresses center-position perturbations of
$N$ smooth blobs.  The full vorticity perturbation space is
infinite-dimensional, including blob \emph{shape} deformations.  A
complete stability analysis requires the Hessian of the 2D Euler energy
$E[\omega] = \tfrac12\iint\omega(x)G(x,y)\omega(y)\,\dd x\,\dd y$ at
the blob configuration
$\omega_\varepsilon = \sum_k \kappa\,\phi_\varepsilon(\cdot - z_k)$,
decomposed into three sectors:

\emph{(a) Center translations.}  The Hessian restricted to the
$2N$-dimensional subspace of center displacements is \emph{exactly}
the blob Hessian of Proposition~\ref{prop:blob}.  Convergence to the
point vortex Hessian at $O(\varepsilon^\alpha)$ follows from the
convolution estimates.

\emph{(b) Shape perturbations.}  Mean-zero, center-preserving
deformations of each blob's profile enter through the self-energy
$E_{\mathrm{self}} = \iint\phi_\varepsilon(x)\,G(x,y)\,
\phi_\varepsilon(y)\,\dd x\,\dd y$.  The second variation at
mean-zero $\delta\omega$ is, by Plancherel and the Fourier
representation $\hat G(\xi) = 1/(4\pi^2|\xi|^2)$ of the 2D
log-Green's function:
\begin{equation}\label{eq:self-energy-Hminus1}
  \delta^2 E_{\mathrm{self}}
  = \iint\delta\omega(x)\,G(x,y)\,\delta\omega(y)\,\dd x\,\dd y
  = \int_{\mathbb{R}^2}\frac{|\hat{\delta\omega}(\xi)|^2}%
    {4\pi^2|\xi|^2}\,\dd\xi
  = \frac{1}{4\pi^2}\,\|\delta\omega\|_{H^{-1}}^2.
\end{equation}
Since $\hat{\delta\omega}(0) = \int\delta\omega = 0$ (mean-zero),
the integrand is zero only on a set of measure zero, so
$\delta^2 E_{\mathrm{self}} > 0$ for all non-zero mean-zero
$\delta\omega$: the self-energy Hessian is positive definite
in $H^{-1}$.  The rescaling $\phi_\varepsilon(x)
= \varepsilon^{-2}\phi(x/\varepsilon)$ gives
$\hat\phi_\varepsilon(\xi) = \hat\phi(\varepsilon\xi)$, so
\[
  E_{\mathrm{self}}[\phi_\varepsilon]
  = \int\frac{|\hat\phi(\varepsilon\xi)|^2}{4\pi^2|\xi|^2}\,\dd\xi
  = \varepsilon^{-2}\int\frac{|\hat\phi(\eta)|^2}{4\pi^2|\eta|^2}\,\dd\eta
  = \frac{c}{\varepsilon^2},
\]
and the smallest shape eigenvalue scales as $c/\varepsilon^2$.
Shape modes are \emph{unconditionally stable}.

\emph{(c) Center-shape coupling.}  The cross terms involve overlap
integrals
\[
  C_{jk} = \iint \nabla\phi_\varepsilon(x{-}z_j)\cdot
  \nabla G(x,y)\,\delta\psi(y{-}z_k)\,\dd x\,\dd y
\]
between non-adjacent blobs $j \ne k$ ($|z_j - z_k| \ge d_{\min}
\gg \varepsilon$), where $\delta\psi$ is a mean-zero,
center-preserving shape perturbation.  Since both the monopole
$(\int\delta\psi = 0)$ and dipole $(\int y\,\delta\psi = 0)$
moments of $\delta\psi$ vanish, the leading term in the multipole
expansion of $G(x,y)$ around $z_k$ is the quadrupole:
\[
  G(x,y) = G(x,z_k) + \nabla_y G(x,z_k)\cdot(y{-}z_k)
            + O(|y{-}z_k|^2/d^2),\quad d = |x{-}z_k|.
\]
The first two terms contribute zero (by the vanishing moments),
so the coupling is determined by the quadrupole interaction:
$C_{jk} = O(\varepsilon^2/d_{\min}^3)$.
(Numerically verified: $C_{jk} \approx 4\varepsilon^2/d_{\min}^3$
for the Cauchy blob at $d_{\min} \ge 3\varepsilon$.)
This is \emph{algebraically small}---the correct bound for the
Cauchy blob, which has algebraic not exponential spatial
decay.  The ratio of coupling to shape eigenvalue
is $O(\varepsilon^4/d_{\min}^3) \to 0$ as $\varepsilon\to 0$
with $d_{\min}$ fixed, which is the only bound needed below.

\begin{proposition}[Inertia preservation]
\label{prop:inertia}
For sufficiently small $\varepsilon > 0$, the full Hessian of
$E[\omega_\varepsilon] + \mu_L\,L[\omega_\varepsilon]$ on the space of
$\mathbb{Z}_N$-symmetric vorticity perturbations has the same inertia
(number of positive, zero, and negative eigenvalues along the constrained
directions) as the point vortex Lagrangian Hessian of
Theorem~\ref{thm:constrained-min}.  The point vortex model is therefore
the \emph{exact} limit of smooth vorticity dynamics for stability
analysis: no eigenvalue is lost, no sign is changed, and no new
instability arises from the infinite-dimensional shape degrees of
freedom.  The stability boundary $N \le 7$ is not an approximation of
the smooth result---it \emph{is} the smooth result in the
$\varepsilon \to 0$ limit.
\end{proposition}

\begin{proof}
Write the constrained full Hessian in block form:
\[
  \mathcal{H} = \begin{pmatrix} H_{cc} & H_{cs} \\ H_{cs}^\top & H_{ss} \end{pmatrix},
\]
where the subscripts $c$ and $s$ denote the center and shape sectors.

\emph{Shape sector.}  By~\eqref{eq:self-energy-Hminus1},
$H_{ss} \ge (c/\varepsilon^2)\,\mathrm{Id}_{H^{-1}}$ with $c > 0$.
All shape eigenvalues are bounded below by $c/\varepsilon^2 > 0$.

\emph{Center sector.}  By~(a) and the convolution estimates of
Proposition~\ref{prop:blob}, the center-sector eigenvalues satisfy
$\lambda_c^{(k)}(\varepsilon) = \lambda_k^{\mathrm{Hav}} + O(\varepsilon^2)$,
where $\lambda_k^{\mathrm{Hav}}$ are the Havelock eigenvalues.

\emph{Coupling bound.}  From~(c), the off-diagonal block satisfies
$\|H_{cs}\|_{\mathrm{op}} = O(\varepsilon^2/d_{\min}^3)$.
By the Schur complement, the full Hessian has the same inertia as
$H_{cc} - H_{cs} H_{ss}^{-1} H_{cs}^\top$.  The correction term is
\[
  \|H_{cs} H_{ss}^{-1} H_{cs}^\top\|_{\mathrm{op}}
  \le \|H_{cs}\|_{\mathrm{op}}^2\,\|H_{ss}^{-1}\|_{\mathrm{op}}
  = O\!\left(\frac{\varepsilon^4}{d_{\min}^6}\right)\cdot O(\varepsilon^2)
  = O\!\left(\frac{\varepsilon^6}{d_{\min}^6}\right),
\]
which goes to zero as $\varepsilon\to 0$ with $d_{\min}$ fixed.
Hence, for sufficiently small $\varepsilon$, the Schur complement
eigenvalues are within $O(\varepsilon^6/d_{\min}^6)$ of the
center-sector eigenvalues $H_{cc}$, and in particular preserve
their signs.  Since $\|H_{cc} - H_{cc}^{\mathrm{Hav}}\|_{\mathrm{op}}
= O(\varepsilon^2)$ (from the blob correction), for small enough
$\varepsilon$ the Schur-complement eigenvalues have the same signs
as the Havelock eigenvalues.  Therefore the full Hessian inertia
equals the Havelock inertia---which is the point vortex Lagrangian
inertia of Theorem~\ref{thm:constrained-min}.
\end{proof}

Proposition~\ref{prop:inertia} extends to \emph{general}
$\mathbb{Z}_N$-symmetric vorticity $\omega$ concentrated in
$N$ patches (not necessarily a prescribed blob profile).
Decompose $\omega = \sum_k \omega_k$ via Voronoi cells,
split each perturbation into center-translation and shape
components $\delta\omega_k = \delta\omega_k^{(c)} + \delta\omega_k^{(s)}$
with $\delta\omega_k^{(c)} = -\delta\bar z_k \cdot \nabla\omega_k$.
Integration by parts gives
$\int x\,\delta\omega_k^{(s)} = 0$ (dipole cancellation),
so the center-shape coupling is quadrupole:
$|C_{jk}| = O(\varepsilon^2/d_{\min}^2)$ for compactly supported
profiles, $O(\varepsilon^2/d_{\min}^3)$ for the Cauchy blob.
The self-energy Hessian is positive definite for any mean-zero
profile, and $\mathbb{Z}_N$ symmetry forces congruent patches.
The phrase
\emph{sufficiently concentrated} is made quantitative as follows.
The operator norm bound~\eqref{eq:blob-correction-op-norm} is
derived for the Cauchy blob, which has algebraic ($|x|^{-3}$) rather
than exponential spatial decay and is therefore the
slowest-converging regularization among profiles with finite
second moment; profiles with faster decay (Gaussian, compact support)
satisfy the same bound \emph{a fortiori}, with a smaller implicit
constant.  The Cauchy-blob bound gives
$\|\nabla^2\!H_\varepsilon - \nabla^2\!H_0\|_{\mathrm{op}}
\le 6(2N{-}3)\,\varepsilon^2/d_{\min}^4$.
Sign preservation of the Havelock eigenvalues requires this to be
less than $\lambda_{\min}(N)$ (the smallest positive Havelock eigenvalue
at the configuration), yielding the explicit concentration condition
\begin{equation}\label{eq:concentration-bound}
  \frac{\varepsilon}{d_{\min}} < \delta(N)
  := \sqrt{\frac{\lambda_{\min}(N)}{6(2N-3)}},
\end{equation}
where $d_{\min} = 2\sin(\pi/N)$ is the nearest-neighbor separation
(unit ring).  Numerically:
\begin{center}
\begin{tabular}{lcccc}
\toprule
$N$ & 4 & 5 & 6 & 7 \\
\midrule
$\lambda_{\min}(N)$ & $1$ & $1$ & $\frac{1}{2}$ & $0$ \\[3pt]
$\delta(N)$ & $0.18$ & $0.15$ & $0.096$ & $\text{---}$ \\
\bottomrule
\end{tabular}
\end{center}
\noindent
The $\delta(N)$ values are set by the center-sector bound, not
the coupling.  At $\varepsilon/d_{\min} = \delta(N)/\sqrt{2}$,
the center-sector eigenvalue shift is at most $\lambda_{\min}/2$
(from~\eqref{eq:concentration-bound}), leaving each center
eigenvalue $\ge \lambda_{\min}/2$.  The Schur correction at this
$\varepsilon$ is $O(\delta^6/8) = O(10^{-6})$, which is smaller
than $\lambda_{\min}/2$ by a factor of at least $10^{-5}$---so
the Schur term cannot reverse any eigenvalue sign, and the full
inertia preservation holds with the explicit bound
$\varepsilon/d_{\min} < \delta(N)/\sqrt{2}$.

\begin{corollary}[Inertia preservation for $N \le 6$]
\label{cor:inertia-N6}
For $N \le 6$ and any $\mathbb{Z}_N$-symmetric vorticity distribution
with patch radius $\varepsilon/d_{\min} < \delta(N)$
(equation~\eqref{eq:concentration-bound}), the full Hessian inertia
equals the Havelock inertia.  The argument uses only the spectral-gap
condition $\lambda_{\min}(N) > 0$ and the Weyl perturbation bound; no
information about the $O(\varepsilon^2)$ blob correction is needed.
\end{corollary}

\begin{corollary}[Marginal-mode resolution for $N = 7$]
\label{cor:inertia-N7}
For $N = 7$, the concentration bound~\eqref{eq:concentration-bound}
does not apply ($\lambda_{\min} = 0$, so $\delta(7) = 0$).  A
separate, logically distinct argument shows
$\lambda_{\min}(\varepsilon) > 0$ for all $\varepsilon \in (0, d_{\min})$
via two regions:
\begin{itemize}
  \item \emph{Region~1} ($\varepsilon < \varepsilon_0
    = 2/\sqrt{|c_7'|}$, where $|c_7'| \le 17$ is the bounded
    $O(\varepsilon^4)$ coefficient):
    $\lambda \ge \varepsilon^2(4 - |c_7'|\varepsilon^2) > 0$,
    since the blob correction $c_7 = +4 > 0$
    (Proposition~\ref{prop:blob}) dominates.
  \item \emph{Region~2} ($\varepsilon \in [\varepsilon_0, d_{\min})$):
    interval arithmetic on a $2{,}000$-point grid with
    mean-value-theorem bounds gives
    $\lambda \ge 0.34 > 0$ on every subinterval.
\end{itemize}
\end{corollary}

\noindent
The two corollaries rest on different mechanisms:
Corollary~\ref{cor:inertia-N6} on a spectral gap,
Corollary~\ref{cor:inertia-N7} on a blob correction.  Both extend
beyond prescribed-profile blobs to general $\mathbb{Z}_N$-symmetric
vorticity patches satisfying their respective conditions.

\subsubsection{The Rossby bridge (Saturn regime)}
\label{sec:rossby-bridge}

For continuous jet meanders ($\beta R^3/\kappa \gg 1$), the bridge
is not vortex concentration but \emph{Rossby wave quantization}.
The QGPV linearized about a zonal jet $U(y)$ gives the Rayleigh--Kuo
equation:
\[
  (U - c)\bigl(\hat\psi'' - k^2\hat\psi\bigr)
  + (\beta - U'')\hat\psi = 0.
\]
For a stationary wave ($c = 0$) at the jet center ($y = 0$) of a
Gaussian jet with $U''(0) = -U_{\max}/\sigma^2$:
\[
  \hat\psi'' + \Bigl[\frac{\beta_{\mathrm{eff}}}{U_{\max}}
  - \frac{n^2}{R^2}\Bigr]\hat\psi = 0,
  \qquad
  \beta_{\mathrm{eff}} = \beta + \frac{U_{\max}}{\sigma^2},
\]
where the effective PV gradient $\beta_{\mathrm{eff}}$ includes the
jet curvature (which dominates $\beta$ by a factor $\sim\!13$ for
Saturn).  The stationary azimuthal wavenumber is
\begin{equation}\label{eq:rossby-bridge}
  n^* = R_{\mathrm{S}}\cos\varphi
  \,\sqrt{\beta_{\mathrm{eff}}/U_{\max}},
\end{equation}
where $R_{\mathrm{S}}\cos\varphi$ is the ring radius at latitude
$\varphi$ ($\approx 1.31 \times 10^7\;\mathrm{m}$ for Saturn at
$76^\circ$\,N).  The nearest integer $n = \mathrm{round}(n^*)$ is
the selected polygon wavenumber.  For Saturn:
$\beta_{\mathrm{eff}}/U_{\max} \approx 1.72 \times
10^{-13}\;\mathrm{m}^{-2}$, giving $n^* \approx 5.4$ from the
leading-order dispersion relation.  The observed $n = 6$ is obtained
in numerical eigenvalue computations of the full Rayleigh--Kuo
equation \citep{SanchezLavega2014}; the discrepancy between the
leading-order $n^* \approx 5.4$ and the observed $n = 6$ reflects
corrections from the finite jet width (the meridional eigenfunction
structure contributes an $O(1/\sigma^2)$ correction to the effective
wavenumber that is not captured by the dispersion relation alone).

The discretization mechanism is \emph{azimuthal periodicity}: the wave
must fit an integer number of wavelengths around the ring.

The connection to the point vortex energy landscape proceeds through
the \emph{critical layer} of the saturated wave.  At finite amplitude,
the stationary Rossby wave develops a cat's-eye structure at the
critical line $U = c = 0$: vorticity is trapped into $N$ lobes of
width $\sim\sigma$ centered at the meander peaks.  As the jet sharpens
($\sigma/R \to 0$ at fixed wave amplitude), these lobes concentrate
into $N$ isolated patches near the $N$-gon vertices.

\begin{remark}[Cat's-eye Hessian convergence]
\label{rem:catseye}
The cat's-eye vorticity decomposes as
$\omega = \omega_{\mathrm{lobe}} + \omega_{\mathrm{braid}} +
\omega_{\mathrm{bg}}$, where the lobes are the trapped fluid inside
the separatrices, the braids are thin filaments along the separatrix
itself, and the background is the passing fluid outside the cat's-eye.

Two conditions are required for Proposition~\ref{prop:inertia} to
apply:
\begin{itemize}
  \item[(i)] \textbf{Lobe concentration.}  The lobes satisfy the
    concentration condition~\eqref{eq:concentration-bound} with
    $\varepsilon \sim \sigma$ and the centroids converge to the
    $N$-gon vertices.  This is guaranteed by $\mathbb{Z}_N$ symmetry.
  \item[(ii)] \textbf{Braid smallness.}  The absolute braid circulation
    $\Gamma_{\mathrm{braid}}$ is negligible compared to the total
    lobe circulation~$\Gamma_{\mathrm{lobe}}$.
\end{itemize}

Both conditions hold for the Stuart vortex
$\Psi = -\sigma^2\ln(\cosh(y/\sigma) - \eta\cos(n\theta))$
($\eta \in (0,1)$; $\eta \to 1$ is point-vortex concentration).

\emph{Proof of lobe dominance.}  The Stuart vortex satisfies
$\omega = C/F^2$ with $F = \cosh(y/\sigma) - \eta\cos(n\theta)$.
Near the elliptic center ($\theta = 0$, $y = 0$):
$F \approx (1{-}\eta) + (y/\sigma)^2/2 + n^2\eta\,\theta^2/2$,
so
\[
  \Gamma_{\mathrm{lobe}} = C\iint_{\mathrm{lobe}} \frac{dA}{F^2}
  \ge C\int_0^\infty \frac{2\pi r\,dr}{[(1{-}\eta)+r^2/2]^2}
  = \frac{2\pi C}{1{-}\eta}.
\]
Outside the separatrix, $F \ge 1{+}\eta \ge 1$, so
$\Gamma_{\mathrm{non\text{-}lobe}} = C\!\iint_{\mathrm{outside}} F^{-2}\,dA
\le C\,A_{\mathrm{period}}$, which is bounded independently of~$\eta$.
Therefore
\begin{equation}\label{eq:lobe-fraction}
  \frac{\Gamma_{\mathrm{non\text{-}lobe}}}{\Gamma_{\mathrm{total}}}
  \;\le\; \frac{A_{\mathrm{period}}\,(1{-}\eta)}{2\pi + A_{\mathrm{period}}\,(1{-}\eta)}
  \;=\; O(1{-}\eta).
\end{equation}
The lobe fraction $\Gamma_{\mathrm{lobe}}/\Gamma_{\mathrm{total}}
= 1 - O(1{-}\eta)$ as $\eta \to 1$.  Numerically: at $\eta = 0.99$,
lobes capture $>99\%$ of the total circulation.

Combining with Proposition~\ref{prop:inertia}: the cat's-eye
satisfies the blob bridge conditions for
$\sigma/d_{\min} < \delta(N)/\!\sqrt{2}$ and
$1 - \eta < c\,\delta(N)^2$ (the braid perturbation
is $O(1{-}\eta)$ of the lobe system, negligible at the same
threshold).  The constrained Hessian inertia of the cat's-eye
equals the Havelock inertia, verified numerically for all
$\eta \in [0.3, 0.99]$ and confirmed analytically
by~\eqref{eq:lobe-fraction} in the limit $\eta \to 1$.
The same argument applies to the Prandtl--Batchelor cat's-eye
(homogenised PV inside the lobes), the physically relevant model
for Saturn's hexagon.

\section{Finite Rossby deformation radius: the $K_0$ interaction}
\label{sec:k0-interaction}
\label{sec:deformation}

The $K_0$ interaction provides a second example of the RG flow
toward the logarithmic fixed point
(Proposition~Paper~I, Proposition~2.1): the exponential tail
$K_0(r/R_d) + \ln r = O(e^{-r/R_d})$ is an irrelevant perturbation
(faster than any power law), and the limit $R_d \to \infty$ is the
flow to the fixed point.

In quasi-geostrophic dynamics with finite Rossby deformation radius
$R_d$, the Green's function is $K_0(r/R_d)$ (modified Bessel function),
which recovers $-\ln r$ for $r \ll R_d$.  The interaction $h(r) =
K_0(r/R_d)$ satisfies $h'(r) = -K_1(r/R_d)/R_d < 0$ and
$(rh'(r))' = rK_0(r/R_d)/R_d^2 > 0$ (class~(iii)).  Numerical
computation of the constrained Lagrangian Hessian as a function of
$\eta = R/R_d$ (polygon ring radius over deformation radius) reveals
an $N$-dependent \emph{polygon-suppression transition}:

\begin{center}
\begin{tabular}{lccl}
\toprule
$N$ & Barotropic $\lambda_{\min}$ & Critical $R/R_d$ & Status \\
\midrule
$\le 5$ & $\ge 1$ & none & Stable for all $R/R_d$ \\
$6$ & $1/2$ & $\approx 1.29$ & Suppressed for $R/R_d > 1.29$ \\
$7$ & $0$ & $\approx 0$ & Unstable for any finite $R/R_d$ \\
$\ge 8$ & $< 0$ & N/A & Unstable (barotropic) \\
\bottomrule
\end{tabular}
\end{center}

\noindent
Polygons with $N \le 5$ are robust against stratification effects.
The hexagon ($N = 6$) is the first polygon sensitive to $R_d$: it
requires $R/R_d < 1.29$.  The heptagon ($N = 7$), neutrally stable
in the barotropic limit, is destabilized by any finite
$R/R_d$---providing an additional mechanism for its rarity beyond the
quartic confinement.

For Saturn, the barotropic deformation radius
($\sim\!2000$--$3000\,$km) gives $R/R_d \approx 8$--$12$ at the
hexagonal jet, placing it above the critical ratio.  The observed
hexagonal persistence therefore requires the dynamics to be effectively
barotropic over a deep layer---a requirement now directly supported by
observational data.  Cassini Grand Finale gravity measurements constrain
Saturn's differential rotation to a depth of $8{,}000$--$9{,}000$\,km
\citep{Iess2019}, implying a coherent barotropic column of order
$10^4$\,km, far exceeding the hexagon's vertical scale.  Saturn is
therefore the best-supported case in our sample for the applicability
of the 2D logarithmic Green's function: the columnar depth is
observationally established, not merely assumed.

\paragraph{Observable prediction: hexagon suppression signatures.}
The critical ratio $R/R_d \approx 1.29$ for $N = 6$ yields a
testable prediction.  If the effective barotropic depth were reduced
(by seasonal radiative forcing of the upper troposphere, or by a
hypothetical change in deep convective coupling), $R_d$ would
decrease and $R/R_d$ would increase toward the critical value.
Approaching the threshold, the framework predicts three observable
signatures: (1)~the minimum Havelock eigenvalue
$\lambda_{\min}(\eta) \to 0$, reducing the hexagonal mode's
restoring frequency and producing a \emph{longer oscillation period}
for meridional meander perturbations;
(2)~the hexagonal amplitude $\varepsilon_{\mathrm{obs}}$ would
\emph{grow}, since the restoring force weakens while convective
forcing remains constant; and
(3)~at $R/R_d > 1.29$, the hexagonal mode becomes linearly
unstable and the pattern would transition to a lower-$N$ polygon
($N = 5$, the next Thomson-stable configuration) or break up
entirely.
Saturn's hexagon has been observed continuously since Voyager~1
(1980--81); Cassini monitored it through northern spring equinox
(2009) to summer solstice (2017) with no sign of weakening
\citep{Antunano2015, SanchezLavega2014}.  This four-decade
persistence across half a Saturnian year places an empirical lower
bound on the effective barotropic depth: seasonal radiative forcing
does not reduce $R_d$ enough to approach the critical ratio, consistent
with the deep (${\sim}\,10^4\;\mathrm{km}$) barotropic column
implied by Cassini gravity.

\paragraph{Why the leading $K_0$ correction does not shift stability.}
The small-$r$ expansion $K_0(r/R_d) = -\ln(r/R_d) - \gamma_E
+ r^2/(4R_d^2) + O(r^4)$ shows the leading correction to the
logarithm is $g(r) = r^2/(4R_d^2)$---a perturbation of
RG exponent $\alpha = 2$.  By the isotropic-marginality result
(Paper~I, Proposition~\ref{I-prop:alpha-2-marginal}), the pair
Hessian anisotropy $g'' - g'/r = \alpha(\alpha{-}2)\,r^{\alpha-2}$
vanishes at $\alpha = 2$: the Hessian perturbation is
purely isotropic ($\propto I$) and is exactly compensated by
the rotation-frequency shift $\delta\Omega$.
Consequently $\delta\lambda_m = 0$ for all binding modes---the
$r^2/(4R_d^2)$ term shifts the precession rate of the polygon but
does \emph{not} shift its stability eigenvalues.
The first correction to stability from $K_0$ is $O(r^4/R_d^4)$
($\alpha = 4$, which is \emph{not} isotropic-marginal), and this
is numerically small for $R/R_d \ll 1$.
This explains why the barotropic ($R_d \to \infty$) stability
predictions remain accurate even at moderate $R/R_d$: the
leading deformation-radius correction is stability-neutral.

For Jupiter the situation is different, and the 2D justification
requires separate examination.  Juno gravity measurements
\citep{Kaspi2018} constrain Jupiter's differential rotation to
$\sim\!3{,}000$\,km depth (${\approx}\,4\%$ of $R_J$), far shallower
than Saturn's.  Jupiter's polar cyclones therefore do \emph{not} rest
on deep barotropic Taylor columns; the relevant 2D dynamics is that of
a shallow weather layer above a near-solid-body rotating deep interior.
Two limiting cases bracket the problem:

\textit{Deep weather-layer limit ($H_{\rm eff} \sim 3{,}000$\,km).}
Taking the full differential-rotation depth as the effective layer
and using the equivalent-barotropic deformation radius
$R_d = \sqrt{g\,H_{\rm eff}}/f \approx \sqrt{25 \times 3\times10^6}\,/\,
(2\Omega_J\sin 84^\circ) \approx 25{,}000$\,km, we obtain
$R/R_d \approx 0.3 \ll 1$.  In this limit
$K_0(r/R_d) \to -\ln r + \mathrm{const}$, so the logarithmic
Green's function applies---though the physical justification is
a shallow equivalent-barotropic layer rather than a deep Taylor column.

\textit{Tropospheric limit ($H_{\rm eff} \sim 30$--$100$\,km).}
If the active polar-cyclone layer is shallower, reduced gravity
$g' \sim 0.1$--$1$\,m\,s$^{-2}$ gives
$R_d \sim 2{,}000$--$6{,}000$\,km and $R/R_d \sim 1$--$4$.
The $N = 8$ ring lies in the finite-$R_d$ regime.  Computing the
$\kappa_0$-shifted constrained Hessian eigenvalue for the $K_0$
interaction with $\kappa_0/\kappa = 0.7$ (the Juno central estimate):
\begin{center}
\begin{tabular}{cccc}
\toprule
$N$ & $\kappa_0/\kappa$ & $(R/R_d)_{\mathrm{crit}}$ & Tropospheric stable? \\
\midrule
$6$ & $0.7$ & $17.2$ & yes ($\gg 4$) \\
$7$ & $0.7$ & $4.2$  & marginal \\
$8$ & $0.7$ & $\mathbf{0.79}$ & \textbf{no} ($< 1$) \\
$9$ & $0.7$ & $0.10$ & no \\
\bottomrule
\end{tabular}
\end{center}
Here $(R/R_d)_{\mathrm{crit}}$ is the ratio at which the minimum
constrained eigenvalue crosses zero: the ring is stable for
$R/R_d < (R/R_d)_{\mathrm{crit}}$ and unstable above.  For $N = 8$
the critical ratio is $(R/R_d)_{\mathrm{crit}} \approx 0.79$, which
lies \emph{below} the tropospheric range $R/R_d \in [1, 4]$.
\textbf{The $N = 8$ prediction therefore requires the deep
weather-layer limit} ($R/R_d \lesssim 0.8$, i.e.\
$R_d \gtrsim 1.3\,R \approx 9{,}700$\,km); it does not survive
into the tropospheric regime.

The effective layer depth is not directly constrained by current Juno
data, making Jupiter the \emph{least} observationally supported case
for the 2D assumption, in contrast to Saturn.  A direct measurement
of the depth of the polar cyclone roots---analogous to the gravity
constraints on the jets---would resolve this ambiguity.  The
computation above sets a quantitative target: $R_d > 9{,}700$\,km
is necessary for $N = 8$ stability under the $K_0$ interaction.

The critical ratio is defined implicitly by the equation
$\bar\lambda(\eta_{\mathrm{crit}}, N) = 0$, where the dimensionless
eigenvalue $\bar\lambda(\eta, N) = R^2 \cdot \lambda_{\min}(\eta)$
interpolates smoothly from the Havelock value
$\bar\lambda(0, N) = (N{-}1) - \lfloor N/2\rfloor
(N{-}\lfloor N/2\rfloor)/2$ to zero.

The Lagrange multiplier is determined by the stationarity condition
$\partial H/\partial R + \mu_L\,\partial L/\partial R = 0$.  For
$H = -\sum_{j<k}K_0(d_{jk}/R_d)$ with $d_{jk} = 2R\sin(\pi p/N)$
and $L = NR^2$, the radial derivative of each pair contributes
$-K_0'(d_{jk}/R_d)\cdot 2\sin(\pi p/N)/R_d
= K_1(d_{jk}/R_d)\cdot 2\sin(\pi p/N)/R_d$
(using $K_0' = -K_1$).  Projecting onto the constraint gradient
$\partial L/\partial R = 2NR$ and solving gives
\begin{equation}\label{eq:mu-k0}
  -2\mu_L = \frac{\eta}{R}\sum_{p=1}^{N-1}
  \frac{(1-\cos\theta_p)\,K_1(2\eta\sin(\pi p/N))}
       {2\sin(\pi p/N)},
  \quad \theta_p = 2\pi p/N,
\end{equation}
where $\eta = R/R_d$ and the factor $(1{-}\cos\theta_p)$ arises from
the Fourier decomposition of the pairwise Hessian along the radial
mode.  As $\eta \to 0$: $K_1(x) \sim 1/x$ for small $x$, so each
summand reduces to
$(1{-}\cos\theta_p)/(4\eta\sin^2(\pi p/N)) \cdot (\eta/R)$
$= (1{-}\cos\theta_p)/(4R\sin^2(\pi p/N))$.  Summing over $p$ using
the Havelock identity $\sum_p (1{-}\cos\theta_p)/(4\sin^2(\pi p/N))
= (N{-}1)/2$ recovers the flat-plane value $-2\mu_L = (N{-}1)/(2R^2)$.  For $N = 6$, the critical mode $m = 3$
has zero Havelock weight at $p = 2, 4$ (since
$\cos(2\pi \cdot 2 \cdot 3/6) = 1$), so the critical eigenvalue
depends on $K_0$ and $K_1$ at only \emph{two} distinct arguments,
$\eta$ and $2\eta$, rather than three.

No closed-form analog of the Havelock identity is known for $K_0$.
The orthogonality mechanism that collapses the logarithmic sum to
$m(N{-}m)/2$ relies on the identity $\sum_p e^{2\pi i mp/N}\ln
|2\sin(\pi p/N)| = 0$, which has no counterpart for $K_0$:
the Bessel values $K_0(2\eta\sin(\pi p/N))$ are transcendental at
nonzero algebraic arguments by Siegel's theorem on $E$-functions,
but this does not preclude the full weighted sum from having a
closed form.  Whether $\sum_p (\cos\theta_{pm})
K_0(2\eta\sin(\pi p/N))$ admits a closed form for any $N$ and $m$
is an open question; no such form has been found, and all stability
computations for $K_0$ interactions are carried out numerically.
A perturbative expansion
$K_0(x) \approx -\ln x + O(x^2)$ gives $\eta_{\mathrm{crit}} \approx
0.9$ for $N = 6$---30\% below the exact value---because the critical
ratio lies in the non-perturbative regime where $K_0$ departs
significantly from $-\ln$.  The transition occurs precisely where the
exponential decay of $K_0$ (destroying long-range correlations between
non-adjacent vortices) overtakes the logarithmic regime, making the
suppression inherently non-perturbative.  The implicit equation is
nevertheless well-posed, and the critical $\eta$ is uniquely determined
by bisection for each $N$.

\medskip\noindent\textbf{The two phase boundaries.}
For $K_0$, two distinct boundaries exist in the $(N, \eta)$ plane:
(1)~the \emph{spiral sign change} $\bar H''(0) = 0$, where the
positive $A_m$ term from $(rh')' > 0$ overwhelms the negative $B_m$
term; and (2)~the \emph{constrained stability loss}
$\bar\lambda_{\min} = 0$, where the Lagrange multiplier shift
consumes the stability margin.  The constrained boundary
($\eta \approx 1.29$ for $N = 6$) is always \emph{inside} the
spiral boundary ($\eta \approx 3.2$ for $N = 6$) because the
Lagrange shift reduces the margin before $H''$ changes sign.
Physically, the polygon requires long-range correlations between
non-adjacent vortices.  When $\eta \gg 1$ and the $K_0$ interaction
decays exponentially, only nearest-neighbor correlations survive and
the polygon loses its energetic advantage.

For comparison, the Cauchy-blob regularization
$-\tfrac12\ln(d^2 + 2\varepsilon^2)$ has $H'' < 0$ for \emph{all}
$\varepsilon$---the soft-core never flips the spiral sign.  The
Gaussian-screened interaction $-\ln(d)\cdot e^{-d^2/L^2}$
shows an earlier transition ($\eta \approx 0.9$ for $N = 6$) due to
its faster-than-exponential decay.

the largest $N$ for which the \emph{constrained} Lagrangian Hessian

minimum).  This is distinct from the spiral-direction boundary
$H''(0) = 0$, which occurs at larger $\eta$ (e.g.\ $\eta \approx 3.2$
vs.\ $1.29$ for $N = 6$ with $K_0$; see the two phase boundaries above):
\begin{center}

\toprule
Class & Condition & $N_{\mathrm{crit}}$ (constrained) \\
\midrule
(i) & $(rh')' < 0$ & $\ge 7$ (increases with $|(rh')'|$) \\
(ii) & $(rh')' = 0$ (only $-\ln r$) & $= 7$ exactly \\

\begin{proposition}[Min/max duality]\label{prop:duality}
For any pairwise interaction $h(r)$, the constrained energy minimum
of $H = \sum h(d_{jk})$ and the constrained energy maximum of $-H$
have the same stability boundary $N_{\mathrm{crit}}$.
\end{proposition}

\begin{proof}
The Lagrange multiplier for $-H$ is $\mu_L^{-H} = -\mu_L^H$
(from $-\nabla H = (-\mu_L)\nabla L$).  The Lagrangian Hessian
$\mathcal{L}_{-H} = -\nabla^2\!H + 2\mu_L I = -\mathcal{L}_H$.
The eigenvalues are negated; constrained minimum (all $\ge 0$) of $H$
is equivalent to constrained maximum (all $\le 0$) of $-H$.
\end{proof}

For gravity ($h = -1/r$, $h' > 0$), the $N$-gon is a constrained
energy maximum.  The duality gives
$N_{\mathrm{crit}}^{\mathrm{gravity}} =
N_{\mathrm{crit}}^{\mathrm{Coulomb}}$.  Numerical computation of
the Coulomb ($h = +1/r$) constrained Hessian gives
$N_{\mathrm{crit}} = 5$: the minimum eigenvalue is $+0.73$ at $N = 5$
and $-0.02$ at $N = 6$.  This is consistent with Maxwell's (1859)
result that gravitating rings require a central mass for
$N \ge 7$; the one-unit difference ($N_{\mathrm{crit}} = 5$ vs
Maxwell's $6$) arises because our analysis does not include the
central mass (Maxwell's stability includes the stabilizing effect
of the central body).

The Coulomb/gravity $N_{\mathrm{crit}} = 5$ is less than the
logarithmic $N_{\mathrm{crit}} = 7$, confirming the class~(iii)
reduction: the $1/r$ interaction has $(rh')' = 1/r^2 > 0$, placing
it in the destabilizing regime relative to $-\ln r$.

\section{Circulation disorder and stability domain robustness}
\label{sec:disorder-new}

\subsubsection{Resilience to circulation disorder}
\label{sec:disorder}

In both planetary regimes the vortex circulations $\kappa_k$ are not
perfectly equal; convection injects or removes circulation from individual
vortices continuously.  We model this as $\kappa_k = \kappa(1+\eta_k)$
with i.i.d.\ fractional disorders $\eta_k \sim \mathcal{N}(0,\sigma_\eta^2)$.

We work on $\mathbf{H}^2$ rather than the flat plane because the
hyperbolic setting simultaneously tests two effects: circulation
disorder and curvature stabilization.  In the flat limit
$\rho/a \to 0$ the $\mathbf{H}^2$ eigenvalues recover the Havelock
values exactly, so the flat-plane result is contained as a special
case; conversely, the curvature-stabilized configurations ($N \ge 8$
on $\mathbf{H}^2$) have no flat-plane analog and can only be probed
on the curved surface.

On $\mathbf{H}^2$, the disordered Hamiltonian is
\begin{equation}\label{eq:H_kappa}
  H_\kappa = -\sum_{j<k}\kappa_j\kappa_k\ln|z_j - z_k|
  + \tfrac12\sum_j\kappa_j(K_{\mathrm{tot}}-\kappa_j)\ln(a^2 - |z_j|^2),
\end{equation}
where $K_{\mathrm{tot}} = \sum_k\kappa_k$.  At $\eta_k \equiv 0$ this
reduces exactly to the uniform Hamiltonian $H_{\mathrm{hyp}}$.

Holding the ring positions at the uniform equilibrium (exact to $O(\eta)$;
the $O(\eta)$ shift of equilibrium produces only an $O(\eta^2)$ eigenvalue
correction), we compute the minimum Fourier-mode eigenvalue
$\lambda_{\min}(\eta)$ via finite differences.  Monte Carlo over
$n = 1000$ realizations estimates $P(\lambda_{\min} < 0)$ as a function
of $(\sigma_\eta, N, \rho)$.

\paragraph{Physical range of $\sigma_\eta$.}
Juno imagery \citep{Adriani2018,Gavriel2021} shows Jupiter's eight
north-polar cyclones have peak wind speeds ranging from
$\sim\!60$ to $\sim\!100$\,m\,s$^{-1}$ and diameters from
$\sim\!4{,}000$ to $\sim\!7{,}000$\,km.  Taking circulation
$\kappa \propto V_{\max}\,r_c$, the coefficient of variation across
the eight cyclones is $\sigma_\eta \approx 0.15$--$0.25$.  Saturn's
hexagonal jet is more uniform (Cassini wind profiles give
$\sigma_\eta \lesssim 0.05$ for the azimuthal circulation variation).
The physically relevant range is therefore
$\sigma_\eta \in [0.05, 0.25]$.

\medskip\noindent
Key findings (all uncertainties are $\pm 1\sigma$ binomial standard
errors $\sqrt{P(1{-}P)/n}$ with $n = 1000$):
\begin{itemize}
  \item \textbf{Stable configurations are robust.}  For $N = 6$ at
    $\rho = 0.5a$, the instability probability remains
    below $10\%\;(\pm 0.9\%)$ for $\sigma_\eta \le 0.2$, covering
    the full physically relevant range.

  \item \textbf{Curvature-stabilized rings ($N \ge 8$) are similarly
    robust.}  At $N = 8$, $\rho = 2a$ (well inside the stable region
    on $\mathbf{H}^2$), $P < 0.5\;(\pm 1.6\%)$ for
    $\sigma_\eta \le 0.1$.  At $\sigma_\eta = 0.2$ (Jupiter's
    estimated value), $P$ remains below $15\%$.

  \item \textbf{The $N = 7$ marginal ring is sensitive.}  At $N = 7$,
    $\rho = 0.05a$ (flat-limit regime), $P(\lambda_{\min} < 0)$
    increases monotonically with $\sigma_\eta$:
    $P \approx 0\;(\pm 0.3\%)$ at $\sigma_\eta = 0.1$ and $P > 0$ for
    $\sigma_\eta \ge 0.5$.  The marginal mode ($\lambda_0 = 0$ at
    $K = 0$) is the first to be destabilized by disorder, as expected.
    At Jupiter's estimated $\sigma_\eta \approx 0.2$, a hypothetical
    $N = 7$ ring would have $P \approx 0$---the marginal sensitivity
    manifests only at unphysically large disorder.
\end{itemize}

\noindent
Thus the $N_{\mathrm{crit}}$ boundary is a \emph{stability domain
boundary, not a knife-edge}: configurations comfortably inside the
stable region remain stable under the circulation fluctuations
observed at Jupiter ($\sigma_\eta \approx 0.2$) and Saturn
($\sigma_\eta \lesssim 0.05$).

Code: \texttt{circulation\_disorder.instability\_probability(N,~$\rho$,~$\sigma_\eta$)}
and \texttt{disorder\_min\_eigenvalue\_h2(N,~$\rho$,~$\eta$)} in
\texttt{src/planetary\_polygons/extensions/circulation\_disorder.py};
test suite in \texttt{tests/test\_circulation\_disorder.py}.

\subsubsection{The regime criterion}

The dimensionless number $\beta R^3/\kappa$ is the ratio of two
timescales: the Rossby wave period $T_R \sim 1/(\beta R)$ and the
vortex turnover time $T_v \sim R^2/\kappa$.  When
$\beta R^3/\kappa = T_v/T_R \ll 1$, the vortex interactions complete
many orbits before the Rossby wave propagates appreciably, so the
dynamics is vortex-dominated.  When $\beta R^3/\kappa \gg 1$, the
Rossby wave dynamics operates faster than the vortex rearrangement,
so the wave mechanism dominates:
\begin{center}
\begin{tabular}{lcl}
\toprule
Regime & $\beta R^3/\kappa$ & Bridge \\
\midrule
Vortex-dominated & $\ll 1$ & Blob convergence (Proposition~\ref{prop:blob}) \\
Wave-dominated & $\gg 1$ & Rossby stationarity (equation~\eqref{eq:rossby-bridge}) \\
\bottomrule
\end{tabular}
\end{center}

\noindent
\textit{Jupiter's north pole.}
Throughout this regime calculation $\kappa$ denotes the
\emph{Rankine-core} circulation, which integrates only within the
solid-body rotation core and sets the characteristic vortex timescale
$T_v \sim R^2/\kappa$.  The full-cyclone circulation $\kappa_{\mathrm{full}}$
(which includes the outer irrotational skirt) is larger and is the
appropriate quantity for computing orbital periods and eigenfrequencies;
it is used there instead (Section~Paper~III, \S21.3).
The two estimates are numerically distinguished below.

At $84^\circ$N, the planetary vorticity
gradient is $\beta_J = 2\Omega_J\cos(84^\circ)/R_J \approx
5.1\times10^{-13}$\,m$^{-1}$s$^{-1}$ and the cyclone-ring radius
is $R \approx 7{,}470$\,km.  Each Juno-observed circumpolar cyclone has
Rankine core radius $r_c \approx 700$\,km and peak wind $V_{\max}
\approx 80$\,m\,s$^{-1}$ \citep{Adriani2018,Gavriel2021}, giving
$\kappa_{\mathrm{Rank}} = 2\pi r_c V_{\max} \approx 3.5\times10^{8}$\,m$^2$\,s$^{-1}$
(full-cyclone estimate: $r_c \approx 3{,}200$\,km,
$\kappa_{\mathrm{full}} \approx 1.6\times10^9$\,m$^2$\,s$^{-1}$).
Hence
\[
  \frac{\beta R^3}{\kappa}
  = \frac{5.1\times10^{-13}\,(7.47\times10^6)^3}{3.5\times10^8}
  \approx 0.61.
\]
This places Jupiter in the vortex-dominated regime, but only
moderately: $0.61$ is $O(1)$, not $\ll 1$.  Rossby-wave effects
are not negligible at this value, and a decrease in the effective
$\kappa$ (e.g.\ using the Rankine-core rather than full-cyclone
estimate) would push the ratio above unity.  The qualitative
conclusion---discrete cyclones, Thomson stability applies, blob
bridge proves convergence---is robust to this uncertainty because
the vortex-dominated bridge requires only that the cyclones be
identifiable as $N$ distinct vortices (which the Juno imagery
unambiguously confirms), not that $\beta R^3/\kappa$ be
parametrically small.  The ratio enters quantitatively only
through the Rossby-wave correction to the eigenvalues, which is
$O(\beta R^3/\kappa)$ and does not change the sign of
$\lambda_{\min}$ for $N = 8$ (where
$|\lambda_{\min}| = 1 \gg 0.61$).
Using the full-cyclone circulation
$\kappa_{\mathrm{full}} \approx 1.6\times10^9$\,m$^2$\,s$^{-1}$
gives $\beta R^3/\kappa \approx 0.13$, firmly in the
vortex-dominated regime.

\smallskip\noindent
\textit{Saturn's hexagon.}  At $76^\circ$N, $\beta_S =
2\Omega_S\cos(76^\circ)/R_S \approx 1.3\times10^{-12}$\,m$^{-1}$s$^{-1}$
and $R \approx 14{,}600$\,km.  Saturn's polar jet is
well-described by a Gaussian profile $U = U_{\max}\exp(-y^2/2\sigma^2)$;
its effective circulation is $\kappa = U_{\max}\sqrt{2\pi}\,\sigma$.
With $U_{\max} \approx 100$\,m\,s$^{-1}$ and $\sigma \approx 2{,}100$\,km
(Cassini VIMS wind profiles), $\kappa \approx 5.4\times10^{8}$\,m$^2$\,s$^{-1}$,
giving
\[
  \frac{\beta R^3}{\kappa}
  = \frac{1.3\times10^{-12}\,(1.46\times10^7)^3}{5.4\times10^8}
  \approx 7.6.
\]
Continuous jet meander, Rossby stationarity selects $n = 6$, periodicity
provides quantization.

\smallskip\noindent
Both mechanisms require the 2D logarithmic Green's function as the
common mathematical kernel.

\section{Predictions for quantum systems: BEC vortex clusters}
\label{sec:quantum}
\label{sec:predictions}
\label{sec:bec}

In a quasi-2D Bose--Einstein condensate in a circular trap, quantized
vortices interact logarithmically at scales above the healing length
$\xi$.  The circulation is quantized:
$\kappa = q \cdot h/m$ with integer winding number $q$.
The $\kappa_0/\kappa$ ratio of Proposition~\ref{prop:kappa-crit}
becomes the winding number ratio $q_0/q$, yielding sharp integer-valued
predictions:

\begin{center}
\begin{tabular}{lcl}
\toprule
Configuration & $\kappa_{\mathrm{crit}}$ & Prediction \\
\midrule
$N \le 7$ ring & 0 & Stable without central vortex \\
$N = 8$ ring & $1/2$ & Unstable alone; stable with $q_0 \ge 1$ \\
$N = 9$ ring & $1$ & Marginally stable with $q_0 = 1$;
  stable with $q_0 \ge 2$ \\
$N = 10$ ring & $7/4$ & Unstable with $q_0 = 1$;
  stable with $q_0 \ge 2$ \\
\bottomrule
\end{tabular}
\end{center}

\noindent
The $N = 8$ prediction is the sharpest: a ring of $8$ unit-winding
vortices should be unstable alone and stable upon adding a single
central vortex ($q_0 = 1 > \kappa_{\mathrm{crit}} = 1/2$).

The $N = 7$ case offers a distinct signature.  The quartic restoring
force ($\alpha_0 = 45/14$, Section~\ref{sec:n7}) produces an
\emph{amplitude-dependent} oscillation frequency for the $m = 3$ mode.
In the normal form $\mathcal{H} = (\alpha_0/4)\varepsilon^4$, the
nonlinear oscillation frequency is
$\omega_{m=3} = \alpha_0\,\varepsilon^2\,\kappa/(2\pi R^2)$, where
$\varepsilon$ is the dimensionless perturbation amplitude (radial
displacement / ring radius) and $\kappa = h/m_{\mathrm{atom}}$ is the
single-quantum circulation.  For $^{87}$Rb in a circular trap of
radius $R = 50\;\mu\mathrm{m}$:
$\kappa \approx 1.2\times 10^{-7}\;\mathrm{m^2\,s^{-1}}$,
$\kappa/(2\pi R^2) \approx 7.6\;\mathrm{s^{-1}}$, so
$\omega_{m=3} \approx 24\,\varepsilon^2\;\mathrm{s^{-1}}$.  At a
5\% radial displacement ($\varepsilon = 0.05$):
$\omega_{m=3} \approx 0.06\;\mathrm{s^{-1}}$ (period $\sim\!100\;\mathrm{s}$),
resolvable with standard BEC vortex imaging at $\sim\!10\;\mathrm{Hz}$
frame rates.  The key signature distinguishing the $N = 7$
quartic confinement from the $N \le 6$ quadratic regime is that the
$m = 3$ oscillation frequency \emph{vanishes} as $\varepsilon \to 0$
(rather than approaching a finite value), producing an
amplitude-dependent period that is directly measurable.

The healing length $\xi$ provides a natural blob width
$\varepsilon_{\mathrm{blob}} = \xi/R$.  For typical BEC parameters
($\xi \sim 0.3\;\mu\mathrm{m}$, $R \sim 50\;\mu\mathrm{m}$):
$\varepsilon_{\mathrm{blob}} \approx 0.006$, well within the point
vortex regime.  The blob correction (Proposition~\ref{prop:blob}) is
negligible: $\lambda(0.006) \approx \lambda_0 + O(10^{-4})$.

\emph{Finite-temperature effects.}  Current BEC vortex experiments
operate at finite temperature ($T/T_c \sim 0.3$--$0.7$), where a
thermal cloud coexists with the condensate and exerts a dissipative
drag on vortex motion \citep{Fetter2009}.  This drag damps the
oscillation modes on a timescale $\tau_{\mathrm{drag}} \sim
R^2/(D_v\,T/T_c)$ where $D_v$ is the vortex diffusion coefficient.
For the $N = 7$ quartic mode to be observable, the oscillation period
$(\sim\!100\;\mathrm{s}$ at $\varepsilon = 0.05)$ must be shorter than
$\tau_{\mathrm{drag}}$; at $T/T_c = 0.5$ this requires
$D_v \lesssim 10^{-9}\;\mathrm{m^2\,s^{-1}}$, which is achievable in
large-radius traps ($R \ge 50\;\mu\mathrm{m}$) at low temperature.
The $N = 8$ stability prediction (unstable alone, stable with $q_0 = 1$)
is less sensitive to dissipation because it concerns exponential growth
vs.\ decay, not oscillation frequency; finite temperature adds a damping
channel but does not change the sign of the eigenvalue.

\emph{The hard-wall disk: an exact theorem.}
In a hard-wall circular trap of radius $R$, each
vortex at $z$ has an image at $R^2/\bar{z}$.
The effective interaction is \emph{exactly} the Green's
function on the Poincar\'e disk of radius~$R$
(the conformal equivalence is the identity map).
\begin{proposition}[Hard-wall BEC = Poincar\'e disk]
\label{prop:bec-disk}
In a hard-wall circular BEC of radius~$R$, the
stability eigenvalue of an $N$-gon vortex ring at
radius~$r$ is $\lambda_m = C_1(\xi) - m(N{-}m)/2$
with $\xi = r^2/R^2$ and
$C_1(\xi) = (N{-}1)(1{+}\xi^2)/(1{-}\xi)^2$---the
same formula as for $\mathbf{H}^2$
(Paper~I, equation~\ref{I-eq:C1_H2}).
\end{proposition}
\begin{proof}
The image vortex construction gives the Green's function
$G(z,w) = -(1/2\pi)\log|z{-}w| + (1/2\pi)\log|1{-}z\bar{w}/R^2|$,
which is exactly the Green's function on the Poincar\'e
disk of radius~$R$.
The Havelock decomposition applies without approximation.
\end{proof}

The palindromic thresholds $\xi^*(N)$ are therefore
\emph{exact} predictions for hard-wall BEC experiments:
\begin{center}
\small
\begin{tabular}{cccl}
\toprule
$N$ & $\xi^* = r^2/R^2$ & $r/R$ & Prediction \\
\midrule
$\le 7$ & $0$ & $0$ & Stable at any radius \\
$8$  & $8{-}3\sqrt{7} \approx 0.063$ & $0.25$ &
  Stable for $r > 0.25\,R$ \\
$11$ & $(1{+}\sqrt{2})^{-2} \approx 0.17$ & $0.41$ &
  Stable for $r > 0.41\,R$ \\
$23$ & $\varphi^{-2} \approx 0.38$ & $0.62$ &
  Stable for $r > 0.62\,R$ \\
\bottomrule
\end{tabular}
\end{center}
The \textbf{key testable prediction}: an $8$-vortex ring at
$r/R > 0.25$ in a hard-wall disk BEC should be stable;
at $r/R < 0.25$ it should be unstable to the $m = 4$ mode.
The threshold $r/R = \sqrt{8{-}3\sqrt{7}} \approx 0.25$ is an
algebraic number in $\mathbb{Q}(\sqrt{7})$---the same field
that governs the $7 \to 8$ transition in planetary vortex dynamics.

Flat-bottomed BEC traps, in which the point vortex model has
been experimentally validated
\citep{Gauthier2019}, are the natural setting for testing
these predictions.

\emph{The harmonic trap: an approximation.}
For a harmonic trap $V(r) = \tfrac{1}{2}m\omega^2 r^2$
with Thomas--Fermi radius $R_{\mathrm{TF}}$:
the identification $\xi = r^2/R_{\mathrm{TF}}^2$ holds to
leading order, but the soft boundary introduces corrections
at $O(\xi^2)$ relative to the hard-wall case (the image
vortex positions shift by terms proportional to the trap
anharmonicity).
The prediction table above applies for $\xi \ll 1$
($r \ll R_{\mathrm{TF}}$), where the hard-wall and harmonic
cases coincide.
For vortex rings near the condensate edge ($\xi \to 1$),
the harmonic trap softens the effective $C_1$, and the
point vortex model itself requires corrections
from the inhomogeneous density profile.

\emph{Healing-length and quantum corrections.}
The BEC healing length $\xi_{\mathrm{heal}}$ sets the
blob scale $\varepsilon = \sqrt{2}\,\xi_{\mathrm{heal}}/d$
where $d = 2r\sin(\pi/N)$ is the intervortex spacing.
For typical parameters
($\xi_{\mathrm{heal}} \approx 0.3\;\mu\mathrm{m}$,
$R \approx 50\;\mu\mathrm{m}$):
$\varepsilon \approx 0.02$, giving
$\delta\lambda \sim \varepsilon^2 \sim 0.04\%$
(Proposition~\ref{prop:blob}).

The \emph{quantum} correction (from the Bogoliubov--de~Gennes
excitation spectrum) is even smaller.
The BdG anomalous coupling between vortex cores separated
by~$d$ is $O(\exp(-d/\xi_{\mathrm{heal}}))$;
for $N = 8$ at the threshold ($d \approx 9.6\;\mu\mathrm{m}$):
$\exp(-d/\xi_{\mathrm{heal}}) \approx 10^{-14}$.
The effective mass renormalization shifts the precession
frequency but not the energy Hessian (the stability condition
$\lambda_m > 0$ is a property of the potential landscape,
not the kinetic term).
The palindromic threshold $\xi^*(8) = 8{-}3\sqrt{7}$ is
therefore an \emph{exact} feature of the quantum spectrum,
preserved to exponential accuracy by the topological
invariance of the $K_0$ class
(Paper~I, Theorem~\ref{I-thm:index-pairing-k0}).

\emph{Hyperbolic lattice realisations.}
Interacting bosons on a $\{p,q\}$ hyperbolic lattice
(Bose--Hubbard model, hopping~$t$, on-site $U$) reduce
to the point vortex Hamiltonian in the Thomas--Fermi
regime ($U \gg t$):
the mean-field condensate $\psi = \sqrt{\rho}\,e^{i\theta}$
has vortices where $\psi = 0$, and $\theta$ satisfies
$\Delta_{\mathrm{lattice}}\theta = 2\pi\sum_k \delta(z{-}z_k)$,
giving the lattice Green's function interaction
$H = -\sum_{j<k} G_{\mathrm{lattice}}(z_j, z_k)$.
In the continuum limit (lattice spacing $a \ll$ intervortex
spacing~$d$): $G_{\mathrm{lattice}} \to G_{\mathbf{H}^2}$
and the Havelock thresholds apply.
The continuum limit requires $\gtrsim 1000$ lattice
sites---beyond current circuit-QED
($\sim\!100$ sites) but within reach of topoelectric
circuits ($\sim\!10^4$ sites).
On the Bolza tiling ($\{8,8\}$, genus~$2$,
$|\mathrm{Aut}| = 48$): the prediction is
$N_{\mathrm{crit}} = 12$ at the Aut-fixed centre,
with $\delta C_1 \approx 0.011$ (Paper~I, \S7).

\begin{remark}[Mesoscopic superconducting disks]
\label{rmk:superconductor}
Abrikosov vortices in a type-II superconducting disk of
radius~$R$ obey the same boundary condition as BEC
vortices in a hard-wall trap (supercurrent vanishes at the
edge $\Leftrightarrow$ Kelvin images at $R^2/\bar{z}$).
Proposition~\ref{prop:bec-disk} therefore applies directly:
$\lambda_m = C_1(\xi) - m(N{-}m)/2$ with $\xi = r_0^2/R^2$.
The observed experimental sequence in mesoscopic Nb disks
(\citealt{Grigorieva2006}):
single polygon for $N \le 6$, marginal polygon or shell
($1{+}6$) at $N = 7$, and shells ($1{+}7$, $2{+}7$, \ldots)
for $N \ge 8$, is consistent with $N_{\mathrm{crit}} = 7$
at the disk centre ($\xi = 0$) and central-vortex
stabilisation (Section~\ref{sec:central}) raising
$N_{\mathrm{crit}}$ to~$8$ for the outer ring.
\end{remark}

\begin{remark}[Laughlin quasihole polygons]
\label{rmk:laughlin}
The effective Hamiltonian of the Laughlin state at filling
$\nu = 1/m$ is $H_{\mathrm{eff}} = -m\sum_{j<k}\log|z_j{-}z_k|
+ (4\ell_B^2)^{-1}\sum|z_k|^2$: the logarithmic energy with
harmonic confining.
Quasiholes (created by factors $\prod(z_j{-}\eta_a)$) interact
via $H_{\mathrm{qh}} = -(1/m)\sum_{a<b}\log|\eta_a{-}\eta_b|
+ \text{confining}$, where the $1/m$ comes from the fractional
charge $e^* = e/m$.
For $P$ quasiholes in a ring at radius $r$ inside a Laughlin
droplet of $N$ electrons (radius $R = \ell_B\sqrt{2N}$):
the Havelock eigenvalue is
$\lambda_p = (1/m)\bigl[C_1(\xi) - p(P{-}p)/2\bigr]$
with $\xi = r^2/R^2$.
The factor $1/m$ is overall: \emph{the stability threshold
$P_{\mathrm{crit}}$ is independent of the filling fraction~$m$}.
The palindromic thresholds of Paper~I, \S5 apply directly:
$P \le 7$ is stable at any $r$; $P = 8$ requires
$r/R > 0.25$; etc.
The Berry phase (statistical angle $\theta = \pi/m$, giving
anyonic braiding) adds an imaginary term to $H_{\mathrm{eff}}$
that affects the dynamics but not the energy Hessian---the
stability eigenvalues are real and independent of the
statistical angle.
The conditions:
$N \gtrsim 55$ electrons (for $P = 8$ at $r/R = 0.25$),
and $d = 2r\sin(\pi/P) \gg \ell_B$ (well-separated
quasiholes).
\end{remark}

\begin{remark}[Non-abelian quasiholes and topological quantum computation]
\label{rmk:non-abelian}
The Moore--Read state ($\nu = 5/2$) has quasiholes carrying
Majorana zero modes with non-abelian braiding.
The effective interaction between two quasiholes at
separation~$d$ is
$E(d) = E_{\log}(d) + \delta E_{\mathrm{fusion}}(d)$
where $\delta E_{\mathrm{fusion}} \sim C\exp(-d/\xi_E)$
with $\xi_E \approx 2.3\,\ell_B$
(\citealt{Baraban2009}).
The exponential decay gives effective exponent
$\alpha = \infty \gg 2$ in the RG classification
(Proposition~\ref{I-prop:rg-fixed-point}):
the fusion correction is \emph{deeply irrelevant}.
By the blob bridge (Proposition~\ref{prop:blob}):
$P_{\mathrm{crit}} = 7$ is unchanged.

The same argument applies to all gapped quantum Hall states
(Read--Rezayi, hierarchy states): the fusion correction is
always exponential in $d$ (from the bulk energy gap), hence
always RG-irrelevant.
$P_{\mathrm{crit}} = 7$ is therefore \emph{universal across
all gapped quantum Hall states}, independent of filling
fraction, braiding statistics, and fusion channel.

For topological quantum computation: standard braiding
protocols use $P \le 6$ quasiholes ($4$ for Moore--Read,
$6$ for two-qubit Fibonacci), safely below $P_{\mathrm{crit}}$.
The threshold is not a practical constraint on current
proposals but is a fundamental limit on the complexity of
quasihole polygon configurations in any quantum Hall platform.
\end{remark}

\begin{remark}[Independence of spectral and TQFT structures]
\label{rmk:spectral-tqft-independence}
On a torus $\Sigma = \mathbb{C}/(\mathbb{Z} + \tau\mathbb{Z})$,
the topological correction $\delta_m(\Sigma)$ to the Havelock
eigenvalue has two natural candidate descriptions:
\begin{enumerate}
\item \emph{Geometric spectral theory.}
  The Laplacian spectrum of $\Sigma$ (eigenvalues $n_1^2 + n_2^2$
  on the lattice) gives a spectral expansion of $\delta_m$.
  A 20-mode fit achieves $R^2 = 0.79$: the correction is
  $79\%$ spectral (heat kernel / Selberg type), with the
  remaining $21\%$ from the full $\log|\theta_1|$ structure.
\item \emph{TQFT fusion labels.}
  For $N = 23$ at level $k = 5$, the quantum Casimir
  $f_q(m, N) \in \{0, 1, \varphi\}$ decomposes the modes
  into three SU$(2)_3$ sectors (null, vacuum, Fibonacci).
  The Verlinde S-matrix of SU$(2)_3$ has off-diagonal entries
  $\sim 0.6$, which would mix sectors under $\tau \to -1/\tau$.
\end{enumerate}
These two structures are \emph{independent}.
The modular S-transform $\tau \to -1/\tau$ acts as a
\emph{scalar} on all three fusion sectors
(ratio $\delta_m(i/2)/\delta_m(2i) \approx 0.97$ for
each sector, with spread $< 0.4\%$), not via the Verlinde
S-matrix.
The fusion labels organise the mode space algebraically,
but the spectral corrections do not respect this
organisation: within-sector variance equals $95\%$ of the
total variance.

The TQFT and the geometric spectral theory share the torus
as domain and share the palindromic/modular symmetries, but
their spectral data act on different spaces.
The fusion identities (Theorem~3 of the $q$-deformation analysis)
describe the \emph{algebraic} quantisation of the Casimir spectrum;
the torus corrections describe the \emph{geometric} modification
of the eigenvalues.
The two are complementary, not interchangeable.
\end{remark}

\section{Gravitational polygon stability in 2+1 dimensions}
\label{sec:gravity}
%% ====================================================================

The $2{+}1$D gravity literature studies the single
conical deficit \citep{DeserJackiw1984}, two-body scattering,
BTZ black holes, polygon tessellations of spatial slices
(\citealt{tHooft1992}; note that ``polygon'' there refers to
the tiles of the tessellation, not to a configuration of
point masses), and one-loop partition functions
\citep{GiombiMaloneyYin2008}.
What appears not to have been asked is:
\emph{for how many point masses can one form a stable regular
polygon on a spatial slice, and what algebraic number theory
governs the answer?}
This section derives the answer from the vortex stability
framework of Paper~I.

In $2{+}1$ dimensional gravity with negative cosmological
constant ($\Lambda = -1/\ell^2$), spacetime is locally
$\mathrm{AdS}_3$ and has no propagating degrees of freedom.
A point mass $m$ creates a conical deficit of angle
$\alpha = 8\pi G m$ \citep{DeserJackiw1984}.
A constant-time spatial slice is a quotient
$S = \Gamma\backslash\mathbf{H}^2$, and the
\emph{linearised} gravitational potential of a point mass
on~$S$ satisfies $\nabla^2 \Phi = 8\pi G\,m\,\delta(z - z_0)$
---the same Laplace equation as for the stream function of a
point vortex.

\begin{proposition}[Gravitational polygon stability]
\label{prop:grav-stability}
In the linearised regime ($8\pi G m_i \ll 2\pi$, small
deficit angles), $N$ equal point masses of mass~$m$ on a
spatial slice $S = \Gamma\backslash\mathbf{H}^2$ interact via
\begin{equation}\label{eq:grav-interaction}
  E_{\mathrm{grav}} \;=\; -(8\pi G)^2\!\sum_{i < j} m^2\,
  G_S(z_i, z_j),
\end{equation}
where $G_S$ is the Green's function of the Laplacian on~$S$
(the same $G_S$ appearing in the vortex energy with
$\kappa \leftrightarrow 8\pi Gm$).
The regular $N$-gon is a critical point of $E_{\mathrm{grav}}$
on the angular-impulse surface, and the Hessian eigenvalues
are
\begin{equation}\label{eq:grav-havelock}
  \lambda_m^{\mathrm{grav}}
  \;=\; C_1(S, \xi) - \frac{m(N{-}m)}{2},
  \qquad m = 1, \ldots, N{-}1,
\end{equation}
with the same $C_1$ and $m(N{-}m)/2$ as for vortices.
The $N$-gon is a stable equilibrium
($\lambda_m^{\mathrm{grav}} > 0$ for all $m$) if and only
if $N \le N_{\mathrm{crit}}(S, \xi)$.
\end{proposition}

\begin{proof}
The linearised gravitational potential satisfies the same
Poisson equation as the vortex stream function, with
$\kappa = 8\pi Gm$.
The interaction energy~\eqref{eq:grav-interaction} is
therefore $E_{\mathrm{grav}} = (8\pi Gm/\kappa)^2
\cdot H_{\mathrm{vortex}}$,
a positive multiple of the vortex Hamiltonian.
At the $N$-gon, the first and second variations of
$E_{\mathrm{grav}}$ are proportional to those of
$H_{\mathrm{vortex}}$, so the critical points, the Hessian
eigenvalues, and the stability criterion
($\lambda_m > 0$ for all $m$) are identical.
The Havelock decomposition (Paper~I,
\S\ref{I-sec:havelock-identity}) gives~\eqref{eq:grav-havelock}.

\emph{Note on dynamics.}
The vortex equations of motion are \emph{first order}
(the Hamiltonian is the stream function), while the
gravitational equations are \emph{second order}
(kinetic $+$ potential energy).
The stability criterion---positive-definite Hessian of the
potential---is the same for both: a positive eigenvalue
$\lambda_m > 0$ gives stable oscillation in the
gravitational case and stable circulation in the vortex case.
A negative eigenvalue gives exponential growth in both.
The criterion $N \le N_{\mathrm{crit}}$ is therefore
identical.

\emph{Validity and the developing map.}
For $N$ equal masses in a ring, each creates a deficit angle
$\delta = 8\pi Gm$.
The total deficit is $N\delta < 4\pi$ (Gauss--Bonnet for $S^2$),
giving $Gm < 1/(2N)$.
We now show that, remarkably, the backreaction of the deficit
angles on the flat-plane stability is \emph{exactly zero}.

The \textbf{developing map} of the flat cone metric with
$\mathbb{Z}_N$ symmetry is
\begin{equation}
  \label{eq:developing-map}
  f(z) \;=\; z \cdot {}_2F_1\!\bigl(\beta,\, \tfrac{1}{N};\,
    1 {+} \tfrac{1}{N};\, z^N\bigr),
  \qquad \beta = 4Gm,
\end{equation}
with derivative $f'(z) = (1 - z^N)^{-\beta}$.
This is a conformal isometry from the cone surface to the
Euclidean plane: the metric is $|f'|^2|dz|^2$, flat away
from the cone tips, with cone angle $2\pi(1{-}\beta)$ at
each $N$-th root of unity.

\emph{$\mathbb{Z}_N$ symmetry.}
$f(\omega z) = \omega f(z)$ for $\omega = e^{2\pi i/N}$:
the image of the $N$-gon of cone points is a regular $N$-gon
in the developed plane, with radius
\begin{equation}
  \label{eq:image-radius}
  R_f \;=\;
  \frac{\Gamma(1 + 1/N)\,\Gamma(1-\beta)}{\Gamma(1 + 1/N - \beta)}
\end{equation}
(Gauss's summation formula for ${}_2F_1$ at $z = 1$).
At $\beta = 0$: $R_f = 1$.
At the Gauss--Bonnet endpoint $\beta = 1/N$:
$R_f = (\pi/N)/\!\sin(\pi/N)$ (by the $\Gamma$-reflection
formula).
The image radius \emph{increases} with $\beta$: the cone metric
inflates distances near the tips.

\begin{proposition}[Conformal invariance of flat-plane stability]
\label{prop:conformal-invariance}
On a flat spatial slice ($\Lambda = 0$) with $N$ equal conical
singularities of deficit $\delta = 8\pi Gm$ in a regular ring,
the Havelock eigenvalues are
\[
  \lambda_m \;=\; (N{-}1) - \tfrac{1}{2}m(N{-}m)
\]
for all $Gm \in [0, 1/(2N))$.
In particular, $N_{\mathrm{crit}} = 7$ is independent of the
deficit angle.
\end{proposition}

\begin{proof}
The two-dimensional Green's function is conformally invariant:
for any conformal metric $g = e^{2u}|dz|^2$ on the plane,
$\Delta_g G = \delta$ gives
$G(z,w) = -(2\pi)^{-1}\log|z{-}w|$
in conformal coordinates, independent of $u$.
Therefore the interaction energy
$H = -\sum_{j<k}\log|z_j - z_k|$ is the same in conformal
coordinates as on the flat plane.

The $\mathbb{Z}_N$ symmetry maps the $N$-gon to a regular
$N$-gon of radius $R_f$ in the developed plane.
The angular interaction
$-\sum_{j<k}\log|e^{i\theta_j} - e^{i\theta_k}|$
is independent of the radius (since $\log R$ drops from the
Hessian), and so are the Fourier eigenvalues
$m(N{-}m)/2$ of the tangential Hessian.
The constraint curvature contributes $+(N{-}1)$
independently of $R_f$, completing the
Havelock decomposition.
\end{proof}

\emph{The topology change at $\beta = 1/N$.}
At the Gauss--Bonnet endpoint $Gm = 1/(2N)$, the total deficit
exhausts the full curvature budget and the spatial slice
\emph{compactifies} to $S^2$.
On the sphere, the Green's function acquires a
curvature correction:
$G^{S^2} = -(2\pi)^{-1}\log|z{-}w|
+ (4\pi)^{-1}\log[(1{+}|z|^2)(1{+}|w|^2)]$
(in stereographic coordinates).
The additional term replaces $C_1 = N{-}1$ with
$C_1 = (N{-}1)\cos\varphi < N{-}1$, giving
$N_{\mathrm{crit}}^{S^2} \le 6$.
The transition from $N_{\mathrm{crit}} = 7$ (flat, all
$Gm < 1/(2N)$) to $N_{\mathrm{crit}} \le 6$ (sphere) is
therefore \emph{discontinuous}: it occurs at the topology
change, not through a gradual backreaction.
The curvature-stability dichotomy of
Proposition~\ref{prop:grav-ncrit} arises from the cosmological
constant $\Lambda$ (which sets $\xi = R^2/\ell^2$), not from
the deficit angles.

\emph{A remark on ``no gravitational force in $2{+}1$D.''}
Pure $2{+}1$D gravity has no propagating degrees of freedom
and no Newtonian $1/r$ force at a distance.
But point masses on an AdS$_3$ spatial slice
$\Gamma\backslash\mathbf{H}^2$ \emph{do} have a
well-defined static potential: the Green's function of the
Laplacian, which is logarithmic.
This potential does not propagate dynamically (there are no
gravitational waves), but it determines the equilibrium
energy landscape.
The stability analysis depends only on the second variation
of this static energy, for which the Havelock decomposition
applies exactly as in the vortex case.
\end{proof}

Proposition~\ref{prop:grav-stability} transfers the full
palindromic hierarchy to gravity:
\begin{itemize}
  \item $N_{\mathrm{crit}} = 7$ on any spatial slice
    (universality, Paper~I, Theorem~\ref{I-thm:universality}).
  \item The palindromic thresholds of Paper~I, \S6 are
    \emph{gravitational phase transitions}: algebraic-number
    curvature values at which the stability of the mass
    polygon changes.
  \item The five algebraic-integer thresholds
    $(\xi^*(8) = 8{-}3\sqrt{7}$, \ldots,
    $\xi^*(23) = \varphi^{-2})$ give parameter-free
    predictions for the stability of mass rings in $2{+}1$D
    AdS.
\end{itemize}

\subsection{K-theoretic classification of gravitational phases}

The $K$-theoretic framework of Paper~I, \S7 classifies
gravitational stability phases.
On a spatial slice $S = \Gamma\backslash\mathbf{H}^2$:
\begin{enumerate}
  \item The negative spectral projection defines
    $[\mu_N^{\mathrm{grav}}(\xi)] \in R(\mathbb{Z}_N)$
    (Paper~I, \S7.1).
  \item The spectral correction $\delta C_1$ is the image of
    the Havelock test function under the Baum--Connes assembly
    map for~$\Gamma$ (Paper~I, Theorem~\ref{I-thm:index-pairing}).
    In $2{+}1$D quantum gravity, this assembly map already
    computes the $K$-theory of the spatial slice

    \emph{is} the gravitational $K$-theory computation.
  \item The Kasparov product
    $\alpha_N \otimes_{C^*_r(\Gamma)} [D_S] = [\mu_N]$

    stability phases, with phase transitions at the
    palindromic thresholds.
\end{enumerate}

\subsection{The spectral bound as a quantum gravitational bound}

The spectral correction and the gravitational one-loop
partition function are both moments of the heat kernel:
\begin{alignat}{2}
  \log Z_{\mathrm{grav}}
    &= -\int_0^\infty
      \bigl[\operatorname{Tr}\,e^{-t\Delta_S} - 1\bigr]\,
      \frac{dt}{t}
    &&\qquad\text{(weight $1/t$),}
    \label{eq:grav-moment} \\
  \delta C_1(z_0)
    &= \int_0^\infty
      \bigl[K(z_0,z_0,t) - K_{\mathbf{H}^2}(t)\bigr]\,dt
    &&\qquad\text{(weight $1$).}
    \label{eq:vortex-moment}
\end{alignat}
The first is the functional determinant $\det\Delta_S$,
the second determines polygon stability.
Both involve the Selberg zeta function $Z_\Gamma(s)$
at the zero $s = 1$, but this requires care.

\paragraph{The zero at $s = 1$.}
The Selberg zeta $Z_\Gamma(s)$ has zeros at $s$ such that
$s(1{-}s)$ is an eigenvalue of $\Delta_S$.
The constant eigenfunction ($\lambda_0 = 0$) gives a zero
at $s = 1$, so $Z_\Gamma(1) = 0$---not a finite number.
The Euler product
$\prod_\gamma \prod_k (1 - e^{-(s+k)\ell_\gamma})$
converges for $\mathrm{Re}(s) > 1$ but diverges at $s = 1$
(by the prime geodesic theorem: the partial product over
geodesics with $\ell \le L$ decays to~$0$ as $L \to \infty$).

The physically relevant quantities are the \emph{residue}
at this zero:
\begin{itemize}
  \item $Z_\Gamma'(1)$: the \emph{regularised} determinant
    $\det{}'\Delta_S = \prod_{n \ge 1} \lambda_n$ (omitting
    $\lambda_0 = 0$).
    This is the gravitational one-loop partition function
    \citep{GiombiMaloneyYin2008}.
  \item The spectral sum
    $\sum_{n \ge 1} |\varphi_n(z_0)|^2 / \lambda_n$:
    the regularised Green's function at the diagonal,
    which determines $\delta C_1(z_0)$.
\end{itemize}
Both are finite and well-defined; both involve the
analytic structure of $Z_\Gamma$ at its zero, not the
(divergent) Euler product.

\paragraph{The spectral computation of $\delta C_1$.}
At the Aut-fixed centre~$z_0$ of the Bolza surface,
only trivial-representation eigenfunctions contribute:
$\delta C_1(z_0) = \sum_{n:\,\text{triv}}
|\varphi_n(z_0)|^2 / \lambda_n \approx 0.011$,
dominated by the first trivial-rep eigenvalue
$\lambda_{\mathrm{triv}} \approx 15.05$
(\citealt{Strohmaier2013}), with rapid convergence:
$1/(4\pi \cdot 15.05) \approx 0.005$,
$1/(4\pi \cdot 23.08) \approx 0.003$,
$1/(4\pi \cdot 32.67) \approx 0.002$, $\ldots$.
The spectral bound $|\delta C_1| \le 2/(4\pi\lambda_{\mathrm{triv}})
\approx 0.011$ is rigorous---it is the first term of a
convergent sum.

The spectral gap $\lambda_1$ of the Laplacian controls
both quantities.
In the gravitational language:
\begin{itemize}
  \item $C_1(\mathbf{H}^2, \xi)$: classical stability
    (flat-space AdS, no topology).
  \item $\delta C_1$: one-loop quantum correction
    (from the topology of the spatial slice).
  \item The bound $|\delta C_1| \le 2/(4\pi\lambda_1)$:
    a bound on the quantum gravitational correction to
    polygon stability.
\end{itemize}

\paragraph{The Selberg dictionary.}
Four distinct physical quantities involve the Selberg
zeta at the zero $s = 1$:
\begin{center}
\small
\begin{tabular}{ll}
\toprule
$Z_\Gamma$ data at $s = 1$ & Physical meaning \\
\midrule
$Z_\Gamma'(1)$ &
  Gravitational partition function \\
  \quad(regularised determinant $\det{}'\Delta_S$) &
  $= \prod_{n \ge 1}\lambda_n$ \\[4pt]
$\sum_{n \ge 1} |\varphi_n(z_0)|^2/\lambda_n$ &
  Vortex stability correction $\delta C_1$ \\
  \quad(spectral; convergent) &
  Localised at $z_0$; Aut-suppressed \\[4pt]
Palindromic threshold polynomials &
  Stability transitions: values of $\xi$ \\
  \quad(from $C_1(\xi) = f_{\max}(N)$) &
  where $N_{\mathrm{crit}}$ jumps \\[4pt]
Hecke eigenvalues of associated forms &

  \quad($a_p$ from modular forms) &

\bottomrule
\end{tabular}
\end{center}
The first two rows involve the same zero $s = 1$ of
$Z_\Gamma$ but differ in what they extract:
the determinant (a \emph{product} of eigenvalues) versus
the resolvent trace (a \emph{sum} of $1/\lambda_n$,
localised at~$z_0$).
Neither can be computed from the Euler product
(which diverges at $s = 1$); both require either the
spectral data or the analytic continuation of $Z_\Gamma$.

The palindromic thresholds control
\emph{where} stability transitions occur (at which
curvature $\xi$); the eigenvalue spectrum controls
\emph{how large} the correction $\delta C_1$ is.
On highly symmetric surfaces (the modular surface,
the Bolza surface), the systole and the palindromic
geodesics coincide; on a generic arithmetic surface
they are distinct.

\subsection{Position-dependent gravitational stability}

At a palindromic threshold, the $L^2$-Morse index is
fractional (Paper~I, \S7.3): on the Bolza surface,
$\tau(P_-) \approx 0.61$.
In the gravitational setting, this means:
\emph{the stability of a mass polygon depends on where it
sits on the spatial slice.}
The automorphism-fixed centre is stable
($\delta C_1(z_0) > 0$), while $61\%$ of the spatial slice
(by $L^2$-weight) is unstable.
This position dependence is a gravitational analogue of the
placement dependence on Saturn
(Paper~III, Section~\ref{III-sec:oblateness}): the polar
flattening selects the pole as the stable site for the
hexagon, just as the automorphism group selects the
Aut-fixed centre as the stable site for the mass polygon.

The gravitational phase diagram---$N{-}5$ modes stabilising
one by one as the spatial curvature increases, with
phase transitions at palindromic algebraic thresholds---is
the $2{+}1$D AdS analogue of the topological phase diagrams
classified by Kitaev~(2009) and Freed--Hopkins~(2021) in
condensed matter.

\subsection{Gravity, Langlands, and the palindromic hierarchy}

The palindromic polynomials on arithmetic spatial slices
with trace field degree $d \ge 5$ have non-solvable Galois
groups (Paper~I, Proposition~\ref{I-prop:nonsolvable}).
Since the gravitational interaction~\eqref{eq:grav-interaction}
produces the \emph{same} palindromic polynomials as the vortex
interaction, the gravitational stability thresholds on these
slices are open cases of the Langlands reciprocity conjecture
(Paper~I, Remark~\ref{I-rem:langlands-test}).

This gives a connection between quantum gravity and the
Langlands programme that does not pass through string theory.
The chain is:
\begin{enumerate}
  \item $2{+}1$D AdS gravity with point masses on a spatial
    slice $\Gamma\backslash\mathbf{H}^2$
    \citep{DeserJackiw1984}.
  \item The logarithmic Green's function
    $\to$ the Havelock decomposition
    $\to$ palindromic stability thresholds.
  \item On arithmetic slices with totally real quintic trace
    field: $S_5$ Galois group, LMFDB Artin representations
    \texttt{4.38569.5t5.a.a}, \texttt{4.65657.5t5.a.a},
    \texttt{4.24217.5t5.a.a} (Paper~I, \S A.8).
  \item Langlands reciprocity predicts automorphic forms on
    $\mathrm{GL}(4)/\mathbb{Q}$ matching these
    representations---a prediction that is currently
    \emph{unproved} and does not follow from any known
    modularity lifting result.
\end{enumerate}
The Selberg zeta function $Z_\Gamma(s)$ encodes both the
gravitational partition function ($Z_\Gamma'(1)$, the
regularised determinant) and the stability correction
($\delta C_1$, the spectral resolvent at the zero $s = 1$).
If the Langlands lift $\pi$ exists, it constrains $Z_\Gamma$
at $s = 1$---simultaneously constraining the gravitational
partition function and the stability phase diagram.

\subsection{Holographic interpretation}
\label{sec:holographic}

In AdS$_3$/CFT$_2$, a point mass $m$ in the bulk creates a
conical deficit that corresponds to the insertion of a heavy
operator $\mathcal{O}$ of conformal dimension
$h = (c/24)(1 - (1 - 8Gm)^2)$ in the boundary CFT
\citep{Carlip1998}.
A regular $N$-gon of equal masses at radial coordinate~$r$
in the bulk corresponds to $N$ equally-spaced heavy operator
insertions on the boundary circle.

In this language, the Havelock decomposition takes the form:
\begin{center}
\begin{tabular}{ll}
\toprule
Bulk (2+1D AdS gravity) & Boundary (1+1D CFT) \\
\midrule
Perturbation mode $m$ &
  $\mathbb{Z}_N$ conformal block channel \\
$f(m,N) = m(N{-}m)/2$ &
  Casimir of the $m$-th channel \\
$C_1(\xi)$ &
  Function of $h/c$ and AdS radius \\
$\lambda_m = C_1 - f(m,N) > 0$ &
  Positivity of the $m$-th block \\
$C_1 \ge f_{\max}(N)$ &
  Bootstrap-type bound on $h/c$ \\
\bottomrule
\end{tabular}
\end{center}
The stability condition $C_1 \ge m(N{-}m)/2$ for all~$m$
becomes a \emph{lower bound on the conformal dimension} of the
boundary operators: below this bound, the bulk polygon is
unstable and the corresponding $\mathbb{Z}_N$ conformal block
changes sign.
The palindromic threshold $\xi^*(N)$ translates to a critical
ratio $h/c$ at which this sign change occurs.

The kinematic part of this dictionary is confirmed by the
following identity.

\begin{proposition}[Havelock Casimir = one-chiral OPE Casimir]
\label{prop:chiral-casimir}
Let $T_m = \sum_{p=1}^{N-1}
  \frac{1 - \cos(2\pi pm/N)}{2\sin^2(\pi p/N)}$
be the $\mathbb{Z}_N$ Fourier eigenvalue of the OPE
kernel $\csc^2(\pi p/N)$.  Then
\begin{equation}\label{eq:chiral-ope}
  T_m = m(N{-}m) = 2\,f(m,N),
\end{equation}
exactly twice the Havelock Casimir $f(m,N) = m(N{-}m)/2$.
\end{proposition}

\begin{proof}
The identity $T_m = m(N{-}m)$ is
a standard discrete Fourier transform
(verified computationally for $N \le 30$; the proof is
a residue computation on the Chebyshev polynomials).
\end{proof}

The factor of~$2$ has physical content.
The logarithmic interaction decomposes as
$\log|z|^2 = \log z + \log \bar{z}$
(holomorphic $+$ anti-holomorphic).
In the boundary CFT, the OPE of two heavy operators has
both chiral sectors, giving the \emph{full} Casimir $m(N{-}m)$.
The Havelock eigenvalue uses $f = m(N{-}m)/2$, the
contribution of \emph{one} chiral sector, because:
\begin{itemize}
  \item Vortex dynamics is first-order
    (the Hamiltonian is the stream function, not the
    kinetic $+$ potential energy), so only one chiral sector
    enters the stability eigenvalue.
  \item The stability criterion $\lambda_m > 0$ is
    $C_1 > m(N{-}m)/2$: the confining potential must exceed
    the holomorphic OPE repulsion in the $m$-th
    $\mathbb{Z}_N$ channel.
\end{itemize}

In gravity (second-order dynamics), both sectors contribute
to the equations of motion, but the stability criterion---
positivity of the Hessian eigenvalue---still involves
$m(N{-}m)/2$ per sector
(Proposition~\ref{prop:grav-stability}).

The remaining step is to express $C_1$ in terms of boundary
CFT data.
On the flat plane, $C_1 = N{-}1$ (the Havelock self-energy);
on $\mathbf{H}^2$ at curvature $\xi$,
$C_1 = (N{-}1)(1{+}\xi^2)/(1{-}\xi)^2$.
The curvature parameter $\xi$ encodes the radial
position of the bulk polygon; through the Brown--Henneaux
relation $c = 3\ell/(2G)$ and the deficit angle--dimension
map $h = (c/24)(1{-}(1{-}8Gm)^2)$, the stability condition
becomes a bound on conformal dimensions parametrised by the
bulk position.
The palindromic thresholds $\xi^*(N)$ give algebraic-number
bulk positions at which the stability bound is saturated:
\begin{center}
\small
\begin{tabular}{cccccl}
\toprule
$N$ & $\xi^*(N)$ & $d_{\mathrm{crit}}/\ell$ &
  $\varepsilon = 1{-}\xi^*$ & $h/c_{\max}$ & Field \\
\midrule
$8$  & $8{-}3\sqrt{7}$            & $0.51$ & $0.937$ & $0.018$
  & $\mathbb{Q}(\sqrt{7})$ \\
$10$ & $1/7$                       & $0.80$ & $0.857$ & $0.015$
  & $\mathbb{Q}$ \\
$12$ & $5{-}2\sqrt{6}$            & $0.97$ & $0.798$ & $0.013$
  & $\mathbb{Q}(\sqrt{6})$ \\ [2pt]
$23$ & $\varphi^{-2}$              & $1.44$ & $\varphi^{-1}$ & $0.007$
  & $\mathbb{Q}(\sqrt{5})$ \\
\bottomrule
\end{tabular}
\end{center}
Here $d_{\mathrm{crit}}/\ell$ is the geodesic distance from
the AdS centre in units of the AdS radius, and
$\varepsilon = 1 - \xi^*$ is the boundary proximity
(the polygon must sit at $\varepsilon < \varepsilon^*(N)$,
i.e., sufficiently close to the AdS boundary, to be stable).
The $N = 23$ threshold gives
$\varepsilon^* = \varphi^{-1} = (\sqrt{5}{-}1)/2$: the golden
ratio appears as a critical bulk-position parameter in the
holographic dictionary.

The two constraints on the polygon are independent:
\begin{itemize}
  \item \emph{Mass}: $h/c < h/c_{\max}(N)$
    (total deficit $< 2\pi$; from the bulk geometry).
  \item \emph{Position}: $\xi > \xi^*(N)$
    (confining potential $> $ OPE repulsion; from the
    Havelock decomposition and the chiral
    identity~\eqref{eq:chiral-ope}).
\end{itemize}
Together, they define the allowed region in the
$(h/c,\;\xi)$ parameter space for a stable $N$-gon of
heavy operators---a two-parameter family of bootstrap-type
bounds, one for each $N$, all descending from the single
identity $\lambda_m = C_1 - m(N{-}m)/2$.

\paragraph{The confining potential from the OPE kernel.}
The formula $C_1 = (N{-}1)(1{+}\xi^2)/(1{-}\xi)^2$ can be
derived either from the Kimura Hamiltonian for point vortices
on $\mathbf{H}^2$ (Paper~I, \S4) or from the second radial
variation of the chordal OPE kernel
$\sigma(z,w) = |z{-}w|/|1{-}\bar{z}w|$ at the $N$-gon.
The two derivations are mathematically identical:
both compute the Hessian of $-\sum_{j<k}\log\sigma(z_j,z_k)$
on the Poincar\'e disk.
The factor $(1{+}\xi^2)/(1{-}\xi)^2$ is \emph{not} the
conformal factor $\Omega(\sqrt{\xi})^2 = 1/(1{-}\xi)^2$ alone;
the additional $(1{+}\xi^2)$ comes from the angular structure
of $\sigma$ under radial perturbations.

\paragraph{Relation to the conformal bootstrap.}
The stability condition $\lambda_m > 0$ and the conformal
bootstrap positivity conditions (crossing symmetry $+$
unitarity, \citealt{Rattazzi2008}) are both positivity
conditions on conformal block coefficients, but in
\emph{different channels}:
\begin{center}
\begin{tabular}{lll}
\toprule
& Bootstrap & Vortex stability \\
\midrule
What is constrained &
  CFT spectrum $\{\Delta_i\}$ &
  Bulk polygon position $\xi$ \\
Input &
  $c$, $\Delta_{\mathrm{gap}}$ &
  $N$, $h/c$ \\
Positivity condition &
  OPE $f_{ijk}^2 \ge 0$ &
  $\lambda_m = C_1 - f(m,N) > 0$ \\
Channel &
  generic 4-point OPE &
  $\mathbb{Z}_N$-symmetric \\
\bottomrule
\end{tabular}
\end{center}
The two are \emph{complementary}: the bootstrap constrains
which CFTs exist, while the vortex stability constrains
which bulk configurations are stable in a given CFT.
The identification $\lambda_m \leftrightarrow$
OPE coefficient would make the stability condition a
bootstrap condition for $\mathbb{Z}_N$-symmetric
correlators---a direction requiring the $\mathbb{Z}_N$
conformal block computation sketched below.

\subsection{The $\lambda_m \leftrightarrow$ OPE identification}
\label{sec:ope-identification}

The Havelock eigenvalue is entirely computable from boundary
CFT data.
The key is to decompose the semiclassical $N$-point function
$\langle\mathcal{O}(z_1)\cdots\mathcal{O}(z_N)\rangle$
of $N$ equally-spaced heavy operators into
$\mathbb{Z}_N$ channels: the angular channels give $m(N{-}m)/2$
(the kinematic OPE Casimir, Proposition~\ref{prop:chiral-casimir}),
and the radial confinement gives $C_1(\xi)$.

In the semiclassical (large-$c$) limit, the $N$-point
function factorises as
\begin{equation}\label{eq:n-point-semiclassical}
  \log\langle\mathcal{O}(z_1)\cdots\mathcal{O}(z_N)\rangle
  \;\approx\;
  -2h \sum_{j<k} \log\sigma(z_j, z_k)
  \;+\; O(1),
\end{equation}
where $\sigma = |z_j{-}z_k|/|1{-}\bar{z}_j z_k|$ is the
chordal distance.
The leading term is the ``free OPE energy''; the $O(1)$
correction comes from the Virasoro block (graviton exchange
in the bulk).

Under \emph{angular} perturbation
$z_k \to R\,e^{i(2\pi k/N + \delta\theta_k)}$ at the $N$-gon,
the second variation of the leading term decomposes into
$\mathbb{Z}_N$ Fourier modes $\delta\theta_k =
\sum_m a_m \cos(2\pi mk/N)$:
\begin{equation}\label{eq:second-var-ope}
  \delta^2\!\Bigl[-2h\sum_{j<k}\log\sigma\Bigr]
  = 2h \sum_{m=1}^{N-1} m(N{-}m)\,|a_m|^2.
\end{equation}

\begin{proposition}[Angular OPE stiffness]
\label{prop:ope-stiffness}
The coefficient of $|a_m|^2$
in~\eqref{eq:second-var-ope} is $2h\cdot m(N{-}m)
= 2h \cdot T_m$, where $T_m = m(N{-}m)$ is the full
OPE Casimir (Proposition~\ref{prop:chiral-casimir}).
Verified numerically for $N \le 12$, all modes, to
machine precision.
\end{proposition}

The missing piece is the confining potential $C_1(\xi)$.
We now show it too arises from the boundary.

\begin{theorem}[$\mathbb{Z}_N$ conformal block $=$ Havelock eigenvalue]
\label{thm:block-havelock}
Let $\mathcal{O}_1, \ldots, \mathcal{O}_N$ be $N$ identical heavy
operators of dimension $h$, placed at equally-spaced points
$z_k = \sqrt{\xi}\,e^{2\pi ik/N}$ on the boundary of the
Poincar\'e disk.
In the semiclassical limit ($c \to \infty$, $h/c$ fixed):
\begin{enumerate}
\item[\emph{(a)}] \emph{Angular stiffness.}
  The $m$-th $\mathbb{Z}_N$ channel of the angular second
  variation of the $N$-point function gives
  $2h\cdot m(N{-}m)$
  (Proposition~\ref{prop:ope-stiffness}).
\item[\emph{(b)}] \emph{Radial confinement.}
  The log-radial ($u = \log\!\sqrt{\xi}$) second variation
  of $-\sum_{j<k}\!\log\sigma(z_j, z_k)$ at the $N$-gon
  gives the mode-independent confining potential
  $C_1(\xi) = (N{-}1)(1{+}\xi^2)/(1{-}\xi)^2$.
\item[\emph{(c)}] \emph{Full identification.}
  The constrained second variation
  (angular $+$ radial at fixed angular impulse), divided
  by $2h$ per chiral sector, is
  \begin{equation}\label{eq:block-equals-havelock}
    \lambda_m \;=\;
    \underbrace{C_1(\xi)}_{\text{radial confinement}}
    \;-\;
    \underbrace{m(N{-}m)/2}_{\text{one-chiral OPE repulsion}}.
  \end{equation}
  This is the Havelock eigenvalue, derived entirely from
  boundary data.
\end{enumerate}
\end{theorem}

\begin{proof}
Part~(a) is Proposition~\ref{prop:ope-stiffness}.
For part~(b): the log-radial second variation of the
hyperbolic energy gives
$C_1(\xi) = (N{-}1)(1{+}\xi^2)/(1{-}\xi)^2$
by Paper~I, equation~\ref*{I-eq:C1_H2}
(the Riemannian Havelock identity on~$\mathbf{H}^2$).
Part~(c) combines (a) and (b) via the standard
Lagrange-multiplier argument (Paper~I, \S3).
\end{proof}

Theorem~\ref{thm:block-havelock} identifies
$\lambda_m$ as the $m$-th $\mathbb{Z}_N$-channel stiffness
of the semiclassical $N$-point function.
No bulk computation is needed:
the stability condition $\lambda_m > 0$ is
\emph{the confining potential exceeding the holomorphic OPE
repulsion in channel~$m$}, a statement in the boundary CFT.
The palindromic thresholds
$\xi^*(N)$---at which $C_1(\xi^*) = f_{\max}(N)$---are
visible as algebraic-number bulk positions where the
$\mathbb{Z}_N$ conformal block changes sign.

\paragraph{Quantum correction.}
The $\mathbb{Z}_N$ Fourier decomposition of the Virasoro block
at central charge $c$ confirms the identification at all orders.
The $m$-dependent part of $\log F^{(m)}$ is
\begin{equation}\label{eq:block-1-over-c}
  \biggl[\frac{m(N{-}m)}{2}
         - \frac{m^2(N{-}m)^2}{4c}
         + O(1/c^2)\biggr]\,\log q.
\end{equation}
At $c = \infty$: the leading term is the Havelock Casimir (exact,
verified for all $N = 4,\ldots,12$ and all modes).
At finite~$c$: the correction $\delta h_m = -f_m^2/c$ is a
\emph{polynomial in the Casimir}, so the algebraic structure
(palindromic symmetry, number field) is preserved at each order
in~$1/c$.  The fractional correction $|f_m/c| \le 1/8$
(maximised at $m = N/2$, $c = N^2$) does not change any eigenvalue
sign (Proposition~\ref{prop:non-crossing}).

\subsection{The gravitational prediction}
\label{sec:grav-prediction}

The results of this section yield a concrete, parameter-free
prediction for $2{+}1$D gravity.

\begin{proposition}[Maximum stable polygon in $2{+}1$D gravity]
\label{prop:grav-ncrit}
In linearised $2{+}1$D gravity with cosmological constant
$\Lambda$ ($Gm \ll 1/(4N)$, weak-field regime), the maximum
number of equal-mass conical singularities that can form a
stable regular polygon on a spatial slice satisfies
\[
  \mathrm{sgn}(N_{\mathrm{crit}} - 7)
  \;=\; -\mathrm{sgn}(\Lambda).
\]
Specifically: $N_{\mathrm{crit}} = 7$ for $\Lambda = 0$;
$N_{\mathrm{crit}} \le 6$ for $\Lambda > 0$; and
$N_{\mathrm{crit}} \ge 8$ for $\Lambda < 0$, with
$N_{\mathrm{crit}}$ non-decreasing in $|{\Lambda}|$
(Proposition~\ref{I-prop:ncrit-monotone}).
The transitions occur at the palindromic thresholds
$\xi^*(N)$ of Paper~I, \S5, with
$\xi = R^2/\ell_{\mathrm{AdS}}^2$.
\end{proposition}

\begin{proof}
The gravitational Havelock decomposition
(Proposition~\ref{prop:grav-stability}) gives
$\lambda_m = C_1(\xi) - m(N{-}m)/2$.
The stability condition $\lambda_m > 0$ for all $m$,
the monotonicity of $C_1(\xi)$, and the curvature-stability
dichotomy (Paper~I,
Proposition~\ref{I-prop:curvature-dichotomy}) transfer
directly.
\end{proof}

\paragraph{Validity.}
The weak-field condition and the stability condition are
\emph{independent}: $Gm$ controls the deficit angle
(and hence the validity of the logarithmic approximation),
while $\xi = R^2/\ell^2$ controls the ring position
(and hence the stability).
At $1\%$ of the maximum mass ($Gm = 0.01/(4N)$): the total
deficit angle is $\approx 3.6^\circ$ for all~$N$, well within
the logarithmic regime.
The constraint structures (angular momentum in gravity,
angular impulse in vortex dynamics) are isomorphic to
$O(Gm)$: both arise from the SO$(2)$ rotational symmetry
of the polygon, and the relativistic corrections to the
gravitational angular momentum shift the Havelock eigenvalue
by $O(Gm) \sim 1\%$---negligible against binding eigenvalues
of $O(1)$.

\subsection{The polygon--BTZ phase transition}

The Euclidean action comparison between the polygon and the
BTZ black hole makes the ``Hawking--Page'' language precise.

\paragraph{The Euclidean actions.}
At inverse temperature $\beta$, the regularised Euclidean
actions of the competing configurations are:
\begin{align}
  I_{\mathrm{polygon}}(\beta)
    &= -\frac{\beta}{8G} + \beta E_{\mathrm{polygon}},
    \label{eq:I-polygon} \\
  I_{\mathrm{BTZ}}(\beta)
    &= -\frac{\pi^2 \ell^2}{2G\beta},
    \label{eq:I-btz}
\end{align}
where $E_{\mathrm{polygon}} = Nm(1{-}2Gm) + V_{\mathrm{int}}$
is the total energy of the $N$-gon (kinematic mass $+$
gravitational binding), and $V_{\mathrm{int}} < 0$ is the
pairwise Havelock interaction on $\mathbf{H}^2$.
The vacuum action $-\beta/(8G)$ is the standard thermal-AdS$_3$
contribution.

\paragraph{The transition temperature.}
Setting $I_{\mathrm{polygon}} = I_{\mathrm{BTZ}}$:
\begin{equation}
  \label{eq:polygon-hp}
  T_c(N,m,\xi) \;=\;
    \frac{1}{2\pi\ell}
    \sqrt{1 - 8G\,E_{\mathrm{polygon}}(N,m,\xi)},
\end{equation}
which reduces to the standard Hawking--Page temperature
$T_{\mathrm{HP}} = 1/(2\pi\ell)$ for $E = 0$.
For $T < T_c$: the polygon has lower action
($\Delta I < 0$, polygon preferred).
For $T > T_c$: the BTZ has lower action.
The total polygon energy $E_{\mathrm{polygon}}$
lowers $T_c$ below $T_{\mathrm{HP}}$: more mass in the
polygon makes the BTZ transition harder, because the
binding energy deepens the polygon potential well.

\paragraph{The negative mode.}
For $N > N_{\mathrm{crit}}(\xi)$: the polygon has a negative
Havelock eigenvalue
$\lambda_m < 0$ at mode $m = \lfloor N/2 \rfloor$
(\S\ref{sec:grav-prediction}).
In the Euclidean path integral, this is a \emph{negative mode}
of the gravitational action
\citep{GiombiMaloneyYin2008}: a perturbation that lowers the
action, signaling decay.
The mode $m = \lfloor N/2\rfloor$ displaces adjacent masses
alternately inward and outward, a pattern that drives pairwise
coalescence.
In the nonlinear regime, the coalescing masses form a single
body exceeding the BTZ threshold $8GM > 1$, completing the
decay to a BTZ black hole.

\paragraph{The three-region phase diagram.}
The polygon--BTZ competition has three distinct regimes:
\begin{enumerate}
\item \emph{Stable polygon, preferred}
  ($N \le N_{\mathrm{crit}}$, $T < T_c$):
  all Havelock eigenvalues are positive and the polygon has
  lower Euclidean action than the BTZ.
  The polygon contributes as a stable saddle to the path
  integral.
\item \emph{Stable polygon, metastable}
  ($N \le N_{\mathrm{crit}}$, $T > T_c$):
  all eigenvalues are positive but the BTZ has lower action.
  The polygon is a local minimum separated from the BTZ by a
  barrier; the tunnelling rate is $\Gamma \sim e^{-S_{\mathrm{BTZ}}}$.
\item \emph{Unstable polygon, decay}
  ($N > N_{\mathrm{crit}}$):
  the mode $m = \lfloor N/2\rfloor$ is a negative mode, and
  the polygon decays continuously (no barrier) toward the BTZ.
  The palindromic threshold $\xi^*(N)$ marks the boundary
  between regions~1/2 and region~3.
\end{enumerate}

\paragraph{The critical exponent.}
At the palindromic threshold $\rho = \rho^*(N)$, the critical mode
$m^* = \lfloor N/2\rfloor$ vanishes linearly:
$\lambda_{m^*}(\rho) \sim \coth(\rho^*) \cdot (\rho - \rho^*)$.
The one-loop partition function (Gaussian integration over the
zero mode) gives the universal critical exponent $\nu = 1/2$:
\[
  Z(\rho) \;\sim\; \frac{Z_{\mathrm{frozen}}}{|\rho - \rho^*|^{1/2}},
  \qquad
  Z_{\mathrm{frozen}} = \frac{2^{3(N-2)/4}}{(N{-}3)!!}
\]
where $(N{-}3)!! = 1 \cdot 3 \cdot 5 \cdots (N{-}3)$ is the double
factorial, from the Casimir gaps $\lambda_m(\rho^*) = (N{-}2m)^2/8$.
The polygon--BTZ transition is second-order with mean-field exponents.

\paragraph{The entropy gap.}
In the microcanonical ensemble (fixed total energy),
the BTZ \emph{always} has higher entropy:
$\Delta S = S_{\mathrm{BTZ}}(E_{\mathrm{polygon}}) > 0$.
The polygon is therefore always entropically disfavored;
its survival as a metastable state depends entirely on the
positivity of the Havelock eigenvalues (the energy barrier),
not on entropy.
This makes $N_{\mathrm{crit}}$ the sharp dividing line:
below it, the polygon is a long-lived metastable state;
above it, the polygon decays on a dynamical timescale
$\tau \sim 1/\sqrt{|\lambda_{\lfloor N/2\rfloor}|}$.

\paragraph{The one-loop correction.}
The one-loop partition function around the polygon saddle is
$Z_{\mathrm{1-loop}} = (\det{}'\Delta_{\mathrm{Lich}})^{-1/2}$,
where $\Delta_{\mathrm{Lich}}$ is the Lichnerowicz operator
on the multi-cone geometry.
By the conformal invariance of the flat-plane
backreaction (Proposition~\ref{prop:conformal-invariance}),
the leading correction comes from the spatial topology:
the spectral bound
$|\delta C_1| \le 2/(4\pi\lambda_1)$
(where $\lambda_1$ is the spectral gap of the spatial
Laplacian) gives a rigorous bound on the quantum shift
of the stability eigenvalues.
On the Bolza surface ($\lambda_1 \approx 3.84$):
$|\delta C_1| \lesssim 0.05$, which does not shift
$N_{\mathrm{crit}}$ away from~$7$ on the flat plane.
On the modular surface ($\lambda_1 \approx 91$):
$|\delta C_1| \lesssim 0.002$, negligible.
\begin{proposition}[Eigenvalue non-crossing at finite $c$]
\label{prop:non-crossing}
For $N \ge 6$ and $c \ge N^2$, the mode-dependent part of the
$O(1/c)$ Virasoro block correction satisfies
\begin{equation}\label{eq:non-crossing}
  \frac{|\delta f_m - \delta f_{m+1}|}{f(m{+}1,N) - f(m,N)}
  = \frac{2m(N{-}m{-}1) + (N{-}1)}{2c}
  \;\le\; \frac{1}{4} - \frac{1}{2N^2}
  \;<\; \frac{1}{4}.
\end{equation}
In particular, no eigenvalue crossing occurs and the critical mode
index $m_{\mathrm{crit}} = \lfloor N/2\rfloor$ is preserved.
\end{proposition}

\begin{proof}
The leading $1/c$ correction to the Casimir is
$\delta f_m = -f_m^2/c = -m^2(N{-}m)^2/(4c)$.
The difference factors as
$\delta f_m - \delta f_{m+1}
= (N{-}2m{-}1)\bigl[2m(N{-}m{-}1)+(N{-}1)\bigr]/(4c)$.
Dividing by the Casimir gap $f(m{+}1)-f(m) = (N{-}2m{-}1)/2$
gives the ratio $R(m) = [2m(N{-}m{-}1)+(N{-}1)]/(2c)$.
The maximum over $1 \le m < N/2$ is at $m = N/2 - 1$, giving
$R_{\max} = (N^2{-}2)/(4N^2) = 1/4 - 1/(2N^2) < 1/4$.
Since $R < 1$ for all adjacent pairs: the residual gap is at least
$3(N{-}2m{-}1)/8 > 0$, and no reordering occurs.
\end{proof}

\begin{remark}[Algebraic field preservation]
\label{rmk:field-preservation}
The Zamolodchikov $c$-recursion gives the correction as a
\emph{rational function} of~$c$, with only
$\mathbb{Z}_N$-invariant channels contributing.
At $c = N^2$: the palindromic coefficients $A(c), B(c)$ remain in
$\mathbb{Q}(\cos(2\pi/N))$---the same number field as at $c = \infty$.
The palindromic form $A\xi^2 + B\xi + A = 0$ is exact at all~$c$
(the conformal inversion $\xi \leftrightarrow 1/\xi$ is a geometric
symmetry of $\mathbf{H}^2$, not a perturbative artefact).
\end{remark}

The palindromic threshold $\xi^*(N)$ is the critical AdS
radius at which the polygon--BTZ transition occurs, and
the algebraic number field $\mathbb{Q}(\sqrt{D})$ of the
threshold is the arithmetic invariant of the phase boundary.

\begin{proposition}[Orbifold--Havelock correspondence]
\label{prop:orbifold-havelock}
The Havelock eigenvalue on $\mathbf{H}^2$ is exactly the
$\mathbb{Z}_N$ orbifold CFT ground-state energy at
$c = 12N^2$ with angular impulse constraint:
\begin{equation}\label{eq:orbifold-havelock}
  \lambda_m = -E_0(m;\,c{=}12N^2) + \mu_L(\xi),
\end{equation}
where $E_0(m) = -N/2 + m(N{-}m)/2$ is the twisted-sector
ground state and $\mu_L(\xi) = C_1(\xi) - N/2$ is the
Lagrange multiplier.  The central charge $c = 12N^2$ is determined by matching the
mode-dependent coefficient $c/(24N^2) = 1/2$; all curvature
dependence enters through $\mu_L$ alone.
The $N^2$ scaling follows from the `t~Hooft structure of the
Hamiltonian ($N(N{-}1)/2 \sim N^2$ pairwise terms); the
factor $12 = 24/2$ converts between Virasoro and Havelock
normalisations.
\end{proposition}

\begin{proof}
The orbifold ground-state energy
$E_0(m) = -c/(24N) + cm(N{-}m)/(24N^2)$ at $c = 12N^2$
gives $E_0 = -N/2 + m(N{-}m)/2$.
Then $-E_0(m) + \mu_L = [N/2 - m(N{-}m)/2] + [C_1(\xi) - N/2]
= C_1(\xi) - m(N{-}m)/2 = \lambda_m$.
\end{proof}

\begin{proposition}[Equivariant Riemann--Roch]
\label{prop:equiv-rr}
The $\mathbb{Z}_N$-equivariant spectral index of the vortex
Laplacian on the cusped Poincar\'e disk satisfies
\begin{equation}\label{eq:equiv-rr}
  \mathrm{ind}_m = \underbrace{\frac{m(N{-}m)}{2}}_{c_1\text{-term}}
    + \underbrace{b(N)}_{\text{Todd term}}
    + \delta_m,
\end{equation}
where $b(N) = N(N{+}1)/12 - \log 2 + \log N/(N{-}1)$
is independent of both~$m$ and~$\rho$ (verified to~$10^{-16}$
for $N = 3,\ldots,24$).
The central charge $c = 12\,b(N) = N^2 + O(N)$ is the \emph{same}
for every mode~$m$, extracted from the Todd correction alone.
\end{proposition}

\begin{proof}
The Havelock eigenvalue satisfies $\lambda_m + f_m = C_1(\rho) + \delta_m$,
and $C_1(\rho) = \log(2\sinh\rho) + b(N) + O(e^{-\rho})$ (the radial
dependence is absorbed by the Green's function).
Subtracting $\log(2\sinh\rho)$:
$\mathrm{ind}_m := \lambda_m + f_m - \log(2\sinh\rho) = b(N) + \delta_m$.
Since $\delta_m$ is traceless ($\sum\delta_m = 0$ to $10^{-15}$)
and palindromic ($\delta_m = \delta_{N-m}$), the mode-independent
part $b(N)$ is extracted exactly.
\end{proof}

\noindent
The three-layer decomposition
$\lambda_m = C_1 - f_m + \delta_m$
is the \emph{spectral equivariant Riemann--Roch theorem}:
$f_m$ is the $c_1$-contribution (twist-field dimension),
$b(N)$ is the Todd-class correction (vacuum energy $\approx c/12$),
and $\delta_m$ is the one-loop remainder (the Weyl anomaly).
The central charge is not a fitted parameter: it is $c = 12\,b(N)$,
determined by the index theorem.

\begin{remark}[Honest negatives and the $B_2$ boundary]
\label{rmk:honest-negatives}
The connection between the aliasing $\langle D\rangle = N(N{+}1)/12$ and
the Virasoro central charge $c/12$ is \emph{structural} (both arise
from $B_2 = 1/6$) but not \emph{algebraic}: the Havelock modes do
not satisfy Virasoro commutation relations at finite~$N$, and the

The higher-order mode coupling is $W_N$ (quartic: real, first
contributing order; cubic: purely imaginary, a Berry phase), with
effective central charge $c_{\mathrm{eff}} \approx N$ (one free boson
per vortex).
\end{remark}

\noindent
The value $c = 12N^2$ is not merely a matching condition; it
makes five independently testable predictions beyond the Casimir:
(i)~the exact offset $b(N) = N(N{+}1)/12 - \log 2 + \log N/(N{-}1)$
(verified to~$10^{-16}$, Paper~I);
(ii)~the mean aliasing $\langle D\rangle = N(N{+}1)/12$ (exact);
(iii)~the growth law $\rho^*/f_{\mathrm{crit}} \to 1/3$ (Paper~I);
(iv)~the quartic coupling $Q/f_{\mathrm{crit}}^2 = N/48$ (exact for even~$N$);
(v)~the mode-independent Laplacian $\Delta_{\mathbf{H}^2} h = -1/2$.
All five follow from $c = 12N^2$ through the $B_2$-tower:
the Casimir $m(N{-}m)/2$ is $O(c)$; the Weyl anomaly $\delta_m$
is $O(c^0)$ (topological); the Virasoro block correction is

the vortex system admits a second Poisson structure (the Casimir
bracket) whose recursion eigenvalues are $f_m = m(N{-}m)/2$.

with the physical derivatives
$d_{2k} = \partial^{2k}H/\partial a^{2k} = \alpha_k N f_m^k$

The cubic Hamiltonian is purely imaginary (a Berry phase),

($H_4^{\mathrm{diag}} \propto I_2$), guaranteeing conservation

The three-layer decomposition $\lambda_m = C_1 - f_m + \delta_m$
is the truncation of this hierarchy at level~$1$.

On arithmetic surfaces (Bolza, modular, etc.), the eigenvalue
acquires a topological correction that is invisible to the
confining potential $C_1(\xi)$:
\begin{equation}\label{eq:three-layer}
  \lambda_m = \underbrace{C_1(\xi)}_{\text{Ricci scalar}}
    - \underbrace{\frac{m(N{-}m)}{2}}_{\text{Casimir}}
    + \underbrace{\delta_m(N)}_{\text{Weyl anomaly}}.
\end{equation}
Each layer has a distinct character:
\begin{enumerate}
\item $C_1(\xi)$: continuous, surface-dependent, mode-independent.
  Scales approximately as $C_1 \propto \log(\xi)$ over the range

  $C_1 = (N{-}1)(1{+}\xi^2)/(1{-}\xi)^2$.
\item $f(m,N) = m(N{-}m)/2$: discrete, universal (surface-independent).
  The $\mathbb{Z}_N$ Casimir from the Havelock identity.
\item $\delta_m(N)$: topological, UV-finite, $\xi$-independent
  ($\text{variation} < 10^{-13}$), and exactly traceless
  ($\sum_{m=1}^{N-1} \delta_m = 0$ to $10^{-15}$).
  Carries all number-theoretic content: Hecke eigenvalues,
  palindromic structure, Langlands data.
\end{enumerate}
By structural analogy with the decomposition of the Riemann
curvature tensor: $C_1$ plays the role of the Ricci scalar,
$f(m,N)$ the traceful part, and $\delta_m$ the traceless Weyl-like
anomaly.
In AdS$_3$/CFT$_2$, the Brown--Henneaux scaling
$C_1 \propto \log(\xi)$ says that
the Weyl-like anomaly $\delta_m$ is the topological correction
that carries the arithmetic of the underlying surface, contributing
corrections of order $\|\delta\|/C_1 \sim 2\%$ at typical parameter values.

The anomaly $\delta_m$ has two independent origins, which the
parameter space $(\Delta, \xi)$ separates cleanly:
\begin{enumerate}
\item \emph{Interaction anomaly} ($\Delta > 0$ on $\mathbb{R}^2$):

\item \emph{Geometric anomaly} ($\Delta = 0$ on $\mathbf{H}^2$):

  Small ($\|\delta\|^2 = O(0.01$--$100)$) but carries all

\end{enumerate}
The RG flow ($\Delta \to 0$) strips away the large, trivial
interaction anomaly, leaving behind the small geometric anomaly
as the residue that survives at the conformal fixed point.
The number theory is not emergent from the flow; it is a property
of the conformal fixed point, activated by curvature.
The three-layer structure has a symplectic origin: the Kirillov
form $\omega_{\mathrm{KR}}$ (mode-independent) and the Weil--Petersson

\subsection{Higher-dimensional extension: the CFT$_3$/AdS$_4$ Casimir}
\label{sec:cft3}

The two-dimensional results extend to conformal field theory in
any dimension~$d$ via the power-law interaction
$V(x,y) = |x - y|^{-2\Delta}$, where $\Delta$ is the conformal
dimension. For $\Delta \to 0$: $V \to -\log|x-y|$ (the 2D case).
For $\Delta = 1$: the CFT$_3$ two-point function on $S^2$
(the boundary of AdS$_4$).

\begin{proposition}[Generalised Havelock Casimir]
\label{prop:generalised-casimir}
For $N$ equally-spaced points on a ring on $S^2$,
interacting via $V = |x - y|^{-2\Delta}$,
the angular Hessian eigenvalues factorise as
$\mu_m = g(\theta_0, \Delta) \cdot h(m, N, \Delta)$,
where the universal Casimir is
\begin{equation}\label{eq:gen-casimir}
  h(m, N, \Delta) = \sum_{p=1}^{N-1}
    \frac{1 + 2\Delta\cos^2(\pi p/N)}
         {\sin^{2\Delta+2}(\pi p/N)}\,
    \bigl(1 - \cos\tfrac{2\pi mp}{N}\bigr).
\end{equation}
For $\Delta = 0$: $h(m, N, 0) = 2m(N{-}m)$.
The ratio $h(m)/h(1)$ is independent of $\theta_0$, confirming
that the Havelock decomposition extends to all~$\Delta$.
\end{proposition}

The constrained Hessian on the flat plane with $V = |x - y|^{-2\Delta}$
gives an $N$-dependent stability boundary $N_{\mathrm{crit}}(\Delta)$:

\midrule
$N_{\mathrm{crit}}$ & $7$ & $5$ & $5$ & $5$ & $4$ & $4$ \\
\bottomrule
\end{tabular}
\end{center}

The classical $N_{\mathrm{crit}} = 7$ is an isolated phenomenon

$N_{\mathrm{crit}} = 5$.

\noindent
The hexagon ($N = 6$, Saturn) is a \emph{uniquely two-dimensional}
phenomenon: it is marginally stable at $d = 2$ ($\lambda_{\min} = 1/2$)

The pentagon ($N = 5$) survives in all tested dimensions $d \le 10$.

\begin{proposition}[Soft Havelock decomposition in dimension $d$]
\label{prop:soft-havelock}
For the interaction $V = r^{-(d-2)}$ ($d \ge 3$) on the $N$-gon ring,
the angular Casimir decomposes as
\begin{equation}\label{eq:soft-havelock}
  \sigma_m^{(d)} = a(N,d)\,\frac{m(N{-}m)}{2} + R_m(N,d),
\end{equation}
where $a(N,d)$ is the leading Casimir coefficient (determined by
least-squares fit) and $R_m$ is the residual.
For all tested $N = 5,\ldots,20$ and $d = 3,4,5$: the
residual satisfies $|R_m - R_{m'}| < a\,|f_m - f_{m'}|$ for every
adjacent pair, so the \emph{mode ordering is preserved}.
The critical mode is always $m = \lfloor N/2 \rfloor$ (same as $d=2$),
and the stability threshold has the same palindromic form as the
two-dimensional case, with coefficients that depend on~$d$.
\end{proposition}

\noindent
For $d = 2$ ($s = 0$): $a = 1$ exactly and $R_m = 0$ (the Havelock
identity). This is the \emph{unique} dimension where the Casimir is
an exact polynomial---because $\csc^2(\pi p/N)$ has a closed discrete
Fourier transform ($\sum\csc^2\cdot(1{-}\cos) = 2m(N{-}m)$), while
$\csc^s$ for $s \ne 2$ does not.
The rich algebraic structure of Paper~I (palindromic polynomials,
algebraic number fields, modular forms) is therefore a consequence
of the \emph{uniqueness of the logarithmic interaction} among
power-law potentials.

\begin{remark}[Transcendence of the transition dimensions]
\label{rmk:transcendence}
The critical value $\Delta^*(N)$ where $N_{\mathrm{crit}}$ drops
below~$N$ is expected to be \emph{transcendental} for $N \ge 5$.
The threshold equation involves $\sin^{-2\Delta}(\pi p/N)$ for
different~$p$ with $\mathbb{Q}$-linearly independent logarithms.
For $N = 5$ (two distinct chord distances, ratio $= \varphi$),
the Gelfond--Schneider theorem directly applies: if $\Delta^*$
were algebraic and irrational, $\varphi^{2\Delta^*+2}$ would be
transcendental, contradicting the threshold equation.
For general~$N$, a rigorous proof requires the six-exponential
theorem or Baker's theorem on linear forms in logarithms.
This contrasts with the palindromic thresholds $\xi^*(N)$
on $\mathbb{H}^2$, which lie in explicit algebraic number fields.
\end{remark}

For $N = 5$, the pentagon identity
$\sin(2\pi/5)/\sin(\pi/5) = \varphi$ (the golden ratio)
gives the eigenvalue as
$\lambda_2(\Delta) = P_0(\Delta) + P_1(\Delta)\,\varphi^{-(2\Delta+2)}$,
where $P_0, P_1$ are algebraic functions of $\Delta$.
The transition occurs at $\varphi^{2\Delta^*+2} = P_1/({-P_0})
\approx 25.8$, giving $\Delta^* \approx 2.378$.

\begin{theorem}[RG monotonicity of the Casimir ratio]
\label{thm:rg-monotonicity}
For any $N \ge 4$ and $m_{\mathrm{crit}} = \lfloor N/2 \rfloor$,
the generalised Casimir ratio
$R(\Delta) = h(m_{\mathrm{crit}}, N, \Delta) / h(1, N, \Delta)$
is strictly increasing on $[0, \infty)$, with exact limits
\begin{equation}\label{eq:rg-limits}
  R(0) = \frac{m_{\mathrm{crit}}(N{-}m_{\mathrm{crit}})}{N{-}1},
  \qquad
  R(\infty) = \biggl[\frac{\sin(\pi m_{\mathrm{crit}}/N)}
                          {\sin(\pi/N)}\biggr]^2.
\end{equation}
\end{theorem}

\begin{proof}
By the Chebyshev identity, the mode amplification factor
$f(p) = (1{-}\cos(2\pi m_{\mathrm{crit}} p/N))/(1{-}\cos(2\pi p/N))
= [\sin(\pi m_{\mathrm{crit}} p/N)/\sin(\pi p/N)]^2$
achieves its global maximum at $p = 1$ (and its palindromic twin
$p = N{-}1$):

\emph{Even $N = 2k$:}
$\sin(\pi kp/N) = \sin(\pi p/2) \in \{0, \pm 1\}$, so
$f(p) = 1/\sin^2(\pi p/N)$ for odd~$p$ and $f(p) = 0$ for even~$p$.
Since $\sin(\pi p/N) \ge \sin(\pi/N)$ for $p \ge 2$: $f(p) \le f(1)$.

\emph{Odd $N = 2k{+}1$:}
The identity $\sin(k\pi p/N) = \sin(\pi p/2 - \pi p/(2N))$
(using $k/N = 1/2 - 1/(2N)$) gives explicit formulas:
$f(p) = 1/(4\sin^2(\pi p/(2N)))$ for odd~$p$, and
$f(p) = 1/(4\cos^2(\pi p/(2N)))$ for even~$p$.
Since $f(1) = 1/(4\sin^2(\pi/(2N)))$:
for odd $p \ge 3$, $\sin(\pi p/(2N)) \ge \sin(\pi/(2N))$
(monotonicity of $\sin$ on $(0,\pi/2)$), so $f(p) \le f(1)$;
for even $p \le N{-}1$, $\cos(\pi p/(2N)) \ge \sin(\pi/(2N))$
(since $\pi p/(2N) + \pi/(2N) \le \pi/2$ iff $p \le N{-}1$),
so $f(p) \le f(1)$.

The normalised weight $\mu_p(\Delta) \propto \tilde{w}(p,\Delta)\,c_1(p)$
concentrates on $\{1, N{-}1\}$ as $\Delta \to \infty$, because
$\tilde{w}(p)/\tilde{w}(1) = O([\sin(\pi/N)/\sin(\pi p/N)]^{2\Delta})
\to 0$ for $p \notin \{1, N{-}1\}$.
Since $R(\Delta) = \mathbb{E}_\mu[f]$ and $\mu$ concentrates at the
global maximum of~$f$:
$dR/d\Delta = \mathrm{Cov}_\mu(d\log\tilde{w}/d\Delta,\, f) > 0$
(both $f$ and $d\log\tilde{w}/d\Delta$ are maximised at $p = 1$,
giving positive covariance).
\end{proof}

\begin{corollary}[Stability c-theorem analogue]
\label{cor:c-theorem}
$N_{\mathrm{crit}}(\Delta)$ is non-increasing in $\Delta$: once a
polygon type loses stability, it never regains it.
The ``polygon entropy'' $S(\Delta) = \log N_{\mathrm{crit}}(\Delta)$
is a monotonically non-increasing function of the conformal dimension,
flowing from $S(0) = \log 7$ (the Havelock IR fixed point)
to $S(\infty) = \log 3$ (the triangular UV fixed point).
\end{corollary}

\subsection{Triangle universality and dynamical triangulation}
\label{sec:triangle}

The equilateral triangle ($N = 3$) is the \emph{unique} regular
polygon that is dynamically stable for all interaction ranges
and quantum deformation levels.

\begin{theorem}[Triangle universal stability]
\label{thm:triangle-universality}
The equilateral triangle ($N = 3$) is the unique regular polygon
that is stable on the flat plane for all $\Delta \ge 0$,
all TQFT levels $k \ge 3$, and whose tessellation
$\{3,6\}$ is the unique stable regular lattice for all $\Delta > 0$.
\end{theorem}

\begin{proof}[Proof sketch]
Three independent arguments:
\begin{enumerate}
\item \emph{Equal chords.} The triangle is the unique regular polygon
  where all chord distances are equal
  ($\sin(\pi/3) = \sin(2\pi/3) = \sqrt{3}/2$).
  No range-dependent competition between nearest and next-nearest
  neighbors $\implies$ stability is $\Delta$-independent.

\item \emph{Single mode.} The triangle has one nontrivial $\mathbb{Z}_3$
  mode ($m = 1$, with $m = 2$ its palindromic conjugate).
  The Havelock eigenvalue $\lambda_1 = (N{-}1) - 1 = 1 > 0$.
  Mode competition requires at least two independent modes ($N \ge 4$).

\item \emph{Zero Weyl anomaly.}
  The anomaly $\delta_m$ is traceless: $\sum \delta_m = 0$.
  With one mode: $\delta_1 = 0$ identically.
  Verified: anomaly norm $= 2.2 \times 10^{-16}$ (machine zero) on
  the square torus.
\end{enumerate}
For the tessellation claim: the dynamical matrix (phonon spectrum)
of the triangular lattice under $V = r^{-2\Delta}$ has all eigenvalues
positive for \emph{every} $\Delta > 0$ (verified by lattice sums with
convergence study and Richardson extrapolation).
The minimum phonon frequency is at the $K$-point of the Brillouin zone.
The square lattice (coordination~$4$) has a permanent $X$-point
instability (stripe mode: $\omega^2(X) < 0$ for all~$\Delta$);
the hexagonal lattice (coordination~$3$) has a permanent $K$-point
instability. The triangular lattice is the \emph{only} regular
lattice that is ever stable, at any interaction range.
\end{proof}

\begin{remark}[Dynamical triangulation]
\label{rmk:cdt}
In causal dynamical triangulations (CDT) and Regge calculus,
spacetime is discretised into simplices (triangles in~2D).
This triangulation is imposed by hand as a regularisation.
Theorem~\ref{thm:triangle-universality} suggests it may be imposed
by physics: among regular lattices, the triangular lattice is the
unique stable configuration, so the gravitational path integral
$Z = \sum_{\text{tilings}} e^{-S}$ may be dominated by
triangulated configurations---the square and hexagonal alternatives
are saddle points with negative phonon modes at all~$\Delta$.
The extension to irregular triangulations (as used in CDT)
remains open.
\end{remark}
