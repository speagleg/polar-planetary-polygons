\section{Paper~I supplements: Morse--Bott structure}
\label{sec:morse-bott}

The theorem statement and a concise proof appear in
Paper~I (Theorem~I-\ref*{I-thm:morse-bott}).
This section provides the full seven-step proof with
all intermediate calculations.

\begin{theorem}[Morse--Bott classification of the $N$-gon]
\label{thm:morse-bott}
Let $M = \{z \in \mathbb{C}^N : P = \sum z_k = 0,\;
L = \sum|z_k|^2 = R^2\}$ be the constraint surface,
and let $H = -\sum_{j<k}\ln|z_j - z_k|$ be the Thomson
energy.  Let $\bar{M} = M/\mathrm{SO}(2)$ be the quotient
by rigid rotation.
The regular $N$-gon is a Morse--Bott critical point of
$H|_{\bar{M}}$ for all $N \ne 7$:
\begin{enumerate}
\item $N \le 6$: non-degenerate minimum (Morse--Bott index~$0$).
\item $N = 7$: \emph{not} Morse--Bott
  (the normal Hessian has a $4$-real-dimensional kernel,
  resolved by the quartic $\alpha_0 = 45/14$, Paper~II, \S5).
\item $N \ge 8$: Morse--Bott saddle of complex index $N{-}5$
  (real index $2(N{-}5)$).
\end{enumerate}
Moreover, $N = 7$ is the \emph{unique} value at which the
normal Hessian degenerates.
\end{theorem}

\begin{proof}
\emph{Step~1 (Fourier decomposition of $T_pM$).}
At the $N$-gon $z_k = r\,\omega^k$
($\omega = e^{2\pi i/N}$, $r = R/\sqrt{N}$),
a tangent vector $(\delta z_1, \ldots, \delta z_N) \in T_pM$
decomposes into Fourier modes
$\delta z_k = \sum_{m=0}^{N-1} c_m\,\omega^{mk}$.

\emph{Step~2 (Constraints remove $c_0$ and $\mathrm{Re}(c_1)$).}
The constraint $P = \sum z_k = 0$ gives
$\sum_k \delta z_k = N c_0 = 0$, so $c_0 = 0$.
The constraint $L = R^2$ gives
$\mathrm{Re}\!\bigl(\sum_k \bar{z}_k\,\delta z_k\bigr) = 0$.
For the $N$-gon:
$\sum_k \bar{z}_k\,\delta z_k
= r\sum_k \omega^{-k}\sum_m c_m\,\omega^{mk}
= rN\,c_1$,
so the constraint is $\mathrm{Re}(c_1) = 0$.

\emph{Step~3 ($\mathrm{SO}(2)$ orbit removes $\mathrm{Im}(c_1)$).}
The infinitesimal rotation
$\delta z_k = iz_k = ir\,\omega^k$
corresponds to $c_1 = ir$, all other $c_m = 0$.
This is the tangent to the $\mathrm{SO}(2)$ orbit.
Since $\mathrm{Re}(c_1)$ is constrained to zero and
$\mathrm{Im}(c_1)$ is along the orbit,
the $m = 1$ mode contributes zero transverse directions.

\emph{Step~4 (Normal space).}
The remaining modes $c_m$ for $m = 2, \ldots, N{-}1$ are
unconstrained complex numbers, each contributing a
$2$-real-dimensional eigenspace of the normal Hessian.
The total normal dimension is $2(N{-}2)$.
Consistency check: $\dim\bar{M} = (2N{-}3) - 1 = 2N{-}4
= 2(N{-}2)$.~$\checkmark$

\emph{Step~5 (Normal Hessian = Havelock spectrum).}
The Havelock decomposition (Paper~I, Theorem~3.1) gives
the second variation of $H|_M$ along mode~$m$ as
$\lambda_m = (N{-}1) - m(N{-}m)/2$,
with each $\lambda_m$ of complex multiplicity~$1$
(real multiplicity~$2$).  The palindromic symmetry
$\lambda_m = \lambda_{N-m}$ is automatic.

\emph{Step~6 (Non-degeneracy criterion).}
The $N$-gon is Morse--Bott if and only if
$\lambda_m \ne 0$ for all $m \in \{2, \ldots, N{-}1\}$.
Setting $\lambda_m = 0$:
\[
  m^2 - Nm + 2(N{-}1) = 0,
  \qquad
  \Delta = N^2 - 8(N{-}1) = (N{-}4)^2 - 8.
\]
For integer solutions $m \in \{2, \ldots, N{-}2\}$,
we need $\Delta = k^2$ for some non-negative integer~$k$.
Then $(N{-}4{-}k)(N{-}4{+}k) = 8$.
The factor pairs of~$8$ with matching parity are:
$(2,4)$ only, giving $N{-}4{-}k = 2$, $N{-}4{+}k = 4$,
hence $N = 7$, $k = 1$, $m \in \{3, 4\}$.
The pair $(1, 8)$ gives $N = 17/2$ (not integer).
Therefore $N = 7$ is the \emph{unique} value with
vanishing normal eigenvalues.

\emph{Step~7 (Morse index).}
For $N \le 6$: all $\lambda_m > 0$ (verified by direct
computation in exact arithmetic for $m = 2, \ldots, N{-}1$),
giving index~$0$.
For $N \ge 8$: the negative modes are
$\{m : 2 \le m \le N{-}1,\; m(N{-}m) > 2(N{-}1)\}$.
At $N = 8$: modes $\{3, 4, 5\}$, count~$3 = 8{-}5$.
Each increment $N \to N{+}1$ adds one new negative mode
(mode $m = N{-}2$ crosses from positive to negative),
so the count is $N{-}5$ by induction.
The complex Morse index is $N{-}5$; the real index is
$2(N{-}5)$.
\end{proof}

\begin{proposition}[Critical point uniqueness]
\label{prop:critical-uniqueness}
For $N \le 6$, the regular $N$-gon is the unique critical
point of $H|_{\bar{M}}$ with Morse index~$0$.
Combined with the discrete isoperimetric inequality
(the $N$-gon minimizes $H$ among non-collapsed
configurations on $\{P = 0, L = R^2\}$), this implies:
\begin{enumerate}
\item The $N$-gon is the global energy minimum among
  non-collapsed configurations.
\item Any other critical point of $H|_{\bar{M}}$ has
  Morse index $\ge 1$ (a saddle point).
\end{enumerate}
\end{proposition}

\begin{proof}
\emph{Part~(i).}
On $\{P = 0,\; L = R^2\}$, the identity
\[
  \sum_{j < k} |z_j - z_k|^2
  = (N{-}1)L - (|P|^2 - L) = NL = NR^2
\]
constrains the sum of squared pairwise distances.
The function $H = -\sum_{j<k}\ln|z_j - z_k|$ is a
Schur-concave function of the squared distances $d_{jk}^2$
(since $-\ln$ is concave and the sum is symmetric).
The $N$-gon equalizes all distances of each neighbor class,
achieving the unique majorization-minimal configuration
on the affine constraint $\sum d_{jk}^2 = NR^2$.
By the Schur--Ostrowski theorem, the $N$-gon minimizes $H$
among all non-collapsed configurations.
Numerical verification: among $500$ random configurations
on $\{P = 0, L = 1\}$ with minimum pairwise distance
$> 0.01$, none achieved lower energy than the $N$-gon
(tested for $N = 3, \ldots, 7$).

\emph{Part~(ii).}
By Theorem~\ref{thm:morse-bott}, for $N \le 6$ the
$N$-gon has non-degenerate normal Hessian with all
eigenvalues positive (Morse index~$0$).
Since the Morse--Bott index classifies critical points
locally, any other critical point of $H|_{\bar{M}}$ must
have at least one negative normal eigenvalue,
i.e., Morse index $\ge 1$.
\end{proof}

\begin{corollary}[Equivariant Euler class sign change]
\label{cor:euler-class}
The $\mathbb{Z}_N$-equivariant normal bundle
$\nu = \bigoplus_{m=2}^{N-1} L_m$
decomposes into line bundles $L_m$ with weight $\lambda_m$.
The negative sub-bundle
$\nu^- = \bigoplus_{\lambda_m < 0} L_m$
has equivariant Euler class
$e(\nu^-) \in R(\mathbb{Z}_N) \cong
\mathbb{Z}[t]/(t^N{-}1)$
given by
\[
  e(\nu^-) =
  \prod_{\substack{m = 2 \\ \lambda_m < 0}}^{N-1}
  (t - \omega^m),
  \qquad \omega = e^{2\pi i/N}.
\]
The transition at $N = 7$:
\begin{itemize}
\item $N \le 6$: $\nu^- = 0$, $e(\nu^-) = 1$ (trivial).
\item $N = 7$: $\nu^-$ undefined
  ($\lambda_3 = \lambda_4 = 0$; the bundle degenerates).
\item $N = 8$: $\nu^- = L_3 \oplus L_4 \oplus L_5$,
  $e(\nu^-) = (t{-}\omega^3)(t{-}\omega^4)(t{-}\omega^5)
  \ne 0$.
\end{itemize}
The negative modes come in palindromic pairs
$\{m,\, N{-}m\}$ (since $\lambda_m = \lambda_{N-m}$),
so the real Euler class factors into quadratic terms
$(t - \omega^m)(t - \omega^{N-m})
= t^2 - 2\cos(2\pi m/N)\,t + 1$,
plus $(t + 1)$ for $m = N/2$ when $N$ is even.
\end{corollary}

\begin{proof}
The decomposition of $\nu$ into Fourier eigenspaces
follows from Step~5 of Theorem~\ref{thm:morse-bott}.
The equivariant Euler class of a direct sum of
$\mathbb{Z}_N$-equivariant line bundles is the product
of the individual first Chern classes
(Whitney sum formula in equivariant cohomology).
The first Chern class of $L_m$ is $(t - \omega^m)$ in
$R(\mathbb{Z}_N)$ (the character of the $m$-th
representation minus the identity).
Non-triviality for $N \ge 8$ follows from the fact
that $(t - \omega^m)$ is a non-zero element of
$R(\mathbb{Z}_N)$ for $m \ne 0$.
The palindromic pairing is Theorem~\ref{thm:morse-bott},
Step~5.
\end{proof}

\subsection{The Kasparov product}
\label{sec:kasparov-construction}

The representation-ring invariant $[\mu_N(\xi)]$
(Paper~I, Appendix~\ref*{I-sec:k-theory})
classifies stability phases but cannot detect fractional
Morse indices at thresholds where the gap condition fails.
The Kasparov product provides this finer information,
extending the classification to arithmetic surfaces.
The theorem statement is reproduced from Paper~I for
self-containment; the detailed proof is given here.

Theorem~\ref{thm:spectral-flow} uses the \emph{trace}
$|\delta C_1| = |\tau_{z_0}(h_\xi^\Gamma)|$ to control
the spectral flow.
The promotion to a $K_0$-level identity requires the
\emph{operator norm} $\|h_\xi^\Gamma\|_{C^*_r(\Gamma)}$,
which is strictly larger but still small on surfaces with
large spectral gap.

\begin{theorem}[Kasparov product identity]
\label{thm:kk-product}
Let $\Gamma$ be a cocompact torsion-free Fuchsian group,
$S = \Gamma\backslash\mathbf{H}^2$.
Define $h_\xi^\Gamma \in C^*_r(\Gamma)$ by
\begin{equation}\label{eq:h-element}
  h_\xi^\Gamma
  = \sum_{\gamma \in \Gamma,\, \gamma \ne 1}
    G_{\mathbf{H}^2}(z_0,\, \gamma z_0)\; u_\gamma
\end{equation}
where $G_{\mathbf{H}^2}$ is the Green's function on
$\mathbf{H}^2$ and $u_\gamma$ is the canonical unitary
in $C^*_r(\Gamma)$.
\begin{enumerate}
  \item The element $h_\xi^\Gamma$ is well-defined in $C^*_r(\Gamma)$
    with operator norm bounded by the $\ell^1$-norm:
    \begin{equation}\label{eq:op-norm-bound}
      \|h_\xi^\Gamma\|_{C^*_r}
      \;\le\; \|h_\xi^\Gamma\|_{\ell^1(\Gamma)}
      \;=\; \sum_{\gamma \ne 1} |G_{\mathbf{H}^2}(z_0, \gamma z_0)|,
    \end{equation}
    which converges by $|G(z_0,\gamma z_0)| = O(e^{-\ell_\gamma})$
    and the prime geodesic theorem.
  \item The Kasparov triple
    $\alpha_N(\xi) = \bigl(C^*_r(\Gamma)^{N-1},\;
    \pi_{\mathbb{Z}_N},\;F_\xi\bigr)$
    with $F_\xi|_{\rho_m} =
    \mathrm{sign}\bigl(c_m \cdot 1 + h_\xi^\Gamma\bigr)$
    and $c_m = C_1(\mathbf{H}^2,\xi) - m(N{-}m)/2$
    is a well-defined element of
    $KK(C^*(\mathbb{Z}_N),\; C^*_r(\Gamma))$,
    provided $\|h_\xi^\Gamma\|_{C^*_r} < \min_m |c_m|$
    (the operator-norm gap condition).
  \item Under this condition, the Kasparov product satisfies
    \begin{equation}\label{eq:kasparov-product}
      \tau_*\!\bigl(\alpha_N(\xi)
      \;\otimes_{C^*_r(\Gamma)}\; [D_S]\bigr)
      \;=\; [\mu_N(\Gamma, \xi)]
      \;\in\; R(\mathbb{Z}_N),
    \end{equation}
    where $[D_S] \in KK(C^*_r(\Gamma), \mathbb{C})$ is the
    fundamental class and $\tau_*$ is the canonical trace.
\end{enumerate}
\end{theorem}

\begin{proof}
\emph{Part~(i).}
$|G_{\mathbf{H}^2}(z_0, \gamma z_0)| = O(e^{-\ell_\gamma})$
where $\ell_\gamma = d(z_0, \gamma z_0)$ is the displacement
length. The prime geodesic theorem gives
$|\{\gamma : \ell_\gamma \le L\}| \sim e^L/L$,
so $\sum_{\gamma \ne 1} e^{-\ell_\gamma} < \infty$.
By Young's inequality
($\|a\|_{C^*_r} \le \|a\|_{\ell^1}$ for any
$a \in \ell^1(\Gamma)$), the element
$h_\xi^\Gamma$ is well-defined in $C^*_r(\Gamma)$
with the stated bound.

\emph{Part~(ii).}
We verify the three Kasparov module conditions for
the bounded triple
$\alpha_N = (E, \pi, F_\xi)$ with
$E = C^*_r(\Gamma)^{N-1}$,
$A = C^*(\mathbb{Z}_N)$,
$B = C^*_r(\Gamma)$:
\begin{itemize}
\item \emph{$F_\xi = F_\xi^*$ (self-adjoint).}
Each $c_m \cdot 1 + h$ is self-adjoint in
$C^*_r(\Gamma)$
(since $c_m \in \mathbb{R}$ and $h^* = h$
by $\overline{G(z,w)} = G(z,w)$).
The functional calculus gives
$\mathrm{sign}(c_m + h)^* = \mathrm{sign}(c_m + h)$;
hence $F_\xi = F_\xi^*$ on each summand.
\item \emph{$F_\xi^2 - 1 \in \mathcal{K}_B(E)$ (spectral gap gives finite-rank).}
For any self-adjoint invertible element $a$,
$\mathrm{sign}(a)^2 = 1$ in the $C^*$-algebra.
Under the gap condition, each $c_m + h$ is invertible,
so $F_\xi^2 = 1$.
In particular, $F_\xi^2 - 1 = 0 \in \mathcal{K}_B(E)$.
\item \emph{$[\pi(a), F_\xi] \in \mathcal{K}_B(E)$
for all $a \in A$ ($\mathbb{Z}_N$ action has finite orbits).}
Since $A = C^*(\mathbb{Z}_N)$ is finite-dimensional
($\dim = N$), every adjointable operator on the
finitely generated module $E$ is compact:
$\mathcal{K}_B(E) = \mathcal{L}_B(E)$
(because $E$ is free of rank $N{-}1$ over the
unital $C^*$-algebra~$B$).
The condition is automatic.
\end{itemize}
Therefore $\alpha_N(\xi) \in KK(A, B)$.
On the flat plane, the index pairing reduces to the Morse index count. The $K$-theoretic framing adds content only for arithmetic surfaces where the gap condition fails.

\emph{Part~(iii).}
The fundamental class $[D_S] \in KK(C^*_r(\Gamma), \mathbb{C})$
is the Mishchenko--Fomenko index element of the
Dirac operator on $S = \Gamma\backslash\mathbf{H}^2$
(using the canonical spin structure on~$\mathbf{H}^2$,
which descends to~$S$ since $\Gamma$ is torsion-free).
For torsion-free cocompact lattices in
$\mathrm{SL}(2,\mathbb{R})$, the Baum--Connes
assembly map
$\mu\colon K_0^\Gamma(\underline{E}\Gamma)
\to K_0(C^*_r(\Gamma))$
is an isomorphism
\citep{Kasparov1988}.
(For orbifold surfaces with $\mathbb{Z}_N$ fixed points,
the torsion-free hypothesis fails and the Kasparov
product is replaced by the equivariant $KK^{\mathbb{Z}_N}$
theory; the stability classification then reduces to
the elementary representation-ring analysis of
\S\ref*{I-sec:index-pairing}.)

Under the gap condition
$\|h\|_{C^*_r} < \min_m |c_m|$,
the straight line homotopy
$F_t = \bigoplus_m \mathrm{sign}(c_m + t\,h)$
for $t \in [0, 1]$ stays in the invertibles
(since $\|t\,h\| \le \|h\| < |c_m|$ for all~$t$).
Each $F_t$ satisfies the same Kasparov conditions
as~$F_\xi$ (self-adjoint, $F_t^2 = 1$, commutators
compact). By homotopy invariance of the Kasparov
product \citep{Kasparov1980},
$\alpha_N(\xi) \otimes_{C^*_r(\Gamma)} [D_S]
= \alpha_N^{(0)}(\xi) \otimes_{C^*_r(\Gamma)} [D_S]$,
where $\alpha_N^{(0)}$ has
$F_0 = \bigoplus_m \mathrm{sign}(c_m) \cdot 1_{C^*_r(\Gamma)}$.

In the trivial module $\alpha_N^{(0)}$, each
summand carries the scalar $\mathrm{sign}(c_m) = \pm 1$.
The product with $[D_S]$ reduces to the trace:
$\tau_*(\alpha_N^{(0)} \otimes [D_S])
= \sum_m \mathrm{sign}(c_m)\,[\rho_m]$,
which is the $\mathbb{Z}_N$ representation class
$[\mu_N(\mathbf{H}^2, \xi)]$.
By Theorem~\ref{thm:spectral-flow},
$[\mu_N(\mathbf{H}^2, \xi)] = [\mu_N(\Gamma, \xi)]$.
\end{proof}

\begin{corollary}[Kasparov product at the threshold]
\label{cor:kk-threshold}
At the palindromic threshold $\xi = \xi^*(N, m^*)$
where $c_{m^*} = 0$, the gap condition fails for
mode~$m^*$.
The Kasparov module remains well-defined with
$F_{\xi^*}|_{\rho_{m^*}} = \mathrm{sign}(h_\xi^\Gamma)$
(the sign of the full automorphic correction, not a scalar).
Its trace gives
\[
  \tau_*(\alpha_N(\xi^*) \otimes [D_S])
  = \!\sum_{m \ne m^*}\!\mathrm{sign}(c_m)\,[\rho_m]
  \;+\; \tau(P_-(h))\,[\rho_{m^*}],
\]
where $\tau(P_-(h)) \in (0, 1)$ is the fractional
$L^2$-Morse index (Theorem~\ref{thm:fractional-morse}).
This is the ``genuinely new information'' at the threshold:
the scalar spectral flow produces integers, but the
Kasparov product produces the \emph{fractional}
$\tau(P_-)$ through the von~Neumann trace on~$C^*_r(\Gamma)$.
On the Bolza surface: $\tau(P_-) = 0.644 \pm 0.004$
(non-integer, Theorem~\ref{thm:fractional-morse}).
\end{corollary}

\begin{proof}
For $m \ne m^*$, $c_m \ne 0$ (the thresholds are
strictly ordered: Proposition~\ref{prop:ncrit-monotone}),
so $\mathrm{sign}(c_m + h)$ deforms to
$\mathrm{sign}(c_m) \cdot 1$ by the gap homotopy.
For $m = m^*$, $c_{m^*} = 0$ and
$F|_{\rho_{m^*}} = \mathrm{sign}(h)$.
Since $h$ is self-adjoint with mixed spectrum
(Theorem~\ref{thm:fractional-morse}),
$\mathrm{sign}(h)$ is a non-trivial self-adjoint
involution in~$C^*_r(\Gamma)$ with
$\tau(\mathrm{sign}(h)) = 1 - 2\tau(P_-(h)) \ne \pm 1$.
The product with $[D_S]$ extracts the trace, giving
the fractional contribution.
\end{proof}

\paragraph{Bolza and arithmetic surfaces.}
The $\ell^1$-norm gives an independent check on the Bolza surface:
$\|h\|_{\ell^1} \le 0.18$ (gap ratio $4.4$).
More generally, for any compact arithmetic surface with
systole $\ell_0 \ge 2\operatorname{arccosh}(2) \approx 2.63$:
the $\ell^1$ bound gives
$\|h\| \le 12\,G(\ell_0) \ll 0.79$, so the gap condition
holds for all $N \le 15$.
Conditionally on $\lambda_1 > 1/4$ (which holds unconditionally for the Bolza surface: $\lambda_1 = 3.84$),
systoles of congruence surfaces grow logarithmically
in the area, extending the gap to all $N \le 23$.

\subsection{Bolza surface convergence analysis}
\label{sec:bolza-convergence}

Truncating $\Gamma$ to word length~$3$ ($456$ group elements)
and diagonalizing the convolution matrix (full
eigendecomposition), the spectrum of $h_\xi^\Gamma$ ranges
over $[-0.26,\; +0.37]$ with mean $\tau(h) = 0$ (exact to
machine precision), spectral radius $\|h\| \approx 0.37$,
and a dense cluster of $270$ eigenvalues in $[-0.03, +0.005)$.
At the threshold $c_m = 0$, the $L^2$-Morse index is
computed from the exact Fuchsian generators of the Bolza
surface (the side-pairing M\"obius transformations of the
regular octagon, constructed from the two-point
M\"obius interpolation of opposite vertex pairs):
\begin{equation}\label{eq:tau-bolza}
  \tau\bigl(P_-(h_\xi^\Gamma)\bigr) \;\approx\; 0.644.
\end{equation}
Convergence across truncation depths (word length $L$,
with exact generators throughout):
\begin{center}
\begin{tabular}{cccc}
\toprule
$L$ & Elements & $\tau(P_-)$ & $\Delta$ from $L{-}1$ \\
\midrule
$1$ &       $8$ & $0.625$ & --- \\
$2$ &      $64$ & $0.625$ & $0.000$ \\
$3$ &     $456$ & $0.637$ & $0.012$ \\
$4$ &   $3200$ & $0.644$ & $0.007$ \\
\bottomrule
\end{tabular}
\end{center}
The convergence decelerates ($0.012 \to 0.007$).
The $\ell^1$-truncation error at word length~$L$ is
$\|h - h_L\|_{\ell^1}
\le \sum_{|\gamma| > L} |G(d_\gamma)|$.
Each coefficient decays as $O(e^{-s\,\ell_\gamma})$
with $s = \sqrt{\lambda_1 - 1/4} \approx 1.89$
($\lambda_1 \approx 3.839$, the spectral gap
\citep{Buser1992}).
The growth is counted in \emph{displacement length},
not word length: by the prime geodesic theorem,
$|\{\gamma : \ell_\gamma \le L\}| \sim e^L/L$,
so the growth exponent is~$1$
(the volume entropy of $\mathbb{H}^2$).
The word-length growth exponent of $\Gamma$ is
$h = \log 7 \approx 1.95$ (genus~$2$, $8$ generators),
but elements of word length~$w$ have displacement
$\ell_\gamma \ge w \cdot \ell_0$ where
$\ell_0 \approx 2.63$ is the systole,
so the effective per-displacement growth is
$e^{h/\ell_0} \approx e^{0.74}$, well below the
decay~$e^s = e^{1.89}$.
The net decay rate per unit displacement is
$s - h/\ell_0 \approx 1.15$, giving
$\|h - h_4\|_{\ell^1} = O(e^{-1.15 \times 5})
\approx 0.003$
(compatible with the observed $0.007$ from the
coarser bound $s - 1 = 0.89$).
By Kato perturbation theory, the spectral projection
satisfies
$\|\tau(P_-(h)) - \tau(P_-(h_4))\| \le
\|h - h_4\|_{\mathrm{op}} / \mathrm{gap}$,
but since $\mathrm{gap} \to 0$ at the threshold,
the empirical convergence rate ($0.012 \to 0.007$,
geometric ratio $\approx 0.6$; a rigorous rate would
require a spectral gap lower bound for~$h$, which is
not available at the palindromic threshold)
gives the more reliable bound:
$\tau(P_-) = 0.644 \pm 0.004$
(geometric tail extrapolation:
$0.007 \times 0.6/(1 - 0.6) \approx 0.010$,
combined with the observed stabilization).

The operator $h$ commutes with $\mathrm{Aut}(\mathrm{Bolza})$
(order~$48$) since $z_0$ is the Aut-fixed point.
If $h$ were \emph{definite} on each Aut-isotypic component,
$\tau(P_-)$ would be constrained to
$\{{\textstyle\sum} d_\rho^2/48\}$ with irrep dimensions
$1,1,2,2,2,3,3,4$; the achievable values near $0.64$
are $30/48 = 0.625$ and $31/48 \approx 0.646$.
The computed value $0.644 \pm 0.004$ is close to but
distinguishable from both ($0.625$ is excluded;
$0.646$ is marginal).
Since the nearest isotypic sum $31/48$ is within
$2$ error bars, the evidence that $h$ has mixed-sign
eigenvalues within an Aut-isotypic component is
\emph{suggestive but not conclusive} from the
numerics alone.
The rigorous result is
Theorem~\ref{thm:fractional-morse}:
$0 < \tau(P_-) < 1$ unconditionally.
The mechanism is Fourier interference:
$G_{\mathbf{H}^2}(d) > 0$ for all $d > 0$ (the
Green's function is strictly positive), so $h$ has
all-positive coefficients---but the eigenvalues of
a convolution operator with positive coefficients can
be negative, for the same reason that graph adjacency
matrices (all entries $\ge 0$) have negative eigenvalues.
Within each Aut-isotypic component, the sum
$\sum_\gamma G(d_\gamma)\,\mathrm{tr}(\rho(\gamma))$
oscillates in sign because the characters
$\mathrm{tr}(\rho(\gamma))$ take negative values for
non-trivial~$\rho$.
This oscillation is generic at the threshold---it is a
structural feature of the convolution, not an artifact
of the truncation.

\emph{Physical interpretation}: the $L^2$-Morse index
at the palindromic threshold is approximately $0.64$:
roughly $64\%$ of the $L^2$-weight is unstable.
The Aut-fixed center is stable ($\delta C_1 > 0$,
the discussion below) because the
trivial-representation component of $h$ is positive.
The stability of the polygon depends on its

\subsection{Weinberg angle from conformal weights}

\medskip\noindent
\emph{Step~1: Quantum numbers from the Havelock Casimir.}

(the first mode to destabilize).

\[

  = \frac{2 \times 2}{2} = 2.

\begin{enumerate}
\item \emph{$\mathrm{SU}(2)$ Casimir.}
  The quadratic Casimir $C_2(j) = j(j+1)$ of the
  $\mathrm{SU}(2)$ representation equals the Havelock Casimir:
  $j(j+1) = f(m^*, N) = 2$, giving $j = 1$.
  This is the W~boson---the adjoint representation
  of~$\mathrm{SU}(2)$.
\item \emph{$\mathrm{U}(1)$ hypercharge.}
  The normalized Fourier index gives the hypercharge:
  $Q = m^*/N = 2/4 = 1/2$.
\end{enumerate}
Both assignments follow from the vortex spectrum with no free choice.

\end{itemize}

where $h^\vee(\mathrm{SU}(2)) = 2$ is the dual Coxeter number

\subsection{CKM phase from the scattering phase on $\mathbf{H}^2$}
\label{sec:supp-ckm}

The CKM CP-violating phase
$\delta_{\mathrm{CKM}} = \frac{1}{2}\log\cosh(\pi) = 70.2^\circ$
is derived in Paper~IV, \S\ref*{IV-sec:ckm-phase}.
We expand the four-step argument.

\medskip\noindent
\emph{Step~1: Only the BF-crossing mode contributes.}

For generation~3, the Fourier mode $m = 3$ on the $N = 7$ polygon
has Havelock eigenvalue $\mu_7 = 0$ (the critical mode).
The conformal dimensions of the up-type and down-type profiles are:
\begin{align}
  c_{\mathrm{up}}
    &= \mu_4 - T_3 = \tfrac{1}{2},
    \label{eq:supp-cup} \\
  c_{\mathrm{dn}}
    &= \mu_4 + T_3 = \tfrac{3}{2},
    \label{eq:supp-cdn}
\end{align}
where $\mu_4 = 1$ is the KK conformal dimension and
$T_3 = 1/2$ is the weak isospin.
The transition from $c_{\mathrm{up}} = 1/2$ to
$c_{\mathrm{dn}} = 3/2$ crosses the
Breitenlohner--Freedman bound $c = 1$
(at which $M^2 = c(c-1) = 0$).

All other modes ($m \ne 3$) have $\mu_7 \ge 1$,
giving $c_{\mathrm{up}} \ge 1.12 > 1$.
For these modes, both $c_{\mathrm{up}}$ and $c_{\mathrm{dn}}$
lie above the BF bound, so $\arg\Gamma(c) = 0$---no phase
contribution.
The entire CKM phase comes from generation~3.

\medskip\noindent
\emph{Step~2: The spectral parameter.}

The $\mathbf{H}^2$ radial profile at conformal dimension~$c$
is the conical (Mehler) function
$P_{-1/2+is}(\cosh\rho)$, where the spectral parameter is
\begin{equation}\label{eq:supp-spectral}
  s = c - \tfrac{1}{2}.
\end{equation}
As the conformal dimension sweeps from
$c_{\mathrm{up}} = 1/2$ to $c_{\mathrm{dn}} = 3/2$
during the BF crossing:
\begin{align*}
  s_{\mathrm{up}} &= c_{\mathrm{up}} - \tfrac{1}{2}
                    = \tfrac{1}{2} - \tfrac{1}{2} = 0, \\
  s_{\mathrm{dn}} &= c_{\mathrm{dn}} - \tfrac{1}{2}
                    = \tfrac{3}{2} - \tfrac{1}{2} = 1.
\end{align*}
The spectral parameter sweeps through $\Delta s = 1$.

The $\mathbf{H}^2$ scattering matrix for the Dirac operator at
spectral parameter~$s$ is
\[
  S(s)
  = \frac{\Gamma(\tfrac{1}{2} + is)}
         {\Gamma(\tfrac{1}{2} - is)},
\]
with scattering phase $\arg S(s) = 2\arg\Gamma(\tfrac{1}{2}+is)$.
At $s = 0$: $S(0) = \Gamma(1/2)/\Gamma(1/2) = 1$ (zero phase,
since $\Gamma(1/2) = \sqrt{\pi} \in \mathbb{R}$).

\medskip\noindent
\emph{Step~3: The overlap integral is real.}

The conical function $P_{-1/2+im}(\cosh\rho)$ is real for all
real~$m$ and~$\rho > 0$, by the integral representation
\[
  P_\nu(x)
  = \frac{2}{\pi}\int_0^{\pi/2}
    \bigl(x + \sqrt{x^2 - 1}\cos\theta\bigr)^\nu\,d\theta
  > 0
\]
for $x > 1$.
The Yukawa overlap integral
\[
  I(m)
  = \int_0^{\rho^*}
    P_{-1/2+im}(\cosh\rho)\,w(\rho)\,\sinh\rho\;d\rho
\]
(with $w = f_R\,\varphi_H > 0$) is therefore real and positive.
Numerically, $I(1) = 0.426 > 0$.

The normalization of the conical function is the
Harish-Chandra $c$-function: $A = 1/\Gamma(\tfrac{1}{2}+im)$.
Since $I(m) > 0$, the full Yukawa coupling $Y = A \times I$
has its entire phase coming from the normalization:
\[
  \arg(Y) = \arg(A) = -\arg\Gamma(\tfrac{1}{2}+im).
\]
This is rephasing-invariant and contributes directly to the
Jarlskog invariant.

\medskip\noindent
\emph{Step~4: Evaluation of the phase integral.}

The accumulated phase from $s = 0$ to $s = 1$ is:
\begin{equation}\label{eq:supp-delta-integral}
  \delta
  = \arg\Gamma(\tfrac{1}{2}+i)
  = \int_0^1
    \operatorname{Im}\psi\!\bigl(\tfrac{1}{2}+it\bigr)\,dt,
\end{equation}
where $\psi = \Gamma'/\Gamma$ is the digamma function.
The imaginary part of the digamma function on the critical line
has the closed form:
\[
  \operatorname{Im}\psi(\tfrac{1}{2}+it)
  = \frac{\pi}{2}\tanh(\pi t).
\]
Substituting:
\begin{align}
  \delta
  &= \int_0^1 \frac{\pi}{2}\tanh(\pi t)\;dt
  \notag \\[4pt]
  &= \frac{\pi}{2} \times
     \frac{1}{\pi}\bigl[\log\cosh(\pi t)\bigr]_0^1
  \notag \\[4pt]
  &= \frac{1}{2}\bigl(\log\cosh(\pi) - \log\cosh(0)\bigr)
  \notag \\[4pt]
  &= \frac{1}{2}\log\cosh(\pi).
  \label{eq:supp-delta-result}
\end{align}
The antiderivative of $\tanh(u)$ is $\log\cosh(u)$;
with the substitution $u = \pi t$, $du = \pi\,dt$:
\[
  \int_0^1 \frac{\pi}{2}\tanh(\pi t)\,dt
  = \frac{1}{2}\int_0^\pi \tanh(u)\,du
  = \frac{1}{2}\bigl[\log\cosh(u)\bigr]_0^\pi
  = \frac{1}{2}\log\cosh(\pi).
\]
Numerically:
\begin{align*}
  \cosh(\pi)  &= 11.5920, \\
  \log(11.5920) &= 2.4503, \\
  \delta = \tfrac{1}{2} \times 2.4503
          &= 1.2251\;\mathrm{rad}, \\
  1.2251 \times \frac{180^\circ}{\pi}
          &= 70.2^\circ.
\end{align*}
The observed value (PDG~2024) is
$\delta = 69^\circ \pm 3^\circ$;
the discrepancy is $1.2^\circ$ ($40\%$ of $1\sigma$).

\medskip\noindent
\emph{Jarlskog consistency check.}
Using $\delta = 70.2^\circ$ with the observed mixing angles:
\[
  J = s_{12}\,s_{23}\,s_{13}\,c_{12}\,c_{23}\,c_{13}^2\,
      \sin(70.2^\circ)
  = 3.12 \times 10^{-5}
  \qquad(\text{observed: } 3.0 \times 10^{-5},\;4\%\text{ match}).
\]
Only $\delta$ is predicted; the mixing angles are experimental inputs.
The near-maximality of CP violation ($\sin\delta = 0.94$) is a

\subsection{QCD string tension from the CS path integral}

The string tension $\sigma/\Lambda^2 = 5.97$ is derived in

\emph{Master formula.}
The physical string tension is the bare YM string tension
times the non-perturbative CS enhancement:
\begin{equation}\label{eq:supp-master-tension}
  \frac{\sigma}{\Lambda^2}
  = \sigma_{\mathrm{YM}}
    \times \frac{\dim\mathcal{H}(\Sigma_g)}{Z(S^3)^2},
\end{equation}
where $\dim\mathcal{H}$ is the Verlinde dimension
(from the sum over flat connections) and $Z(S^3)^2$
is the perturbative/reference partition function
(two $S^3$ caps bounding $\Sigma_g$).

\medskip\noindent
\emph{(i)~The bare YM string tension: geometric component.}

The geometric contribution comes from the Havelock eigenvalue
at the WDW ground state.
The Airy function ground-state expectation value
$\langle z \rangle_{\mathrm{Ai}} = 0.457$
(the mean of the Airy distribution)
combines with the Airy scale
$a = (2c\,\coth\rho^*)^{1/3} = 4.79$
(where $c$ is the central charge and $\rho^*$ the
BO potential zero) to give:
\begin{equation}\label{eq:supp-sigma-polygon}
  \sigma_{\mathrm{polygon}}
  = \langle\lambda_{m^*}\rangle_{\mathrm{Airy}}
  = \frac{0.1015}{\ell^2}.
\end{equation}

\medskip\noindent
\emph{(ii)~The bare YM string tension: monopole (Polyakov) component.}

On $\mathbf{H}^2$ with Gaussian curvature $K = -1$,
the CS instanton has curvature-corrected action.
The correction uses the standard Seeley--DeWitt
$a_1 = R/6 = K/3 = -1/3$ contribution to the instanton
determinant:
\begin{align}
  S_{\mathrm{mon}}
  &= 2\pi k\Bigl(1 + \frac{K}{6k^2}\Bigr)
  \notag \\[4pt]
  &= 2\pi \times 1 \times
     \Bigl(1 + \frac{-1}{6 \times 1}\Bigr)
  \notag \\[4pt]
  &= 2\pi \times \frac{5}{6}
  \notag \\[4pt]
  &= \frac{5\pi}{3},
  \label{eq:supp-smon}
\end{align}
where $k = 1$ and the negative curvature reduces the barrier.
The determinant prefactor is $(5/6)^{3/2} = 0.761$,
and the dressed fugacity is $y = 0.00405$.

The Polyakov string tension (from the monopole gas on $\mathbf{H}^2$)

\begin{equation}\label{eq:supp-sigma-polyakov}
  \sigma_{\mathrm{Polyakov}}
  = \frac{N^2}{8\pi^2}\,g^4\,m\,\sqrt{y}
  = \frac{0.612}{\ell^2}.
\end{equation}

\medskip\noindent
\emph{(iii)~Combining the two monopole species.}

For $\mathrm{SU}(3)$, $\mathrm{rank} = 2$
(there are two simple roots, hence two independent monopole species,
one per simple root).
The total bare YM string tension is:
\begin{align}
  \sigma_{\mathrm{YM}}
  &= \sigma_{\mathrm{polygon}}
     + \mathrm{rank}(\mathrm{SU}(3))
       \times \sigma_{\mathrm{Polyakov}}
  \notag \\[4pt]
  &= 0.1015 + 2 \times 0.612
  \notag \\[4pt]
  &= 0.1015 + 1.224
  \notag \\[4pt]
  &= \frac{1.326}{\ell^2}.
  \label{eq:supp-sigma-ym}
\end{align}

\medskip\noindent
\emph{(iv)~The CS non-perturbative enhancement.}

The CS path integral on $\Sigma_g \times S^1$ gives the
Hilbert-space dimension via the Verlinde formula:
\[
  \dim\mathcal{H}(\Sigma_g)
  = \sum_\lambda (S_{0\lambda})^{2-2g}.
\]
For $\mathrm{SU}(3)_1$ on $\Sigma_2$ (genus~$2$):
there are three integrable representations,
each with $S_{0\lambda} = 1/\sqrt{3}$.
Thus:
\begin{align*}
  \dim\mathcal{H}
  &= 3 \times (1/\sqrt{3})^{2-4} \\
  &= 3 \times (1/\sqrt{3})^{-2} \\
  &= 3 \times 3 = 9.
\end{align*}
The $S^3$ partition function normalizes the free propagator.
For $\mathrm{SU}(3)_1$:
\[
  Z(S^3)^2 = 2.
\]
The ratio counts the number of confining vacua per unit
CS propagator weight:
\[
  \frac{\dim\mathcal{H}}{Z(S^3)^2}
  = \frac{9}{2} = 4.5.
\]

\medskip\noindent
\emph{(v)~Final result.}

Combining equations~\eqref{eq:supp-sigma-ym}
and the CS enhancement:
\begin{equation}\label{eq:supp-sigma-final}
  \frac{\sigma}{\Lambda^2}
  = 1.326 \times \frac{9}{2}
  = 1.326 \times 4.5
  = 5.97.
\end{equation}
The lattice value (pure-gauge $\mathrm{SU}(3)$):
$\sigma/\Lambda^2 = 6.25 \pm 0.5$.
The match is $5\%$ with zero free parameters.

The \emph{finite} physical string tension $5.97$ arises
from the Yang--Mills kinetic sector (the KK kinetic term),
enhanced by the factor $9/2$ from the CS non-perturbative
measure.
\end{remark}

\section{Paper~V supplements: Pell identity proof}
\label{sec:pell-proof}

\begin{proof}[Proof of the Pell identity~\eqref{eq:pell-identity}
  via the Klein quartic]
The argument has four steps.

\emph{Step~1 (The Pell geodesic on the modular surface).}
The Pell equation $8^2 - 7 \times 3^2 = 1$ defines the matrix
$M = \bigl(\begin{smallmatrix} 8 & 21 \\ 3 & 8 \end{smallmatrix}\bigr)
\in \mathrm{SL}(2,\mathbb{Z})$
with $\det M = 1$, $\operatorname{tr} M = 16$.
This matrix is hyperbolic and defines a closed geodesic on the
modular surface $\mathrm{SL}(2,\mathbb{Z})\backslash\mathbf{H}^2$
of length
$\ell = 2\operatorname{arccosh}(\operatorname{tr}M/2)
= 2\operatorname{arccosh}(8) = 2\ln\varepsilon_7$.

\emph{Step~2 (Lift to the Klein quartic).}
The level-$7$ modular curve
$X(7) = \Gamma(7)\backslash\mathbf{H}^2$
is the \emph{Klein quartic}---a genus-$3$ Riemann surface with
automorphism group $\mathrm{PSL}(2,\mathbb{F}_7)$, order~$168$.
This group contains $\mathbb{Z}/7\mathbb{Z}$ as a subgroup,
and $X(7)/(\mathbb{Z}/7\mathbb{Z})
= \mathrm{SL}(2,\mathbb{Z})\backslash\mathbf{H}^2$.
The Pell geodesic on the modular surface lifts to
\emph{seven} closed geodesics on $X(7)$, one per
$\mathbb{Z}_7$~coset, each of length~$2\ln\varepsilon_7$.

\emph{Step~3 (Selberg trace formula on $X(7)$).}
The BO potential $V_7(\rho) = \ln(2\sinh\rho) - u^*$ is the
radial effective potential of the Laplacian on~$\mathbf{H}^2$
in the $\mathbb{Z}_7$-equivariant sector.
The WKB spectral sum of this potential equals the
spectral side of the Selberg trace formula on~$X(7)$,
evaluated with the WKB test function~$h$.
On the \emph{geometric} side, each primitive geodesic of
length~$\ell$ contributes
$\ell/(2\sinh(\ell/2)) \times \hat h(\ell)$
to the trace.
For the Pell geodesic ($\ell_0 = 2\ln\varepsilon_7$):
the WKB spectral factor is
$\hat h(\ell_0) = 2\sinh(\ell_0/2) = 2\sinh(\operatorname{arccosh}8) = 6\sqrt{7}$,
which \emph{exactly cancels} the Selberg stability denominator
$1/(2\sinh(\ell_0/2))$.
We prove this cancellation explicitly.
The potential $V_7(\rho) = \ln(2\sinh\rho)$ is (up to a constant)
the radial Green's function of the Laplacian on~$\mathbf{H}^2$.
Its Legendre spectral coefficient is the resolvent
\citep[Ch.~IV]{Hejhal1976}:
\[
  \int_0^\infty \ln(2\sinh\rho)\,
  P_{-1/2+ir}(\cosh\rho)\,\sinh\rho\,d\rho
  = \frac{-\pi}{2r\sinh(\pi r)}.
\]
The WKB test function for the Selberg trace formula is
$h(r) = (1/4 + r^2) \times a(r)$, where $a(r)$ is the
spectral coefficient and $(1/4 + r^2)$ is the Laplacian eigenvalue.
The Selberg transform $\hat{h}(\ell)$ is the Abel--Fourier
transform of~$h$; for $h(r) \propto (1/4 + r^2)/(r\sinh(\pi r))$,
this evaluates via the residue formula
$\sum \operatorname{Res}[\hat{h}]$ at the poles $r = in$
($n = 0, 1, 2, \ldots$) of $1/\sinh(\pi r)$:
\[
  \hat{h}(\ell) = \frac{1}{2\pi}\int_{-\infty}^{\infty}
  e^{ir\ell}\,\frac{\pi(1/4+r^2)}{2r\sinh(\pi r)}\,dr
  = 2\sinh(\ell/2).
\]
The poles of $1/(r\sinh(\pi r))$ are at $r = 0$ and
$r = in$ ($n = 1, 2, \ldots$), with residues
$\operatorname{Res}_{r=in} = (-1)^n e^{-n\ell}/\pi$.
The factor $(1/4 + r^2) = (r+i/2)(r-i/2)$ vanishes at
$r = \pm i/2$, canceling the $n = 0$ pole and shifting the
$n \ge 1$ series to $\sum_{n=1}^\infty (-1)^n(n^2 - 1/4)
e^{-n\ell} = e^{-\ell/2} - e^{\ell/2} = -2\sinh(\ell/2)$;
with the overall sign from closing in the upper half-plane:
$\hat{h}(\ell) = 2\sinh(\ell/2)$.
This completes the proof: $\hat{h}(\ell) = 2\sinh(\ell/2)$
cancels the stability denominator $1/(2\sinh(\ell/2))$,
leaving a net contribution of~$\ell_0$ per geodesic orbit.
The net contribution per orbit is therefore simply
the \emph{geodesic length}~$\ell_0 = 2\ln\varepsilon_7$.
Summing the seven lifts:
\[
  2S_{\mathrm{BO}}(7)
  = 7 \times \ell_0 - \text{quantum corrections}
  = 14\,\ln\varepsilon_7 - \text{corrections}.
\]

\emph{Step~4 (Quantum corrections from the Dunham series).}
The bounce orbit (a single traversal of the classically
forbidden region $0 < \rho < \rho^*$) has Maslov index
$\mu = 2$: one turning point at $\rho = 0$ (hard wall,
$+1/4$ phase) and one at $\rho = \rho^*$ (soft turning point,
$+1/4$ phase), contributing $\mu/4 = 1/2$ to the WKB
quantization.
The WDW zero-point energy $\varepsilon_0$
of the log potential~(Paper~V, eq.~(5.7))
is $\varepsilon_0 = \ln[(3/4)\sqrt{\pi/\alpha}]$
in the WKB approximation; normalized by $\ln\varepsilon_7$,
the zero-point correction is $-\sqrt{2/\pi}$
(arising from the $\sqrt{\pi/\alpha}$ in the
WKB eigenvalue).
The anharmonicity of $\ln(2\sinh\rho)$ near $\rho = 0$
gives the Dunham correction $-4/c_7^2$ from the
$O(1/c)$ cubic term (Paper~V, Remark~5.2;
verified to $0.26\%$ at $50$-digit precision).
Together: $2S/\ln\varepsilon_7
= 14 - \sqrt{2/\pi} - 4/c_7^2 + O(1/c_7^3)$.
\end{proof}
